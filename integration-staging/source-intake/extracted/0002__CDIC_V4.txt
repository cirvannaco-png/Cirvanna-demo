src/cdic/common/utils.py
import warnings
from threading import Lock

_import_cache = {}
_import_lock = Lock()

def lazy_import(name: str):
with _import_lock:
if name not in _import_cache:
try:
if name == 'numpy':
import numpy as np
_import_cache[name] = np
elif name == 'numba':
import numba
_import_cache[name] = numba
elif name == 'cupy':
try:
import cupy as cp
_import_cache[name] = cp
except ImportError:
_import_cache[name] = None
elif name == 'redis':
import redis
_import_cache[name] = redis
elif name == 'msgpack':
import msgpack
_import_cache[name] = msgpack
elif name == 'cryptography':
from cryptography.hazmat.primitives import hashes, hmac
from cryptography.hazmat.primitives.kdf.hkdf import HKDF
_import_cache[name] = {'hashes': hashes, 'hmac': hmac, 'HKDF': HKDF}
elif name == 'psutil':
import psutil
_import_cache[name] = psutil
else:
import importlib
module = importlib.import_module(name)
_import_cache[name] = module
except ImportError as e:
warnings.warn(f"Failed to import {name}: {e}")
_import_cache[name] = None
return _import_cache[name]

import json
import time
from threading import Lock
from collections import deque, defaultdict

from ..common.utils import lazy_import
from ..common.caches import LRUCache

class SecureHasherHP:
def init(self, secret_key: bytes, cache_size: int = 10000):
self.secret_key = secret_key
self.cache = LRUCache(cache_size)
crypto = lazy_import('cryptography')
if crypto is None:
raise ImportError("Cryptography library not available")
self.hashes = crypto['hashes']
self.HKDF = crypto['HKDF']
self.hkdf = self.HKDF(
algorithm=self.hashes.SHA256(),
length=32,
salt=None,
info=b'cdic-hp-hash'
)
self.derived_key = self.hkdf.derive(secret_key)

class CircuitBreakerHP:
def init(self, threshold: int = 100, window: int = 60):
self.threshold = threshold
self.window = window
self.failures = deque(maxlen=threshold * 2)
self.state = "CLOSED"
self.last_state_change = time.time()
self.lock = Lock()
self.stats = {'successes': 0, 'failures': 0, 'trips': 0}
