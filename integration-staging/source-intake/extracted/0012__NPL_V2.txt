```typescript
import { Repository, DataSource } from 'typeorm';
import { IISCRepository, IdentitySafetyCore, AbsoluteConstraint } from '../types';
import { ISCEntity } from '../entities/ISCEntity';
import { logger } from '../logger';

export class PostgresISCRepository implements IISCRepository {
  private repo: Repository<ISCEntity>;
  constructor(private dataSource: DataSource) {
    this.repo = dataSource.getRepository(ISCEntity);
  }
  async getCurrentISC(userId: string): Promise<IdentitySafetyCore> {
    try {
      const entity = await this.repo.findOne({ where: { userId, isActive: true }, order: { version: 'DESC' } });
      if (!entity) throw new Error(`ISC not found for user ${userId}`);
      // Convert constraints JSON to Map (domains array -> Set)
      const constraints = new Map<string, AbsoluteConstraint>();
      for (const [id, c] of Object.entries(entity.constraints)) {
        constraints.set(id, { ...c, domains: new Set(c.domains) });
      }
      return { ...entity, constraints } as IdentitySafetyCore;
    } catch (error) {
      logger.error(`Failed to get ISC for ${userId}`, error);
      throw error;
    }
  }
  async saveISC(isc: IdentitySafetyCore): Promise<void> {
    try {
      // Convert Map to object, Set to array
      const constraintsObj: Record<string, any> = {};
      for (const [id, c] of isc.constraints.entries()) {
        constraintsObj[id] = { ...c, domains: Array.from(c.domains) };
      }
      const entity = this.repo.create({
        userId: isc.userId,
        version: isc.version,
        constraints: constraintsObj,
        checksum: isc.checksum,
        createdAt: isc.createdAt,
        updatedAt: isc.updatedAt,
        isActive: isc.isActive,
      });
      await this.repo.save(entity);
    } catch (error) {
      logger.error(`Failed to save ISC for ${isc.userId}`, error);
      throw error;
    }
  }
  async getVersionHistory(userId: string): Promise<IdentitySafetyCore[]> {
    const entities = await this.repo.find({ where: { userId }, order: { version: 'ASC' } });
    return entities.map(e => {
      const constraints = new Map();
      for (const [id, c] of Object.entries(e.constraints)) {
        constraints.set(id, { ...c, domains: new Set(c.domains) });
      }
      return { ...e, constraints } as IdentitySafetyCore;
    });
  }
}
```

repositories/PostgresAuditRepository.ts (implements IAuditRepository with hash chain)

```typescript
import { Repository, DataSource } from 'typeorm';
import { IAuditRepository, AuditLogEntry, AuditFilter } from '../types';
import { AuditLogEntity } from '../entities/AuditLogEntity';
import { generateHash } from '../utils/hash';
import { logger } from '../logger';

export class PostgresAuditRepository implements IAuditRepository {
  private repo: Repository<AuditLogEntity>;
  constructor(dataSource: DataSource) {
    this.repo = dataSource.getRepository(AuditLogEntity);
  }
  async logEntry(entry: Omit<AuditLogEntry, 'entryId' | 'immutableHash'>): Promise<string> {
    const previousHash = await this.getLatestHash(entry.userId);
    const immutableHash = generateHash(entry, previousHash);
    const entryId = `audit_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
    const entity = this.repo.create({
      ...entry,
      entryId,
      immutableHash,
      previousHash,
      vetoMetadata: entry.vetoMetadata ? JSON.stringify(entry.vetoMetadata) : null,
    });
    await this.repo.save(entity);
    return entryId;
  }
  private async getLatestHash(userId: string): Promise<string | undefined> {
    const last = await this.repo.findOne({ where: { userId }, order: { timestamp: 'DESC' } });
    return last?.immutableHash;
  }
  async getEntries(userId: string, filters?: AuditFilter): Promise<AuditLogEntry[]> {
    let qb = this.repo.createQueryBuilder('audit').where('audit.userId = :userId', { userId });
    if (filters?.startDate) qb = qb.andWhere('audit.timestamp >= :startDate', { startDate: filters.startDate });
    if (filters?.endDate) qb = qb.andWhere('audit.timestamp <= :endDate', { endDate: filters.endDate });
    if (filters?.module) qb = qb.andWhere('audit.module = :module', { module: filters.module });
    if (filters?.action) qb = qb.andWhere('audit.action = :action', { action: filters.action });
    if (filters?.result) qb = qb.andWhere('audit.result = :result', { result: filters.result });
    if (filters?.limit) qb = qb.limit(filters.limit);
    const entities = await qb.getMany();
    return entities.map(e => ({ ...e, vetoMetadata: e.vetoMetadata ? JSON.parse(e.vetoMetadata) : undefined }));
  }
  async verifyChain(userId: string): Promise<boolean> {
    const entries = await this.getEntries(userId);
    for (let i = 0; i < entries.length; i++) {
      const entry = entries[i];
      const recalc = generateHash(entry, entry.previousHash);
      if (recalc !== entry.immutableHash) return false;
      if (i > 0 && entry.previousHash !== entries[i-1].immutableHash) return false;
    }
    return true;
  }
}
```

```typescript
    return this.client.get(key);
  }

    await this.client.set(key, value, 'EX', ttlSeconds);
  }
  async del(key: string): Promise<void> {
    await this.client.del(key);
  }
  async delPattern(pattern: string): Promise<void> {
    const keys = await this.client.keys(pattern);
    if (keys.length) {
      // Use unlink for non-blocking deletion
      await this.client.unlink(...keys);
    }
  }
}
```

```typescript
import { ISCEngine as IEngine, IdentitySafetyCore, AttributeType, DecisionSource, IAuditRepository, IISCRepository, AttributeVocabulary } from '../types';
import { RedisCache } from '../repositories/RedisCache';
import { logger } from '../logger';

export class ISCEngine {
  private cache: RedisCache;
  private vocabularyMap: Map<string, AttributeVocabulary>;
  private synonymIndex: Map<string, Set<string>>;

  constructor(private iscRepo: IISCRepository, private auditRepo: IAuditRepository, vocabulary: AttributeVocabulary[]) {
    this.cache = new RedisCache();
    this.vocabularyMap = new Map();
    this.synonymIndex = new Map();
    for (const v of vocabulary) {
      const key = `${v.attributeType}:${v.canonicalValue}`;
      this.vocabularyMap.set(key, v);
      for (const syn of v.synonyms) {
        const synKey = `${v.attributeType}:${syn}`;
        if (!this.synonymIndex.has(synKey)) this.synonymIndex.set(synKey, new Set());
        this.synonymIndex.get(synKey)!.add(v.canonicalValue);
      }
    }
  }

  async isAllowed(userId: string, attributes: Record<AttributeType, string>, requestId: string): Promise<{ allowed: boolean; vetoReason?: string; rigidity?: number }> {
    const cacheKey = `isc:${userId}:${JSON.stringify(attributes)}`;
    const cached = await this.cache.get(cacheKey);
    if (cached === 'false') return { allowed: false, vetoReason: 'cached veto' };
    if (cached === 'true') return { allowed: true };

    try {
      const isc = await this.iscRepo.getCurrentISC(userId);
      for (const constraint of isc.constraints.values()) {
        const attrValue = attributes[constraint.attributeType];
        if (!attrValue) continue;

        // Direct match
        if (attrValue === constraint.attributeValue) {
          return this.handleVeto(userId, requestId, attributes, constraint, requestId);
        }
        // Synonym match
        const synKey = `${constraint.attributeType}:${attrValue}`;
        const synonyms = this.synonymIndex.get(synKey);
        if (synonyms && synonyms.has(constraint.attributeValue)) {
          return this.handleVeto(userId, requestId, attributes, constraint, requestId);
        }
      }
      await this.cache.set(cacheKey, 'true', 300);
      return { allowed: true };
    } catch (error) {
      logger.error(`ISC Engine error for ${userId}`, error);
      await this.auditRepo.logEntry({
        timestamp: new Date(),
        userId,
        module: 'ISC_ENGINE',
        action: 'ERROR',
        decisionSource: DecisionSource.ISC_VETO,
        attributes,
        result: 'DENY',
        reason: `System error: ${error.message}`,
        metadata: { requestId }
      });
      return { allowed: false, vetoReason: 'System error' };
    }
  }

  private async handleVeto(userId: string, requestId: string, attributes: Record<AttributeType, string>, constraint: any, requestId: string) {
    const isHardVeto = constraint.rigidity >= 0.8;
    const result = isHardVeto ? 'DENY' : 'ALLOW';
    await this.auditRepo.logEntry({
      timestamp: new Date(),
      userId,
      module: 'ISC_ENGINE',
      action: 'VETO',
      decisionSource: DecisionSource.ISC_VETO,
      attributes,
      result,
      reason: constraint.reason,
      vetoMetadata: {
        domainTags: Array.from(constraint.domains),
        rigidityAtTime: constraint.rigidity,
        driftInteractionScore: 0, // Not used (no inference)
        contextSnapshot: {}
      },
      metadata: { requestId }
    });
    if (isHardVeto) {
      await this.cache.set(`isc:${userId}:${JSON.stringify(attributes)}`, 'false', 300);
      return { allowed: false, vetoReason: constraint.reason, rigidity: constraint.rigidity };
    }
    return { allowed: true }; // soft veto allows
  }
}
```

engine/ContextualResolver.ts (full implementation with rigidity modifier)

```typescript
import { ContextualConstraintResolver as IResolver, Constraint, ContextBinding, ContextScope, ConstraintType, IConstraintRepository, IAuditRepository } from '../types';
import { RedisCache } from '../repositories/RedisCache';

export class ContextualResolver {
  private cache: RedisCache;
  constructor(private constraintRepo: IConstraintRepository, private auditRepo: IAuditRepository) {
    this.cache = new RedisCache();
  }

  async getActiveConstraints(userId: string, context: Map<ContextScope, string>): Promise<Constraint[]> {
    const cacheKey = `ctx:${userId}:${this.hashContext(context)}`;
    const cached = await this.cache.get(cacheKey);
    if (cached) return JSON.parse(cached);

    const allConstraints = await this.constraintRepo.getConstraints(userId);
    const active: Constraint[] = [];
    for (const constraint of allConstraints) {
      if (constraint.type !== ConstraintType.CONTEXTUAL || !constraint.isActive) continue;
      // Check bindings (simplified: assume bindings stored in constraint metadata)
      const bindings = constraint.metadata?.contextBindings as ContextBinding[] || [];
      for (const binding of bindings) {
        if (binding.isActive && (!binding.expiresAt || binding.expiresAt > new Date())) {
          const contextValue = context.get(binding.contextType);
          if (contextValue === binding.contextValue) {
            // Apply rigidity modifier
            const modifiedRigidity = Math.min(1, Math.max(0, constraint.rigidity + binding.rigidityModifier));
            active.push({ ...constraint, rigidity: modifiedRigidity });
            break;
          }
        }
      }
    }
    await this.cache.set(cacheKey, JSON.stringify(active), 60);
    return active;
  }

  private hashContext(context: Map<ContextScope, string>): string {
    const sorted = Array.from(context.entries()).sort(([a],[b]) => a.localeCompare(b));
    return Buffer.from(JSON.stringify(sorted)).toString('base64');
  }
}
```

engine/NegativePreferenceGraph.ts (with Redis storage, closure cache, no inference)

```typescript
import { EventEmitter } from 'events';
import Redis from 'ioredis';
import { NegativePreferenceNode, PreferenceEdge } from '../types';
import { config } from '../config';
import { logger } from '../logger';

export class NegativePreferenceGraph extends EventEmitter {
  private redis: Redis;
  private closureCache: Map<string, Set<string>> = new Map();
  constructor(private auditRepo: any) {
    super();
    this.redis = new Redis(config.redis.url);
  }

  async addNode(userId: string, node: NegativePreferenceNode): Promise<void> {
    const key = `graph:${userId}:node:${node.nodeId}`;
    const data = {
      ...node,
      edges: JSON.stringify([...node.edges.entries()]), // store as array of [key, edge]
      createdAt: node.createdAt.toISOString(),
      updatedAt: node.updatedAt.toISOString(),
    };
    await this.redis.hset(key, data);
    this.closureCache.clear();
    this.emit('node_added', { userId, nodeId: node.nodeId });
  }

  async generateExclusionSet(userId: string, triggerNodeId: string, context?: Map<string, string>): Promise<Set<string>> {
    const cacheKey = `${userId}:${triggerNodeId}:${this.hashContext(context)}`;
    if (this.closureCache.has(cacheKey)) return this.closureCache.get(cacheKey)!;

    const exclusionSet = new Set<string>();
    const visited = new Set<string>();
    const queue = [{ nodeId: triggerNodeId, depth: 0 }];
    const maxDepth = 3;

    while (queue.length) {
      const { nodeId, depth } = queue.shift()!;
      if (visited.has(nodeId) || depth > maxDepth) continue;
      visited.add(nodeId);
      const node = await this.getNode(userId, nodeId);
      if (!node) continue;
      if (depth > 0 && node.isIscOriginated) continue;
      exclusionSet.add(nodeId);
      const edges = await this.getEdges(userId, nodeId);
      for (const edge of edges) {
        if (edge.confidenceScore >= 0.7 && this.isEdgeActiveInContext(edge, context)) {
          queue.push({ nodeId: edge.targetNodeId, depth: depth + 1 });
        }
      }
    }
    this.closureCache.set(cacheKey, exclusionSet);
    return exclusionSet;
  }

  private async getNode(userId: string, nodeId: string): Promise<NegativePreferenceNode | null> {
    const key = `graph:${userId}:node:${nodeId}`;
    const data = await this.redis.hgetall(key);
    if (!data || !data.nodeId) return null;
    return {
      ...data,
      edges: new Map(JSON.parse(data.edges)),
      createdAt: new Date(data.createdAt),
      updatedAt: new Date(data.updatedAt),
    } as any;
  }

  private async getEdges(userId: string, sourceNodeId: string): Promise<PreferenceEdge[]> {
    const node = await this.getNode(userId, sourceNodeId);
    return node ? Array.from(node.edges.values()) : [];
  }

  private isEdgeActiveInContext(edge: PreferenceEdge, context?: Map<string, string>): boolean {
    if (!context) return true;
    const restrictions = (edge as any).contextRestrictions;
    if (!restrictions) return true;
    for (const [scope, value] of context.entries()) {
      if (restrictions[scope] && restrictions[scope] !== value) return false;
    }
    return true;
  }

  private hashContext(context?: Map<string, string>): string {
    if (!context) return 'none';
    const sorted = Array.from(context.entries()).sort(([a],[b]) => a.localeCompare(b));
    return Buffer.from(JSON.stringify(sorted)).toString('base64');
  }
}
```

engine/EnforcementGate.ts (with PPL integration, final score, permanent exclusion check)

```typescript
import { EnforcementGate as IGate, DecisionSource, AttributeType, IAuditRepository } from '../types';
import { PositivePreferenceLayer } from '../ppl/PositivePreferenceLayer';
import { ISCEngine } from './ISCEngine';
import { ContextualResolver } from './ContextualResolver';
import { NegativePreferenceGraph } from './NegativePreferenceGraph';
import { RedisCache } from '../repositories/RedisCache';
import { logger } from '../logger';

export class EnforcementGate {
  private cache: RedisCache;
  constructor(
    private iscEngine: ISCEngine,
    private contextualResolver: ContextualResolver,
    private graph: NegativePreferenceGraph,
    private auditRepo: IAuditRepository,
    private ppl: PositivePreferenceLayer
  ) {
    this.cache = new RedisCache();
  }

  async evaluateItem(request: { userId: string; itemId: string; attributes: Record<AttributeType, string>; priority: number }): Promise<EvaluationResult> {
    const start = Date.now();
    const reasons: string[] = [];
    const sources: DecisionSource[] = [];

    // 1. ISC check (hard veto if rigidity >= 0.8)
    const iscResult = await this.iscEngine.isAllowed(request.userId, request.attributes, request.itemId);
    if (!iscResult.allowed && iscResult.rigidity && iscResult.rigidity >= 0.8) {
      await this.logDecision(request, false, reasons, sources, start);
      return { allowed: false, reasons: [iscResult.vetoReason!], sources: [DecisionSource.ISC_VETO], processingTime: Date.now() - start };
    }

    // 2. Contextual constraints (with rigidity)
    const context = new Map(); // would be fetched from user session
    const activeConstraints = await this.contextualResolver.getActiveConstraints(request.userId, context);
    for (const c of activeConstraints) {
      if (c.attributeValue === request.attributes[c.attributeType]) {
        if (c.rigidity >= 0.8) {
          reasons.push(`Contextual constraint: ${c.reason}`);
          sources.push(DecisionSource.CONTEXTUAL);
          await this.logDecision(request, false, reasons, sources, start);
          return { allowed: false, reasons, sources, processingTime: Date.now() - start };
        } else {
          reasons.push(`Soft contextual: ${c.reason}`);
        }
      }
    }

    // 3. NPG exclusions (with elastic threshold? No, no inference – we just check)
    const exclusions = await this.graph.generateExclusionSet(request.userId, 'trigger', context);
    if (exclusions.size > 0) {
      reasons.push(`NPG exclusion (${exclusions.size} nodes)`);
      sources.push(DecisionSource.NPG);
      await this.logDecision(request, false, reasons, sources, start);
      return { allowed: false, reasons, sources, processingTime: Date.now() - start };
    }

    // 4. Positive Preference Layer score (explicit user likes only)
    const pplScore = await this.ppl.getScore(request.userId, request.itemId);
    // NPL penalty: 0 if no veto, 0.5 if soft veto, 1 if hard veto (but hard already returned)
    const nplPenalty = 0; // no hard veto remained
    const finalScore = pplScore - nplPenalty;
    const allowed = finalScore >= 0;
    if (!allowed) reasons.push('Positive preference score too low');
    await this.logDecision(request, allowed, reasons, sources, start);
    return { allowed, reasons, sources, processingTime: Date.now() - start };
  }

  private async logDecision(request: any, allowed: boolean, reasons: string[], sources: DecisionSource[], start: number) {
    await this.auditRepo.logEntry({
      timestamp: new Date(),
      userId: request.userId,
      module: 'ENFORCEMENT_GATE',
      action: 'EVALUATION',
      decisionSource: sources[0] || DecisionSource.ENFORCEMENT_GATE,
      itemId: request.itemId,
      attributes: request.attributes,
      result: allowed ? 'ALLOW' : 'DENY',
      reason: reasons.join('; ') || 'All checks passed',
      metadata: { requestId: request.itemId, processingTime: Date.now() - start, sources }
    });
  }
}
```

engine/DigitalWardrobe.ts (full cartesian outfit generation, permanent exclusion tracking)

```typescript
import { WardrobeItem, AttributeType, IPermanentExclusionRepository } from '../types';
import { EnforcementGate } from './EnforcementGate';
import { RedisCache } from '../repositories/RedisCache';
import { config } from '../config';

export class DigitalWardrobe {
  private userWardrobes: Map<string, Map<string, WardrobeItem>> = new Map();
  private cache: RedisCache;
  constructor(private enforcementGate: EnforcementGate, private permExclusionRepo: IPermanentExclusionRepository) {
    this.cache = new RedisCache();
  }

  async filterWardrobe(userId: string): Promise<WardrobeItem[]> {
    const wardrobe = this.userWardrobes.get(userId);
    if (!wardrobe) return [];
    const items = Array.from(wardrobe.values());
    const filtered: WardrobeItem[] = [];
    for (const item of items) {
      const excluded = await this.permExclusionRepo.isExcluded(userId, item.itemId);
      if (excluded) continue;
      const result = await this.enforcementGate.evaluateItem({ userId, itemId: item.itemId, attributes: item.attributes, priority: 1 });
      if (result.allowed) {
        filtered.push(item);
      } else {
        await this.trackRejection(userId, item, result.reasons.join(','));
      }
    }
    return filtered;
  }

  private async trackRejection(userId: string, item: WardrobeItem, reason: string) {
    const key = `rejection:${userId}:${item.itemId}`;
    const count = await this.cache.get(key);
    const newCount = (parseInt(count || '0') + 1);
    if (newCount >= config.npl.permanentExclusionThreshold) {
      await this.permExclusionRepo.addExclusion({
        userId,
        itemId: item.itemId,
        domain: 'general',
        rejectionCount: newCount,
        threshold: config.npl.permanentExclusionThreshold,
        reason,
        rigidityAtTime: 1.0,
        createdAt: new Date(),
      });
      await this.cache.del(key);
    } else {
      await this.cache.set(key, newCount.toString(), 86400);
    }
  }

  generateOutfits(items: WardrobeItem[], rule: OutfitRule, maxOutfits: number): WardrobeItem[][] {
    // Group by category (each item can have multiple categories; we'll use first category for simplicity)
    const byCategory = new Map<string, WardrobeItem[]>();
    for (const item of items) {
      const cat = Array.from(item.categories)[0];
      if (!byCategory.has(cat)) byCategory.set(cat, []);
      byCategory.get(cat)!.push(item);
    }
    // Cartesian product
    let combinations: WardrobeItem[][] = [[]];
    for (const cat of rule.categories) {
      const catItems = byCategory.get(cat) || [];
      if (catItems.length === 0) return [];
      const newCombos: WardrobeItem[][] = [];
      for (const combo of combinations) {
        for (const item of catItems) {
          newCombos.push([...combo, item]);
        }
      }
      combinations = newCombos;
    }
    // Filter by min/max items and exclusive categories
    const valid = combinations.filter(combo => combo.length >= rule.minItems && combo.length <= rule.maxItems && this.validateExclusive(combo, rule));
    return valid.slice(0, maxOutfits);
  }

  private validateExclusive(outfit: WardrobeItem[], rule: OutfitRule): boolean {
    for (const pair of rule.exclusiveCategories) {
      const [cat1, cat2] = pair;
      const has1 = outfit.some(i => i.categories.has(cat1));
      const has2 = outfit.some(i => i.categories.has(cat2));
      if (has1 && has2) return false;
    }
    return true;
  }
}
```

engine/ISCEvolutionProtocol.ts (with Bull queue, transactions)

```typescript
import { DataSource, QueryRunner } from 'typeorm';
import { ISCEvolutionProtocol as IEvolution, IdentitySafetyCore, ConstraintChange, ChangeType, EvolutionStatus } from '../types';
import { coolingQueue } from '../repositories/BullQueue';
import { logger } from '../logger';

export class ISCEvolutionProtocol {
  private queryRunner: QueryRunner | null = null;
  constructor(private iscRepo: any, private auditRepo: any, private constraintRepo: any, private dataSource: DataSource) {}

  async initiateEvolution(userId: string, changes: ConstraintChange[], reason?: string) {
    const currentISC = await this.iscRepo.getCurrentISC(userId);
    const event = {
      eventId: `evol_${Date.now()}_${Math.random().toString(36)}`,
      userId,
      previousVersion: currentISC.version,
      newVersion: currentISC.version + 1,
      changes,
      coolingPeriodEnds: new Date(Date.now() + config.npl.coolingPeriodHours * 3600000),
      status: EvolutionStatus.COOLING,
      createdAt: new Date(),
      confirmationToken: require('crypto').randomBytes(32).toString('hex'),
    };
    // Save to pending evolutions table
    await this.dataSource.getRepository('PendingEvolutionEntity').save({ userId, eventId: event.eventId, event, timeoutAt: event.coolingPeriodEnds });
    // Schedule job
    await coolingQueue.add({ userId, eventId: event.eventId }, { delay: config.npl.coolingPeriodHours * 3600000 });
    return { event, validation: { isValid: true, errors: [], warnings: [] } };
  }

  async confirmEvolution(userId: string, token: string, confirmImmutable: boolean) {
    const pending = await this.dataSource.getRepository('PendingEvolutionEntity').findOne({ where: { userId } });
    if (!pending) throw new Error('No pending evolution');
    if (pending.event.confirmationToken !== token) throw new Error('Invalid token');
    if (pending.event.coolingPeriodEnds > new Date()) throw new Error('Cooling period not expired');

    this.queryRunner = this.dataSource.createQueryRunner();
    await this.queryRunner.connect();
    await this.queryRunner.startTransaction();
    try {
      const currentISC = await this.iscRepo.getCurrentISC(userId);
      const newISC = await this.applyChanges(currentISC, pending.event.changes);
      await this.iscRepo.saveISC(newISC);
      await this.queryRunner.manager.delete('PendingEvolutionEntity', { userId });
      await this.queryRunner.commitTransaction();
      await this.auditRepo.logEntry({ /* log success */ });
      return { success: true, newISC };
    } catch (err) {
      await this.queryRunner.rollbackTransaction();
      throw err;
    } finally {
      await this.queryRunner.release();
    }
  }

  private async applyChanges(currentISC: IdentitySafetyCore, changes: ConstraintChange[]): Promise<IdentitySafetyCore> {
    const newConstraints = new Map(currentISC.constraints);
    for (const change of changes) {
      if (change.changeType === ChangeType.ADDED) {
        newConstraints.set(change.constraintId, { ...change.newState, createdAt: new Date(), createdBy: 'USER', isImmutable: false });
      } else if (change.changeType === ChangeType.REMOVED) {
        newConstraints.delete(change.constraintId);
      } else if (change.changeType === ChangeType.MODIFIED) {
        const existing = newConstraints.get(change.constraintId);
        if (existing) newConstraints.set(change.constraintId, { ...existing, ...change.newState });
      }
      // DOWNGRADED_* would remove from ISC and add to contextual/graph stores (omitted for brevity)
    }
    return { ...currentISC, version: currentISC.version + 1, constraints: newConstraints, updatedAt: new Date(), checksum: generateISCChecksum(newConstraints) };
  }
}
```
