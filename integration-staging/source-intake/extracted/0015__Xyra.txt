"""
Enhanced connection management with security and connection pooling.
"""
import asyncio
import time
from collections import deque
from contextlib import asynccontextmanager
import redis.asyncio as redis
from databases import Database
import aiohttp
from .config import XyraConfig

class SecureConnectionPool:
    _instance = None
    _initialized = False

    def __init__(self, config: XyraConfig):
        self.config = config
        self.redis_pool = None
        self.db_pool = None
        self.http_session = None
        self.model_cache = {}
        self._lock = asyncio.Lock()
        self._redis_connections = deque(maxlen=config.DB_POOL_SIZE)
        self._db_connections = deque(maxlen=config.DB_POOL_SIZE)
        self._connection_semaphore = asyncio.Semaphore(config.DB_POOL_SIZE)
        self._rate_limiter = {}

    @classmethod
    def get_instance(cls, config: XyraConfig = None):
        if cls._instance is None:
            with asyncio.Lock():
                if cls._instance is None and config is not None:
                    cls._instance = cls(config)
        return cls._instance

    async def initialize(self):
        if self._initialized:
            return
        async with self._lock:
            try:
                self.redis_pool = await redis.Redis.from_url(
                    "redis://localhost:6379",
                    max_connections=self.config.DB_POOL_SIZE,
                    decode_responses=False,
                    ssl=False,
                    health_check_interval=30,
                )
                self.db_pool = Database(
                    "postgresql://user:pass@localhost/xyra",
                    min_size=5,
                    max_size=self.config.DB_POOL_SIZE,
                    ssl=False,
                )
                await self.db_pool.connect()
                timeout = aiohttp.ClientTimeout(total=2.0)
                connector = aiohttp.TCPConnector(limit=100, ssl=False)
                self.http_session = aiohttp.ClientSession(
                    timeout=timeout,
                    connector=connector,
                    headers={"User-Agent": "Xyra-System/1.0"}
                )
                await self._warm_connections()
                self._initialized = True
            except Exception as e:
                raise

    async def _warm_connections(self):
        tasks = []
        for _ in range(5):
            tasks.append(self._create_redis_connection())
        await asyncio.gather(*tasks, return_exceptions=True)

    async def _create_redis_connection(self):
        conn = await self.redis_pool.connection()
        self._redis_connections.append(conn)
        return conn

    @asynccontextmanager
    async def get_redis(self):
        async with self._connection_semaphore:
            if self._redis_connections:
                conn = self._redis_connections.pop()
            else:
                conn = await self._create_redis_connection()
            try:
                yield conn
            finally:
                self._redis_connections.append(conn)

    @asynccontextmanager
    async def get_db(self):
        yield self.db_pool

    async def check_rate_limit(self, user_id: str, endpoint: str) -> bool:
        key = f"rate_limit:{user_id}:{endpoint}"
        now = time.time()
        window = self.config.RATE_LIMIT_WINDOW
        max_requests = self.config.RATE_LIMIT_REQUESTS
        async with self.get_redis() as redis_conn:
            # Use sorted set to track timestamps
            await redis_conn.zadd(key, {str(now): now})
            await redis_conn.zremrangebyscore(key, 0, now - window)
            count = await redis_conn.zcard(key)
            await redis_conn.expire(key, window)
            return count <= max_requests

    async def close(self):
        async with self._lock:
            if self.redis_pool:
                await self.redis_pool.close()
            if self.db_pool:
                await self.db_pool.disconnect()
            if self.http_session:
                await self.http_session.close()
            self._initialized = False

"""
Intelligent caching with prefetching and access pattern learning.
"""
import asyncio
from collections import OrderedDict, deque
import orjson
import mmh3

class PredictiveCache:
    def __init__(self, config):
        self.config = config
        self.local_cache = OrderedDict()
        self.access_patterns = {}
        self.prefetch_queue = asyncio.Queue()
        self.hits = 0
        self.misses = 0
        self.prefetch_hits = 0
        self._prefetch_worker = asyncio.create_task(self._prefetch_worker_task())

    async def _prefetch_worker_task(self):
        while True:
            try:
                key, redis_pool = await self.prefetch_queue.get()
                await self._prefetch_related(key, redis_pool)
                self.prefetch_queue.task_done()
            except asyncio.CancelledError:
                break
            except Exception:
                pass

    def get_local(self, key):
        if key in self.local_cache:
            self.hits += 1
            value = self.local_cache.pop(key)
            self.local_cache[key] = value
            return value
        self.misses += 1
        return None

    def set_local(self, key, value):
        if key in self.local_cache:
            self.local_cache.move_to_end(key)
        else:
            if len(self.local_cache) >= self.config.MAX_CACHE_SIZE:
                self.local_cache.popitem(last=False)
            self.local_cache[key] = value

    async def get_distributed(self, redis_pool, key):
        if key in self.access_patterns:
            self.prefetch_hits += 1
        async with redis_pool.pipeline() as pipe:
            await pipe.get(key)
            await pipe.expire(key, self.config.CACHE_TTL_SECONDS)
            result, _ = await pipe.execute()
        if result:
            try:
                if result.startswith(b'compressed:'):
                    import zlib
                    result = zlib.decompress(result[11:])
                value = orjson.loads(result)
                self.set_local(key, value)
                await self.prefetch_queue.put((key, redis_pool))
                return value
            except Exception:
                pass
        return None

    async def set_distributed(self, redis_pool, key, value, compress=True):
        serialized = orjson.dumps(value)
        if compress and len(serialized) > 1000:
            import zlib
            compressed = zlib.compress(serialized)
            if len(compressed) < len(serialized) * 0.7:
                serialized = b'compressed:' + compressed
        async with redis_pool.pipeline() as pipe:
            await pipe.set(key, serialized, ex=self.config.CACHE_TTL_SECONDS)
            await pipe.execute()
        self.set_local(key, value)

    async def _prefetch_related(self, key, redis_pool):
        if key not in self.access_patterns:
            return
        related = self.access_patterns[key].get('related', [])
        for related_key in related[:5]:
            try:
                await redis_pool.get(related_key)
            except Exception:
                pass

    def record_access(self, key, next_key=None):
        if key not in self.access_patterns:
            self.access_patterns[key] = {'count': 0, 'related': {}}
        self.access_patterns[key]['count'] += 1
        if next_key:
            if next_key not in self.access_patterns[key]['related']:
                self.access_patterns[key]['related'][next_key] = 0
            self.access_patterns[key]['related'][next_key] += 1
            related_items = self.access_patterns[key]['related']
            if len(related_items) > 20:
                sorted_items = sorted(related_items.items(), key=lambda x: x[1], reverse=True)
                self.access_patterns[key]['related'] = dict(sorted_items[:15])

    @staticmethod
    def generate_key(prefix, *args):
        key_string = ":".join(str(arg) for arg in args)
        hash_int = mmh3.hash(key_string, signed=False)
        return f"{prefix}:{hash_int:x}"

    def get_stats(self):
        return {
            "local_cache_size": len(self.local_cache),
            "hits": self.hits,
            "misses": self.misses,
            "hit_rate": self.hits / max(self.hits + self.misses, 1),
            "prefetch_hits": self.prefetch_hits,
            "access_patterns": len(self.access_patterns),
        }

"""
Privacy-first user data management with anonymization, encryption, and differential privacy.
"""
import hashlib
import time
from datetime import datetime
import orjson
import numpy as np

class PrivacyManager:
    def __init__(self, config):
        self.config = config
        self.anonymization_salt = b"xyra_privacy_salt_2026"

    def anonymize_user_id(self, user_id):
        if not self.config.ANONYMIZE_USER_IDS:
            return user_id
        anonymized = hashlib.blake2b(
            user_id.encode(),
            key=self.anonymization_salt,
            digest_size=16
        ).hexdigest()
        return anonymized

    def encrypt_sensitive_data(self, data):
        if not self.config.ENCRYPT_SENSITIVE_DATA:
            return orjson.dumps(data)
        serialized = orjson.dumps(data)
        encrypted = self.config.cipher.encrypt(serialized)
        return encrypted

    def decrypt_sensitive_data(self, encrypted_data):
        if not self.config.ENCRYPT_SENSITIVE_DATA:
            return orjson.loads(encrypted_data)
        decrypted = self.config.cipher.decrypt(encrypted_data)
        return orjson.loads(decrypted)

    def add_differential_privacy_noise(self, data, epsilon=None):
        if not self.config.ENABLE_DIFFERENTIAL_PRIVACY:
            return data
        epsilon = epsilon or self.config.DIFFERENTIAL_PRIVACY_EPSILON
        sensitivity = 1.0
        scale = sensitivity / epsilon
        noise = np.random.laplace(0, scale, data.shape)
        return data + noise

    def generate_audit_log(self, user_id, action, data):
        anonymized_id = self.anonymize_user_id(user_id)
        log_entry = {
            "timestamp": datetime.now().isoformat(),
            "anonymized_user_id": anonymized_id,
            "action": action,
            "data_type": type(data).__name__,
            "data_size": len(str(data)),
            "retention_days": self.config.DATA_RETENTION_DAYS,
        }
        log_entry["data_hash"] = hashlib.sha256(
            orjson.dumps(data)
        ).hexdigest()[:16]
        return log_entry

"""
Adaptive batch processing with dynamic optimization.
"""
import asyncio
import time
from collections import deque
import numpy as np

class AdaptiveBatchProcessor:
    def __init__(self, config):
        self.config = config
        self.batch_size = config.VECTOR_SIMILARITY_BATCH_SIZE
        self.batch_size_history = deque(maxlen=100)
        self.processing_times = deque(maxlen=100)
        self.optimal_batch_size = config.VECTOR_SIMILARITY_BATCH_SIZE
        self.thresholds = {
            "fast": 10,
            "optimal": 50,
            "slow": 100
        }

    async def process_batch(self, items, processor):
        start_time = time.perf_counter()
        if len(items) <= self.batch_size:
            result = await processor(items)
            elapsed = (time.perf_counter() - start_time) * 1000
            self.record_processing_time(elapsed, len(items))
            self.adapt_batch_size()
            return result
        chunk_size = self.optimal_batch_size
        chunks = [
            items[i:i + chunk_size]
            for i in range(0, len(items), chunk_size)
        ]
        semaphore = asyncio.Semaphore(self.config.MAX_CONCURRENT_ML)
        async def process_with_limit(chunk):
            async with semaphore:
                return await processor(chunk)
        tasks = [process_with_limit(chunk) for chunk in chunks]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        combined = []
        for i, result in enumerate(results):
            if isinstance(result, Exception):
                try:
                    fallback_result = await processor(chunks[i])
                    combined.extend(fallback_result)
                except Exception:
                    pass
                continue
            combined.extend(result)
        elapsed = (time.perf_counter() - start_time) * 1000
        self.record_processing_time(elapsed, len(items))
        self.adapt_batch_size()
        return combined

    def record_processing_time(self, elapsed_ms, batch_size):
        self.processing_times.append(elapsed_ms)
        self.batch_size_history.append(batch_size)

    def adapt_batch_size(self):
        if len(self.processing_times) < 5:
            return
        recent_times = list(self.processing_times)[-5:]
        recent_sizes = list(self.batch_size_history)[-5:]
        avg_time = np.mean(recent_times)
        avg_size = np.mean(recent_sizes)
        if avg_time < self.thresholds["fast"] and avg_size == self.optimal_batch_size:
            self.optimal_batch_size = min(
                self.config.MAX_BATCH_SIZE,
                int(self.optimal_batch_size * 1.5)
            )
        elif avg_time > self.thresholds["slow"]:
            self.optimal_batch_size = max(
                10,
                int(self.optimal_batch_size * 0.7)
            )

    def get_metrics(self):
        return {
            "optimal_batch_size": self.optimal_batch_size,
            "avg_processing_time": np.mean(self.processing_times) if self.processing_times else 0,
            "total_batches_processed": len(self.processing_times),
        }

"""
Real-time performance monitoring and optimization.
"""
import time
from collections import deque
import numpy as np

class RealTimeOptimizer:
    def __init__(self, config):
        self.config = config
        self.metrics_buffer = deque(maxlen=1000)
        self.optimization_history = []
        self.last_optimization = time.time()
        self.optimization_cooldown = 60
        self.thresholds = {
            "p95_response_ms": config.P95_RESPONSE_MS,
            "error_rate": 0.01,
            "cache_hit_rate": 0.7,
            "memory_usage_percent": 80,
        }
        self.system_state = {
            "mode": config.MODE,
            "ml_timeout": config.ML_TIMEOUT_MS,
            "cache_ttl": config.CACHE_TTL_SECONDS,
            "batch_size": config.VECTOR_SIMILARITY_BATCH_SIZE,
        }

    def record_metrics(self, metrics):
        self.metrics_buffer.append({
            **metrics,
            "timestamp": time.time()
        })
        if time.time() - self.last_optimization > self.optimization_cooldown:
            self.analyze_and_optimize()

    def analyze_and_optimize(self):
        if len(self.metrics_buffer) < 100:
            return
        recent_metrics = list(self.metrics_buffer)[-100:]
        p95 = np.percentile([m.get('latency', 0) for m in recent_metrics], 95)
        error_rate = np.mean([m.get('error_rate', 0) for m in recent_metrics])
        cache_hit_rate = np.mean([m.get('cache_hit_rate', 0) for m in recent_metrics])
        optimizations = []
        if p95 > self.thresholds["p95_response_ms"]:
            optimizations.append(self._apply_latency_optimizations(p95))
        if error_rate > self.thresholds["error_rate"]:
            optimizations.append(self._apply_error_optimizations(error_rate))
        if cache_hit_rate < self.thresholds["cache_hit_rate"]:
            optimizations.append(self._apply_cache_optimizations(cache_hit_rate))
        if optimizations:
            self.last_optimization = time.time()

    def _apply_latency_optimizations(self, current_p95):
        optimizations = {}
        new_ml_timeout = max(20, self.system_state["ml_timeout"] - 10)
        if new_ml_timeout != self.system_state["ml_timeout"]:
            self.system_state["ml_timeout"] = new_ml_timeout
            self.config.ML_TIMEOUT_MS = new_ml_timeout
            optimizations["ml_timeout_reduced"] = new_ml_timeout
        new_cache_ttl = min(3600, self.system_state["cache_ttl"] + 60)
        if new_cache_ttl != self.system_state["cache_ttl"]:
            self.system_state["cache_ttl"] = new_cache_ttl
            self.config.CACHE_TTL_SECONDS = new_cache_ttl
            optimizations["cache_ttl_increased"] = new_cache_ttl
        if current_p95 > self.config.MAX_RESPONSE_MS:
            self.system_state["mode"] = "mvp"
            self.config.MODE = "mvp"
            optimizations["mode_switched"] = "mvp"
        return {"type": "latency", "actions": optimizations}

    def _apply_error_optimizations(self, error_rate):
        optimizations = {}
        self.config.ENABLE_CIRCUIT_BREAKERS = True
        new_batch_size = max(10, int(self.system_state["batch_size"] * 0.8))
        if new_batch_size != self.system_state["batch_size"]:
            self.system_state["batch_size"] = new_batch_size
            self.config.VECTOR_SIMILARITY_BATCH_SIZE = new_batch_size
            optimizations["batch_size_reduced"] = new_batch_size
        return {"type": "error_rate", "actions": optimizations}

    def _apply_cache_optimizations(self, cache_hit_rate):
        optimizations = {}
        new_ttl = min(3600, self.system_state["cache_ttl"] + 120)
        if new_ttl != self.system_state["cache_ttl"]:
            self.system_state["cache_ttl"] = new_ttl
            self.config.CACHE_TTL_SECONDS = new_ttl
            optimizations["cache_ttl_increased"] = new_ttl
        if not self.config.ENABLE_PREDICTIVE_LOADING:
            self.config.ENABLE_PREDICTIVE_LOADING = True
            optimizations["predictive_loading_enabled"] = True
        return {"type": "cache", "actions": optimizations}

    def get_system_state(self):
        return {
            **self.system_state,
            "buffer_size": len(self.metrics_buffer),
            "optimizations_applied": len(self.optimization_history),
            "last_optimization": self.last_optimization,
        }

                "makeup_zones": face_results.get("face_analysis", {}).get("makeup_zones", {})
            })
        if body_results:
            recs = body_results.get("recommendations", {})
            extra.update({
                "body_measurements": recs.get("measurements", {}),
                "recommended_size": recs.get("recommended_size", "unknown")
            })

        explanation = self.explanation_generator.generate(
            recommendations=recommendations or [],
            tier=tier,
            context=context,
            extra=extra
        )
        modulated = self.emotional_modulator.modulate(
            explanation,
            user_mood=self._detect_mood(query, history),
            context=context,
            user_embedding=user_embedding
        )
        dialogue = self._assemble_dialogue(
            intent=intent,
            tier=tier,
            explanation=modulated,
            recommendations=recommendations[:3] if recommendations else [],
            context=context
        )
        await self.memory_manager.store_interaction(
            user_id=user_id,
            query=query,
            response=dialogue,
            tier=tier,
            intent=intent
        )
        elapsed = (time.perf_counter() - start_time) * 1000
        return {
            "dialogue": dialogue,
            "metadata": {
                "tier": tier,
                "intent": intent,
                "confidence": confidence,
                "generation_time_ms": elapsed,
                "history_length": len(history),
                "emotional_tone": modulated.get("tone", "neutral")
            },
            "recommendations": recommendations[:3] if recommendations else []
        }

    def _determine_tier(self, user_id, intent, confidence, history, user_embedding=None):
        tier = "summary"
        if history:
            recent_tiers = [h.get("tier", "summary") for h in history[-5:]]
            if "scientific" in recent_tiers:
                tier = "scientific"
            elif "detailed" in recent_tiers:
                tier = "detailed"
        if intent == "explanation" and confidence > 0.8:
            tier = "detailed"
        elif intent == "scientific_inquiry":
            tier = "scientific"
        # Personalisation via user embedding (simplified)
        if user_embedding is not None and hasattr(self, 'persona_classifier'):
            persona = self.persona_classifier(user_embedding)
            if persona == "scientific":
                tier = "scientific"
            elif persona == "detailed" and tier == "summary":
                tier = "detailed"
        return tier

    def _detect_mood(self, query, history):
        query_lower = query.lower()
        positive_words = ["great", "good", "happy", "excited", "thanks", "thank you"]
        negative_words = ["bad", "worried", "concerned", "frustrated", "disappointed"]
        if any(word in query_lower for word in positive_words):
            return "positive"
        elif any(word in query_lower for word in negative_words):
            return "concerned"
        return "neutral"

    def _assemble_dialogue(self, intent, tier, explanation, recommendations, context):
        template = self.templates.get(intent, {}).get(tier, "")
        if not template and intent in self.templates:
            template = self.templates[intent].get("summary", "")
        filled_template = template.format(**explanation.get("template_data", {}))
        return {
            "text": filled_template,
            "tone": explanation.get("tone", "neutral"),
            "tier": tier,
            "suggested_followups": self._generate_followups(intent, tier, context),
            "visualization_hint": self._get_visualization_hint(intent, tier),
            "timestamp": datetime.now().isoformat()
        }

    def _generate_followups(self, intent, tier, context):
        followups = []
        if tier != "scientific":
            followups.append("Can you explain that in more detail?")
        if intent == "recommendation":
            followups.append("How does this compare to other options?")
            followups.append("What are the expected results?")
        return followups[:3]

    def _get_visualization_hint(self, intent, tier):
        if tier == "scientific" and intent in ["explanation", "recommendation"]:
            return {
                "type": "biochemical_pathway",
                "suggested": True,
                "complexity": "high"
            }
        return None

                await pipe.lpush(cache_key, orjson.dumps(interaction))
                await pipe.ltrim(cache_key, 0, self.max_history - 1)
                await pipe.expire(cache_key, 86400)  # 24 hours
                await pipe.execute()

"""
Natural language processing for intent parsing.
"""
import spacy

class NLPProcessor:
    def __init__(self):
        try:
            self.nlp = spacy.load("en_core_web_sm")
        except:
            self.nlp = None

    async def parse_intent(self, query):
        if self.nlp:
            return self._parse_with_spacy(query)
        else:
            return self._parse_with_keywords(query)

    def _parse_with_spacy(self, query):
        doc = self.nlp(query.lower())
        intents = {
            "recommendation": ["recommend", "suggest", "should i use", "what product"],
            "explanation": ["why", "how does", "explain", "what is"],
            "comparison": ["compare", "difference", "versus", "vs", "better"],
            "scientific_inquiry": ["study", "research", "evidence", "clinical", "scientific"]
        }
        for intent, keywords in intents.items():
            if any(keyword in query.lower() for keyword in keywords):
                return {"intent": intent, "confidence": 0.9}
        if any(token.text in ["what", "how", "why", "when"] for token in doc):
            return {"intent": "general_inquiry", "confidence": 0.7}
        return {"intent": "general", "confidence": 0.5}

    def _parse_with_keywords(self, query):
        query_lower = query.lower()
        if any(word in query_lower for word in ["recommend", "suggest", "what should i"]):
            return {"intent": "recommendation", "confidence": 0.8}
        elif any(word in query_lower for word in ["why", "how does", "explain"]):
            return {"intent": "explanation", "confidence": 0.85}
        elif any(word in query_lower for word in ["compare", "difference", "vs"]):
            return {"intent": "comparison", "confidence": 0.9}
        return {"intent": "general", "confidence": 0.5}

xyra/skin/ingredient_analyzer.py
"""
Vectorized ingredient analysis.
"""
import numpy as np
from enum import Enum, auto

class SkinType(Enum):
    DRY = auto()
    OILY = auto()
    COMBINATION = auto()
    NORMAL = auto()
    SENSITIVE = auto()

class IngredientAnalyzer:
    def __init__(self):
        self.ingredient_vectors = self._load_ingredient_vectors()
        self.ingredient_index = {ing: idx for idx, ing in enumerate(self.ingredient_vectors.keys())}
        self.compatibility_matrix = np.array([
            [0.9, 0.1, 0.3, 0.7, 0.2],
            [0.1, 0.9, 0.5, 0.3, 0.2],
            [0.8, 0.2, 0.6, 0.5, 0.9],
            [0.1, 0.8, 0.4, 0.2, 0.1],
            [0.2, 0.1, 0.2, 0.1, 0.0],
        ])

    def _load_ingredient_vectors(self):
        return {
            "hyaluronic_acid": [1, 0, 0, 0, 0],
            "salicylic_acid": [0, 1, 0, 0, 0],
            "ceramides": [0, 0, 1, 0, 0],
            "niacinamide": [0, 0, 0, 1, 0],
            "fragrance": [0, 0, 0, 0, 1],
        }

    def analyze_batch(self, skin_types, ingredient_lists):
        n_skin_types = len(skin_types)
        n_products = len(ingredient_lists)
        ingredient_matrix = np.zeros((n_products, len(self.ingredient_vectors)))
        for i, ingredients in enumerate(ingredient_lists):
            for ing in ingredients:
                if ing in self.ingredient_index:
                    ingredient_matrix[i, self.ingredient_index[ing]] = 1
        skin_type_indices = np.array([st.value - 1 for st in skin_types])
        compatibility = np.zeros((n_skin_types, n_products))
        for i, st_idx in enumerate(skin_type_indices):
            compat_vector = self.compatibility_matrix[:, st_idx]
            compatibility[i, :] = ingredient_matrix @ compat_vector
        compatibility = np.clip(compatibility / 5.0, 0.0, 1.0)
        return compatibility

        monk_map = {1:2, 2:3, 3:5, 4:6, 5:8, 6:9}

            b_skin = b[skin_mask]
            if len(L_skin) > 0:
                L_mean = np.mean(L_skin)
                b_mean = np.mean(b_skin)
                if abs(b_mean) < 1e-6:
                    ita = 90.0 if L_mean > 50 else -90.0
                else:
                    ita = np.arctan((L_mean - 50) / b_mean) * 180 / np.pi

                ita = 0.0
        else:
            lab = cv2.cvtColor(image, cv2.COLOR_BGR2LAB)

                ita = 90.0 if L_mean > 50 else -90.0

                ita = np.arctan((L_mean - 50) / b_mean) * 180 / np.pi

"""
Texture analysis: pore density, roughness, elasticity, sebum, micro-line topology.
"""
import cv2
import numpy as np
from skimage import feature, filters, measure
from ..config import XyraConfig

class TextureAnalyzer:
    def __init__(self, config: XyraConfig):
        self.config = config

    def analyze(self, image: np.ndarray) -> dict:
        gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
        h, w = gray.shape

        # Pore detection
        pores = feature.blob_log(gray, max_sigma=5, num_sigma=5, threshold=0.1)
        pore_density = len(pores) / (h * w) * 10000
        pore_diameter = np.mean([p[2] for p in pores]) if pores else 0
        pore_distribution = np.std([p[0] for p in pores]) / w if pores else 0

        # Roughness
        gabor = filters.gabor(gray, frequency=0.1)[0]
        surface_roughness_ra = np.std(gabor)
        surface_roughness_rz = np.percentile(gabor, 95) - np.percentile(gabor, 5)

        # Wrinkles and lines
        edges = cv2.Canny(gray, 50, 150)
        wrinkle_density = np.sum(edges) / (h * w)
        sobelx = cv2.Sobel(gray, cv2.CV_64F, 1, 0, ksize=3)
        sobely = cv2.Sobel(gray, cv2.CV_64F, 0, 1, ksize=3)
        grad_mag = np.sqrt(sobelx**2 + sobely**2)
        wrinkle_depth = np.mean(grad_mag[edges > 0]) if np.sum(edges) > 0 else 0
        fine_line_density = wrinkle_density * 0.5

        # Micro-line topology (e.g., using line detection)
        lines = cv2.HoughLinesP(edges, 1, np.pi/180, threshold=50, minLineLength=10, maxLineGap=5)
        micro_line_topology = len(lines) / (h * w) * 10000 if lines is not None else 0

        # Elasticity proxy
        lbp = feature.local_binary_pattern(gray, P=8, R=1, method='uniform')
        lbp_var = np.var(lbp)
        elasticity = lbp_var / 100
        firmness = 1.0 - elasticity
        resilience = elasticity * 0.8

        # Microrelief pattern
        hist, _ = np.histogram(lbp, bins=np.arange(0, 11))
        hist = hist / hist.sum()
        entropy = -np.sum(hist * np.log(hist + 1e-7))
        microrelief_pattern = entropy / 3.0

        # Smoothness
        smoothness = 1.0 / (1.0 + surface_roughness_ra)

        # Sebum and oil
        hsv = cv2.cvtColor(image, cv2.COLOR_BGR2HSV)
        value_mean = np.mean(hsv[:,:,2])
        sebum_level = value_mean / 255.0
        oil_distribution = np.std(hsv[:,:,2]) / 255.0

        # Dryness
        sat_mean = np.mean(hsv[:,:,1]) / 255.0
        dry_patch_density = max(0, 1.0 - sat_mean * 2)
        skin_flake_density = max(0, dry_patch_density - 0.5) * 2

        return {
            "pore_density": float(pore_density),
            "pore_diameter": float(pore_diameter),
            "pore_distribution": float(pore_distribution),
            "surface_roughness_ra": float(surface_roughness_ra),
            "surface_roughness_rz": float(surface_roughness_rz),
            "wrinkle_count": float(wrinkle_density * 1000),
            "wrinkle_depth": float(wrinkle_depth),
            "fine_line_density": float(fine_line_density * 1000),
            "micro_line_topology": float(micro_line_topology),
            "elasticity": float(elasticity),
            "firmness": float(firmness),
            "resilience": float(resilience),
            "microrelief_pattern": float(microrelief_pattern),
            "smoothness": float(smoothness),
            "sebum_level": float(sebum_level),
            "oil_distribution": float(oil_distribution),
            "dry_patch_density": float(dry_patch_density),
            "skin_flake_density": float(skin_flake_density),
            "confidence": 0.7
        }

"""
Fast vector similarity search using FAISS (with NumPy fallback).
"""
import numpy as np

class VectorSimilarityEngine:
    def __init__(self, embedding_dim=256):
        self.embedding_dim = embedding_dim
        self.product_embeddings = None
        self.product_ids = None
        try:
            import faiss
            self.index = faiss.IndexFlatIP(embedding_dim)
            self.use_faiss = True
        except ImportError:
            self.use_faiss = False

    def build_index(self, product_embeddings, product_ids):
        self.product_embeddings = product_embeddings
        self.product_ids = product_ids
        if self.use_faiss and len(product_embeddings) > 100:
            import faiss
            faiss.normalize_L2(product_embeddings)
            self.index.add(product_embeddings)
        else:
            norms = np.linalg.norm(product_embeddings, axis=1, keepdims=True)
            norms[norms == 0] = 1
            self.product_embeddings = product_embeddings / norms

    def search(self, query_embedding, k=10, threshold=0.0):
        if len(self.product_embeddings) == 0:
            return []
        query_norm = np.linalg.norm(query_embedding)
        if query_norm > 0:
            query_embedding = query_embedding / query_norm
        if self.use_faiss and len(self.product_embeddings) > 100:
            import faiss
            query_embedding = query_embedding.reshape(1, -1)
            distances, indices = self.index.search(query_embedding, k)
            results = [
                (self.product_ids[idx], float(dist))
                for dist, idx in zip(distances[0], indices[0])
                if dist > threshold and idx != -1
            ]
        else:
            similarities = np.dot(self.product_embeddings, query_embedding)
            top_indices = np.argpartition(similarities, -k)[-k:]
            top_indices = top_indices[np.argsort(-similarities[top_indices])]
            results = [
                (self.product_ids[idx], float(similarities[idx]))
                for idx in top_indices
                if similarities[idx] > threshold
            ]
        return results

        async with pool.get_redis() as redis_pool:
            cached = await self.cache.get_distributed(redis_pool, cache_key)
            if cached:
                return cached

            # Retrieve user embedding
            user_embedding = await self._get_user_embedding_privately(anonymized_id, redis_pool)

            # Collaborative filtering: find similar users (simplified, would use a user-user matrix)
            # For now, use content-based on embedding.

            similar_products = self.similarity_engine.search(
                user_embedding,
                k=50,
                threshold=0.1
            )
            product_batch = []
            for pid, similarity in similar_products:
                product = self.product_catalog.get(pid)
                if product:
                    product_batch.append({
                        "id": pid,
                        "product": product,
                        "similarity": similarity
                    })
            recommendations = await self._process_batch_privately(
                product_batch, context, skin_data
            )
            final_recommendations = self._apply_business_rules(
                recommendations, context
            )
            dialogue = None
            if query:
                dialogue = await self.dialogue_orchestrator.generate_response(
                    user_id=anonymized_id,
                    query=query,
                    context=context,

            }
            await self.cache.set_distributed(redis_pool, cache_key, result)
            self.performance_monitor.record_metrics({
                "latency": (time.perf_counter() - start_time) * 1000,
                "cache_hit_rate": 0,
                "recommendations_generated": len(final_recommendations)
            })
            return result

    async def _get_user_embedding_privately(self, user_id, redis_pool):
        cache_key = f"user_embedding:{user_id}"
        cached = await redis_pool.get(cache_key)
        if cached:
            return np.frombuffer(cached, dtype=np.float32)
        query = """
            SELECT embedding, privacy_level FROM user_embeddings
            WHERE user_id = :user_id
            LIMIT 1
        """
        pool = await SecureConnectionPool.get_instance()
        async with pool.get_db() as db:
            result = await db.fetch_one(query, {"user_id": user_id})
            if result and result["embedding"]:
                embedding = np.frombuffer(result["embedding"], dtype=np.float32)
                privacy_level = result.get("privacy_level", "standard")
                if privacy_level == "high" and self.config.ENABLE_DIFFERENTIAL_PRIVACY:
                    embedding = self.privacy_manager.add_differential_privacy_noise(embedding)
            else:
                embedding = np.random.randn(self.config.USER_EMBEDDING_DIM).astype(np.float32)
                embedding = embedding / np.linalg.norm(embedding)
                await db.execute("""
                    INSERT INTO user_embeddings
                    (user_id, embedding, privacy_level, created_at, updated_at)
                    VALUES (:user_id, :embedding, :privacy_level, NOW(), NOW())
                    ON CONFLICT (user_id) DO UPDATE
                    SET embedding = EXCLUDED.embedding,
                        updated_at = NOW()
                """, {
                    "user_id": user_id,
                    "embedding": embedding.tobytes(),
                    "privacy_level": "standard"
                })
        await redis_pool.setex(cache_key, 3600, embedding.tobytes())
        return embedding

    async def _process_batch_privately(self, products, context, skin_data):
        if not products:
            return []
        async def process_chunk(chunk):
            return await self._process_product_chunk(chunk, context, skin_data)
        return await self.batch_processor.process_batch(products, process_chunk)

    async def _process_product_chunk(self, products, context, skin_data):
        processed = []
        for product_info in products:
            compatibility = await self.skin_analyzer.analyze(
                skin_type=skin_data.get("skin_type") if skin_data else "NORMAL",
                ingredients=product_info["product"].get("ingredients", []),
                image_data=skin_data.get("image") if skin_data else None
            )
            final_score = (
                0.5 * product_info["similarity"] +
                0.3 * compatibility.get("score", 0.5) +
                0.1 * product_info["product"].get("popularity", 0.5) +
                0.1 * product_info["product"].get("safety_score", 0.8)
            )
            processed.append({
                **product_info,
                "score": float(final_score),
                "compatibility": compatibility,
                "safety_score": product_info["product"].get("safety_score", 0.8)
            })
        return processed

    def _apply_business_rules(self, recommendations, context):
        price_sensitive = context.get("price_sensitive", False)
        max_price = context.get("max_price", float('inf'))
        min_safety = context.get("min_safety_score", 0.7)
        filtered = []
        for rec in recommendations:
            product = rec["product"]
            if product["price"] > max_price:
                continue
            if product.get("safety_score", 1.0) < min_safety:
                continue
            adjusted_score = rec["score"]
            if price_sensitive and product["price"] > 30:
                adjusted_score *= 0.8
            safety_boost = (product.get("safety_score", 1.0) - 0.7) * 0.1
            adjusted_score *= (1 + safety_boost)
            filtered.append({
                **rec,
                "final_score": adjusted_score,
                "reasoning": {
                    "price_appropriate": product["price"] <= max_price,
                    "safety_met": product.get("safety_score", 1.0) >= min_safety,
                    "personalization_strength": rec["similarity"]
                }
            })
        filtered.sort(key=lambda x: x["final_score"], reverse=True)
        return filtered

    async def record_feedback(self, user_id: str, interaction_id: str, rating: int, comment: Optional[str] = None):
        """
        Store explicit feedback and update user embedding.
        """
        anonymized_id = self.privacy_manager.anonymize_user_id(user_id)
        pool = await SecureConnectionPool.get_instance()
        async with pool.get_db() as db:
            await db.execute("""
                INSERT INTO user_feedback (user_id, interaction_id, rating, comment, created_at)
                VALUES (:user_id, :interaction_id, :rating, :comment, NOW())
            """, {
                "user_id": anonymized_id,
                "interaction_id": interaction_id,
                "rating": rating,
                "comment": comment
            })
        # Update user embedding asynchronously (could be background task)
        if self.user_embedding_updater:
            await self.user_embedding_updater.update_from_feedback(anonymized_id, interaction_id, rating)

"""

"""
from fastapi.security import HTTPBearer
from fastapi.responses import JSONResponse
import jwt
import os
from ..connection_pool import SecureConnectionPool
from ..config import XyraConfig

security = HTTPBearer()

class SecurityMiddleware:
    def __init__(self, config: XyraConfig):
        self.config = config
        self.secret_key = os.getenv("JWT_SECRET", config.JWT_SECRET)
        self.algorithm = "HS256"

    async def __call__(self, request, call_next):
        token = request.headers.get("Authorization")
        if not token or not token.startswith("Bearer "):
            return JSONResponse(
                status_code=401,
                content={"error": "Missing or invalid token"}
            )
        try:
            payload = jwt.decode(
                token[7:],
                self.secret_key,
                algorithms=[self.algorithm]
            )
            request.state.user_id = payload["sub"]
            request.state.role = payload.get("role", "user")
        except jwt.ExpiredSignatureError:
            return JSONResponse(
                status_code=401,
                content={"error": "Token expired"}
            )
        except jwt.InvalidTokenError:
            return JSONResponse(
                status_code=401,
                content={"error": "Invalid token"}
            )

        endpoint = request.url.path
        pool = await SecureConnectionPool.get_instance()
        if not await pool.check_rate_limit(request.state.user_id, endpoint):
            return JSONResponse(
                status_code=429,
                content={"error": "Rate limit exceeded", "retry_after": 60},
                headers={"Retry-After": "60"}
            )

        response = await call_next(request)
        response.headers["X-Content-Type-Options"] = "nosniff"
        response.headers["X-Frame-Options"] = "DENY"
        response.headers["X-XSS-Protection"] = "1; mode=block"
        response.headers["Strict-Transport-Security"] = "max-age=31536000; includeSubDomains"
        return response

            (BodySegment.LEFT_THIGH, BodySegment.HEIGHT): SegmentConstraint(0.22, 0.26, 0.240, 0.7),

"""
Probabilistic body model with shape classification and weight distribution.
"""
import time
import json
import zlib
import base64
import numpy as np
from typing import Dict, List, Any, Optional
from collections import defaultdict
from ..math.core import fast_bayesian_update
from ..data_structures import RingBuffer
from .body import BodySegment, AnthropometricDatabase, BodyShape

class ProbabilisticSegment:
    __slots__ = ['mean', 'std', 'confidence', 'sources', 'flags', '_ring_buffer']

    def __init__(self, mean: float = 0.0, std: float = 0.1,
                 confidence: float = 0.0, window_size: int = 5):
        self.mean = mean
        self.std = std
        self.confidence = confidence
        self.sources = []
        self.flags = []
        self._ring_buffer = RingBuffer(window_size)

    def update(self, observation: float, obs_std: float,
               source_id: int, obs_confidence: float) -> bool:
        if obs_confidence < 0.1:
            return False
        new_mean, new_std = fast_bayesian_update(
            self.mean, self.std,
            observation, obs_std
        )
        self._ring_buffer.push(observation)
        if self._ring_buffer.size >= 3:
            smoothed_mean = self._ring_buffer.get_smoothed(0.3)
            blend_factor = min(0.3, 1.0 / (len(self.sources) + 1))
            self.mean = (1 - blend_factor) * new_mean + blend_factor * smoothed_mean
        else:
            self.mean = new_mean
        self.std = new_std
        if len(self.sources) == 0:
            self.confidence = obs_confidence
        else:
            alpha = 0.3
            self.confidence = (1 - alpha) * self.confidence + alpha * obs_confidence
        self.sources.append(source_id)
        return True

    @property
    def reliability_score(self) -> float:
        if self.confidence < 0.3:
            return 0.0
        consistency = 1.0 / (1.0 + self.std / (abs(self.mean) + 1e-6))
        source_factor = min(1.0, len(self.sources) / 5.0)
        return self.confidence * 0.6 + consistency * 0.3 + source_factor * 0.1

    def to_dict(self) -> Dict[str, Any]:
        return {
            'mean': float(self.mean),
            'std': float(self.std),
            'confidence': float(self.confidence),
            'reliability': float(self.reliability_score),
            'sources_count': len(self.sources),
            'flags': self.flags.copy()
        }

class PBody:
    def __init__(self, config):
        self.config = config
        self.segments: Dict[BodySegment, ProbabilisticSegment] = {}
        self.height: Optional[ProbabilisticSegment] = None
        self.weight: Optional[ProbabilisticSegment] = None
        self.posture: Dict[str, np.ndarray] = {}
        self.overall_confidence = 0.0
        self.flags: List[str] = []
        self.metadata: Dict[str, Any] = {}
        for segment in BodySegment:
            if segment == BodySegment.HEIGHT:
                self.height = ProbabilisticSegment(window_size=config.smoothing_window)
            elif segment == BodySegment.WEIGHT:
                self.weight = ProbabilisticSegment(window_size=config.smoothing_window)
            else:
                self.segments[segment] = ProbabilisticSegment(window_size=config.smoothing_window)
        self.update_times = RingBuffer(100)
        self._last_update = time.time()
        self.shape: Optional[BodyShape] = None
        self.visceral_fat: Optional[float] = None
        self.subcutaneous_fat: Optional[float] = None
        self.ratios: Dict[str, float] = {}

    def update_segment(self, segment: BodySegment, observation: float,
                       obs_std: float, source_id: int, obs_confidence: float) -> bool:
        start = time.time()
        if segment == BodySegment.HEIGHT:
            success = self.height.update(observation, obs_std, source_id, obs_confidence)
        elif segment == BodySegment.WEIGHT:
            success = self.weight.update(observation, obs_std, source_id, obs_confidence)
        elif segment in self.segments:
            success = self.segments[segment].update(observation, obs_std, source_id, obs_confidence)
        else:
            success = False
        self.update_times.push(time.time() - start)
        return success

    def compute_overall_confidence(self) -> float:
        if not self.segments:
            self.overall_confidence = 0.0
            return 0.0
        importance_weights = {
            BodySegment.CHEST: 0.20,
            BodySegment.WAIST: 0.20,
            BodySegment.HIPS: 0.15,
            BodySegment.HEIGHT: 0.15,
            BodySegment.SHOULDER_WIDTH: 0.10,
            BodySegment.INSEAM: 0.10,
            BodySegment.LEFT_UPPER_ARM: 0.05,
            BodySegment.LEFT_THIGH: 0.05,
        }
        total_weight = 0.0
        weighted_conf = 0.0
        for segment, weight in importance_weights.items():
            if segment == BodySegment.HEIGHT:
                seg_confidence = self.height.confidence if self.height else 0.0
            elif segment in self.segments:
                seg_confidence = self.segments[segment].confidence
            else:
                continue
            weighted_conf += seg_confidence * weight
            total_weight += weight
        if total_weight > 0:
            self.overall_confidence = weighted_conf / total_weight
        else:
            self.overall_confidence = 0.0
        return self.overall_confidence

    def check_plausibility(self, anthropo_db) -> List[str]:
        issues = []
        if not self.height or self.height.mean < 0.1:
            return ["Missing or invalid height"]
        height = self.height.mean
        for segment, constraint in anthropo_db.height_constraints.items():
            if segment in self.segments:
                segment_val = self.segments[segment].mean
                ratio = segment_val / height
                if ratio < constraint.min_ratio or ratio > constraint.max_ratio:
                    issues.append(
                        f"{segment.value}: ratio {ratio:.3f} outside range "
                        f"[{constraint.min_ratio:.3f}, {constraint.max_ratio:.3f}]"
                    )
        bilateral_pairs = [
            (BodySegment.LEFT_UPPER_ARM, BodySegment.RIGHT_UPPER_ARM),
            (BodySegment.LEFT_THIGH, BodySegment.RIGHT_THIGH),
            (BodySegment.LEFT_CALF, BodySegment.RIGHT_CALF),
        ]
        for left, right in bilateral_pairs:
            if left in self.segments and right in self.segments:
                left_val = self.segments[left].mean
                right_val = self.segments[right].mean
                asymmetry = abs(left_val - right_val) / ((left_val + right_val) / 2 + 1e-6)
                if asymmetry > 0.15:
                    issues.append(f"Asymmetry in {left.value.replace('left_', '')}: {asymmetry:.1%}")
        self.flags.extend(issues)
