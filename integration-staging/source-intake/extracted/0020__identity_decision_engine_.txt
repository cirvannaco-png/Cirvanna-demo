```typescript
export class AtomicValue<T> {
  private value: T;
  private version: number = 0;
  constructor(initialValue: T) { this.value = initialValue; }
  get(): { value: T; version: number } { return { value: this.value, version: this.version }; }
  set(newValue: T): boolean { this.value = newValue; this.version++; return true; }
  compareAndSwap(expected: T, newValue: T): boolean {
    if (this.value === expected) {
      this.value = newValue;
      this.version++;
      return true;
    }
    return false;
  }
}
```

```typescript
export class RingBuffer<T> {
  private buffer: T[];
  private head: number = 0;
  private tail: number = 0;
  private size: number = 0;
  private readonly capacity: number;
  constructor(capacity: number) {
    this.capacity = capacity;
    this.buffer = new Array(capacity);
  }
  push(value: T): void {
    this.buffer[this.head] = value;
    this.head = (this.head + 1) % this.capacity;
    if (this.size < this.capacity) this.size++;
    else this.tail = (this.tail + 1) % this.capacity;
  }
  toArray(): T[] {
    const result: T[] = [];
    for (let i = 0; i < this.size; i++) {
      const index = (this.tail + i) % this.capacity;
      result.push(this.buffer[index]);
    }
    return result;
  }
  clear(): void { this.head = this.tail = this.size = 0; }
}
```

```typescript
export class DeterministicRandom {
  private static readonly A = 1664525;
  private static readonly C = 1013904223;
  private static readonly M = 2 ** 32;
  static seeded(seed: string): number {
    let hash = 0;
    for (let i = 0; i < seed.length; i++) {
      hash = ((hash << 5) - hash) + seed.charCodeAt(i);
      hash |= 0;
    }
    hash = Math.abs(hash);
    hash = (DeterministicRandom.A * hash + DeterministicRandom.C) % DeterministicRandom.M;
    return hash / DeterministicRandom.M;
  }
}
```

```typescript
import { DecisionInput, CulturalRequirement, ValidSeason, CriticalRegion } from '../types/analysis';
import { VALID_SEASONS, SEASON_MAP } from '../constants/seasons';
import { DeterministicHasher } from '../core/hashing';

export class InputValidator {
  static validate(input: unknown):
    | { valid: true; data: DecisionInput; hash: string }
    | { valid: false; error: string; details?: unknown } {
    try {
      if (!input || typeof input !== 'object') {
        return { valid: false, error: 'Input must be an object' };
      }
      const typed = input as Record<string, unknown>;

      const items = this.validateClothingItems(typed.clothingItems);
      if (!items.valid) return items;

      const season = this.validateSeason(typed.season, typed.timestamp);
      if (!season.valid) return season;

      const location = this.validateLocation(typed.location);
      if (!location.valid) return location;

      const timestamp = this.validateTimestamp(typed.timestamp);
      if (!timestamp.valid) return timestamp;

      const patterns = this.validatePatterns(typed.clothingPatterns);
      const identity = this.validateIdentity(typed.declaredIdentity);
      const cultural = this.validateCultural(typed.culturalRequirements);
      const correlationId = typeof typed.correlationId === 'string' ? typed.correlationId : undefined;

      const data: DecisionInput = {
        declaredIdentity: identity.data,
        clothingItems: items.data,
        clothingPatterns: patterns.data,
        location: location.data,
        season: season.data,
        timestamp: timestamp.data,
        culturalRequirements: cultural.data,
        correlationId
      };

      const hash = DeterministicHasher.generateInputHash(data as any);
      return { valid: true, data, hash };
    } catch (error) {
      return { valid: false, error: 'Validation processing error', details: error };
    }
  }

  private static validateClothingItems(items: unknown):
    | { valid: true; data: readonly string[] }
    | { valid: false; error: string } {
    if (!Array.isArray(items)) return { valid: false, error: 'clothingItems must be an array' };
    if (items.length === 0) return { valid: false, error: 'clothingItems cannot be empty' };
    if (items.length > 100) return { valid: false, error: 'Too many clothing items (max 100)' };
    const validated: string[] = [];
    for (const item of items) {
      if (typeof item !== 'string') return { valid: false, error: 'All clothing items must be strings' };
      const trimmed = item.trim();
      if (trimmed.length === 0) return { valid: false, error: 'Clothing item cannot be empty' };
      if (trimmed.length > 200) return { valid: false, error: 'Clothing item too long (max 200 chars)' };
      validated.push(trimmed.toLowerCase());
    }
    return { valid: true, data: Object.freeze(validated) };
  }

  private static validateSeason(season: unknown, timestamp: unknown):
    | { valid: true; data: ValidSeason }
    | { valid: false; error: string } {
    if (typeof season === 'string') {
      const norm = season.trim().toLowerCase() as ValidSeason;
      if (VALID_SEASONS.includes(norm)) return { valid: true, data: norm };
    }
    if (typeof timestamp === 'number' && timestamp > 0) {
      const date = new Date(timestamp);
      const month = date.getMonth();
      return { valid: true, data: SEASON_MAP[month] };
    }
    const now = new Date();
    return { valid: true, data: SEASON_MAP[now.getMonth()] };
  }

  private static validateLocation(location: unknown):
    | { valid: true; data: string }
    | { valid: false; error: string } {
    if (typeof location !== 'string') return { valid: false, error: 'location must be a string' };
    const trimmed = location.trim();
    if (trimmed.length === 0) return { valid: false, error: 'location cannot be empty' };
    if (trimmed.length > 200) return { valid: false, error: 'location too long (max 200 chars)' };
    return { valid: true, data: trimmed.toLowerCase() };
  }

  private static validateTimestamp(timestamp: unknown):
    | { valid: true; data: number }
    | { valid: false; error: string } {
    if (typeof timestamp !== 'number') return { valid: false, error: 'timestamp must be a number' };
    if (timestamp <= 0) return { valid: false, error: 'timestamp must be positive' };
    const normalized = Math.floor(timestamp / 1000) * 1000;
    if (normalized > Date.now() + 86400000) {
      return { valid: false, error: 'timestamp cannot be more than 24 hours in future' };
    }
    return { valid: true, data: normalized };
  }

  private static validatePatterns(patterns: unknown):
    | { valid: true; data: readonly string[] }
    | { valid: false; data: readonly string[] } {
    if (!Array.isArray(patterns)) return { valid: false, data: Object.freeze([]) };
    const validated: string[] = [];
    for (const p of patterns) {
      if (typeof p === 'string') {
        const trimmed = p.trim().toLowerCase();
        if (trimmed.length > 0 && trimmed.length <= 50) validated.push(trimmed);
      }
    }
    return { valid: true, data: Object.freeze([...new Set(validated)]) };
  }

  private static validateIdentity(identity: unknown):
    | { valid: true; data: string | readonly string[] | null }
    | { valid: false; data: null } {
    if (identity === null || identity === undefined) return { valid: false, data: null };
    if (typeof identity === 'string') {
      const trimmed = identity.trim().toLowerCase();
      return trimmed ? { valid: true, data: trimmed } : { valid: false, data: null };
    }
    if (Array.isArray(identity)) {
      const validated: string[] = [];
      for (const id of identity) {
        if (typeof id === 'string') {
          const trimmed = id.trim().toLowerCase();
          if (trimmed) validated.push(trimmed);
        }
      }
      if (validated.length === 0) return { valid: false, data: null };
      return { valid: true, data: Object.freeze([...new Set(validated)].sort()) };
    }
    return { valid: false, data: null };
  }

  private static validateCultural(requirements: unknown):
    | { valid: true; data: readonly CulturalRequirement[] | undefined }
    | { valid: false; data: undefined } {
    if (!Array.isArray(requirements)) return { valid: false, data: undefined };
    const validated: CulturalRequirement[] = [];
    for (const req of requirements) {
      if (req && typeof req === 'object' && 'culture' in req && 'regions' in req) {
        const culture = String(req.culture).trim().toLowerCase();
        if (!culture) continue;
        if (Array.isArray(req.regions)) {
          const regions: CriticalRegion[] = [];
          for (const r of req.regions) {
            if (typeof r === 'string' && ['head','torso','arms','legs'].includes(r)) {
              regions.push(r as CriticalRegion);
            }
          }
          if (regions.length) {
            validated.push({
              culture,
              regions: Object.freeze([...new Set(regions)].sort())
            });
          }
        }
      }
    }
    if (validated.length === 0) return { valid: false, data: undefined };
    return { valid: true, data: Object.freeze(validated.sort((a,b) => a.culture.localeCompare(b.culture))) };
  }
}
```

```typescript
import { CriticalRegion, CulturalRequirement } from '../types/analysis';
import { DEFAULT_CRITICAL, COVERAGE_MAP } from '../constants/regions';

export class CoverageAnalyzer {
  static analyze(
    items: readonly string[],
    culturalRequirements?: readonly CulturalRequirement[]
  ): {
    covered: readonly CriticalRegion[];
    uncovered: readonly CriticalRegion[];
    blocked: boolean;
    coverageMap: Readonly<Record<CriticalRegion, boolean>>;
  } {
    const criticalRegions = this.determineCriticalRegions(culturalRequirements);
    const coverage = this.analyzeCoverage(items, criticalRegions);
    const covered = criticalRegions.filter(r => coverage[r]);
    const uncovered = criticalRegions.filter(r => !coverage[r]);
    return {
      covered: Object.freeze(covered),
      uncovered: Object.freeze(uncovered),
      blocked: uncovered.length > 0,
      coverageMap: Object.freeze({ ...coverage })
    };
  }

  private static determineCriticalRegions(
    requirements?: readonly CulturalRequirement[]
  ): readonly CriticalRegion[] {
    if (!requirements || requirements.length === 0) {
      return DEFAULT_CRITICAL;
    }
    const regionSet = new Set<CriticalRegion>();
    for (const req of requirements) {
      for (const region of req.regions) {
        regionSet.add(region);
      }
    }
    return Object.freeze([...regionSet].sort());
  }

  private static analyzeCoverage(
    items: readonly string[],
    criticalRegions: readonly CriticalRegion[]
  ): Record<CriticalRegion, boolean> {
    const coverage: Record<CriticalRegion, boolean> = {
      head: false,
      torso: false,
      arms: false,
      legs: false
    };
    // Reset only critical regions
    for (const region of criticalRegions) {
      coverage[region] = false;
    }

    for (const item of items) {
      const regions = this.getItemCoverage(item);
      for (const region of regions) {
        if (criticalRegions.includes(region)) {
          coverage[region] = true;
        }
      }
      if (criticalRegions.every(r => coverage[r])) break;
    }
    return coverage;
  }

  private static getItemCoverage(item: string): readonly CriticalRegion[] {
    const lowerItem = item.toLowerCase();
    // Exact match first
    for (const [garment, regions] of Object.entries(COVERAGE_MAP)) {
      if (lowerItem === garment) {
        return regions;
      }
    }
    // Partial match
    for (const [garment, regions] of Object.entries(COVERAGE_MAP)) {
      if (lowerItem.includes(garment) || garment.includes(lowerItem)) {
        return regions;
      }
    }
    // Keyword fallback
    const regions: CriticalRegion[] = [];
    if (lowerItem.includes('head') || lowerItem.includes('scarf') || lowerItem.includes('hat')) {
      regions.push('head');
    }
    if (lowerItem.includes('shirt') || lowerItem.includes('top') || lowerItem.includes('blouse') ||
        lowerItem.includes('jacket') || lowerItem.includes('coat') || lowerItem.includes('vest')) {
      regions.push('torso');
    }
    if (lowerItem.includes('sleeve') || lowerItem.includes('arm') || lowerItem.includes('glove')) {
      regions.push('arms');
    }
    if (lowerItem.includes('pant') || lowerItem.includes('leg') || lowerItem.includes('skirt') ||
        lowerItem.includes('jean') || lowerItem.includes('short')) {
      regions.push('legs');
    }
    return Object.freeze([...new Set(regions)].sort());
  }
}
```

```typescript
import { IdentityAnalysis } from '../types/analysis';

export class IdentityAnalyzer {
  private static readonly REGISTRY = new Set<string>([
    'muslim', 'indian', 'chinese', 'japanese', 'korean',
    'western', 'african', 'latin', 'middle_eastern'
  ]);
  private static readonly VERSION = '7.0.0';

  static analyze(
    declaredIdentity: string | readonly string[] | null | undefined
  ): IdentityAnalysis {
    if (!declaredIdentity) {
      return {
        active: false,
        identities: Object.freeze([]),
        isPartial: false,
        version: this.VERSION
      };
    }
    const identities = Array.isArray(declaredIdentity)
      ? declaredIdentity
      : [declaredIdentity];
    const valid: string[] = [];
    for (const identity of identities) {
      if (this.REGISTRY.has(identity.toLowerCase())) {
        valid.push(identity);
      }
    }
    return {
      active: valid.length > 0,
      identities: Object.freeze(valid),
      isPartial: valid.length > 0 && valid.length < identities.length,
      version: this.VERSION
    };
  }
}
```

```typescript
import { CriticalRegion, ValidSeason } from '../types/core';
import { WardrobeAnalysis } from '../types/analysis';
import { PATTERN_DATABASE, MATERIAL_MAP } from '../constants/patterns';
import { COVERAGE_MAP } from '../constants/regions';

export class WardrobeAnalyzer {
  static analyze(
    items: readonly string[],
    patterns: readonly string[],
    season: ValidSeason,
    coverage: Readonly<Record<CriticalRegion, boolean>>,
    criticalRegions: readonly CriticalRegion[]
  ): WardrobeAnalysis {
    try {
      const derived = this.analyzePatterns(patterns, items, season, criticalRegions);
      const material = this.analyzeMaterials(items, criticalRegions);
      const coverageLevel = this.analyzeCoverageLevel(coverage);
      const allAttributes = [...derived, ...material, ...coverageLevel];
      const score = this.calculateCompositeScore(allAttributes);
      const descriptor = this.generateWeightedDescriptor(allAttributes, score);
      return {
        active: allAttributes.length > 0,
        patterns: Object.freeze(allAttributes.map(a => ({ type: a.value, confidence: a.confidence, weight: a.weight }))),
        score,
        descriptor
      };
    } catch (error) {
      return {
        active: false,
        patterns: [],
        score: 0,
        descriptor: 'wardrobe:undetermined'
      };
    }
  }

  private static analyzePatterns(
    patterns: readonly string[],
    items: readonly string[],
    season: ValidSeason,
    criticalRegions: readonly CriticalRegion[]
  ): Array<{ type: string; value: string; confidence: number; weight: number }> {
    const result: Array<{ type: string; value: string; confidence: number; weight: number }> = [];
    for (const pattern of patterns) {
      const patternData = PATTERN_DATABASE[pattern as keyof typeof PATTERN_DATABASE];
      if (!patternData) continue;
      const patternItems = items.filter(item =>
        item.toLowerCase().includes(pattern.replace('pattern_', ''))
      );
      if (patternItems.length === 0) continue;
      const weight = this.calculatePatternWeight(patternItems, criticalRegions);
      const seasonalFactor = patternData.seasonalRelevance[season] || 0.5;
      const confidence = patternData.confidence * seasonalFactor;
      result.push({
        type: 'PATTERN',
        value: pattern,
        confidence,
        weight
      });
      patternData.styles.forEach(style => {
        result.push({
          type: 'STYLE_INFERENCE',
          value: style,
          confidence: confidence * 0.8,
          weight
        });
      });
    }
    return result;
  }

  private static calculatePatternWeight(items: string[], criticalRegions: readonly CriticalRegion[]): number {
    if (items.length === 0) return 0;
    let totalWeight = 0;
    for (const item of items) {
      const regions = this.getItemCoverage(item);
      const coveredCritical = regions.filter(r => criticalRegions.includes(r));
      totalWeight += coveredCritical.length / criticalRegions.length;
    }
    return Math.min(1, totalWeight / items.length);
  }

  private static analyzeMaterials(
    items: readonly string[],
    criticalRegions: readonly CriticalRegion[]
  ): Array<{ type: string; value: string; confidence: number; weight: number }> {
    const materialCounts = new Map<string, { count: number; items: string[] }>();
    for (const item of items) {
      const lower = item.toLowerCase();
      for (const [material, keywords] of Object.entries(MATERIAL_MAP)) {
        if (keywords.some(k => lower.includes(k))) {
          const existing = materialCounts.get(material) || { count: 0, items: [] };
          existing.count++;
          existing.items.push(item);
          materialCounts.set(material, existing);
          break;
        }
      }
    }
    const result: Array<{ type: string; value: string; confidence: number; weight: number }> = [];
    for (const [material, data] of materialCounts.entries()) {
      const weight = this.calculatePatternWeight(data.items, criticalRegions);
      result.push({
        type: 'MATERIAL_INFERENCE',
        value: material,
        confidence: 0.7,
        weight
      });
    }
    return result;
  }

  private static analyzeCoverageLevel(coverage: Record<CriticalRegion, boolean>): Array<{ type: string; value: string; confidence: number; weight: number }> {
    const covered = Object.values(coverage).filter(Boolean).length;
    const total = Object.keys(coverage).length;
    if (covered === 0) return [];
    const levels = [
      { threshold: 4, value: 'FULL_COVERAGE', confidence: 0.95 },
      { threshold: 3, value: 'HIGH_COVERAGE', confidence: 0.85 },
      { threshold: 2, value: 'MODERATE_COVERAGE', confidence: 0.75 },
      { threshold: 1, value: 'MINIMAL_COVERAGE', confidence: 0.65 }
    ];
    for (const level of levels) {
      if (covered >= level.threshold) {
        return [{
          type: 'COVERAGE_LEVEL',
          value: level.value,
          confidence: level.confidence,
          weight: covered / total
        }];
      }
    }
    return [];
  }

  private static calculateCompositeScore(attributes: Array<{ confidence: number; weight: number }>): number {
    if (attributes.length === 0) return 0;
    let weightedSum = 0, totalWeight = 0;
    for (const attr of attributes) {
      weightedSum += attr.confidence * attr.weight;
      totalWeight += attr.weight;
    }
    return totalWeight > 0 ? weightedSum / totalWeight : 0;
  }

  private static generateWeightedDescriptor(
    attributes: Array<{ type: string; value: string; confidence: number; weight: number }>,
    compositeScore: number
  ): string {
    if (attributes.length === 0) return 'wardrobe:undetermined';
    const groups = new Map<string, string[]>();
    for (const attr of attributes) {
      const type = attr.type.toLowerCase().replace('_inference', '');
      if (!groups.has(type)) groups.set(type, []);
      groups.get(type)!.push(attr.value);
    }
    const parts: string[] = [];
    for (const [type, values] of groups.entries()) {
      const top = [...new Set(values)].sort().slice(0, 2);
      if (top.length) parts.push(`${type}:${top.join('+')}`);
    }
    const confidencePercent = Math.round(compositeScore * 100);
    return `wardrobe:${parts.sort().join('|')}|confidence:${confidencePercent}%`;
  }

  private static getItemCoverage(item: string): readonly CriticalRegion[] {
    const lowerItem = item.toLowerCase();
    for (const [garment, regions] of Object.entries(COVERAGE_MAP)) {
      if (lowerItem === garment || lowerItem.includes(garment) || garment.includes(lowerItem)) {
        return regions;
      }
    }
    return [];
  }
}
```

```typescript
import { ValidSeason } from '../types/core';
import { TrendAnalysis } from '../types/analysis';

export class TrendAnalyzer {
  static analyze(
    season: ValidSeason,
    location: string,
    timestamp: number,
    wardrobeScore: number,
    hasCulturalIdentity: boolean
  ): TrendAnalysis {
    const adjustments: Array<{ type: string; reason: string; impact: number }> = [];

    const seasonal = this.analyzeSeasonal(season, timestamp);
    if (seasonal.impact > 0) adjustments.push({ type: 'seasonal', reason: seasonal.reason, impact: seasonal.impact });

    const locationImpact = this.analyzeLocation(location);
    if (locationImpact.impact > 0) adjustments.push({ type: 'location', reason: locationImpact.reason, impact: locationImpact.impact });

    const temporal = this.analyzeTemporal(timestamp);
    if (temporal.impact > 0) adjustments.push({ type: 'temporal', reason: temporal.reason, impact: temporal.impact });

    if (hasCulturalIdentity) {
      const cultural = this.analyzeCulturalCoherence(season);
      if (cultural.impact > 0) adjustments.push({ type: 'cultural_seasonal', reason: cultural.reason, impact: cultural.impact });
    }

    const maxImpact = adjustments.length ? Math.max(...adjustments.map(a => a.impact)) : 0;
    return {
      adjustments: Object.freeze(adjustments.sort((a, b) => a.type.localeCompare(b.type))),
      maxImpact
    };
  }

  private static analyzeSeasonal(season: ValidSeason, timestamp: number): { impact: number; reason: string } {
    const date = new Date(timestamp);
    const month = date.getMonth();
    const expected: Record<ValidSeason, number[]> = {
      spring: [2, 3, 4],
      summer: [5, 6, 7],
      fall: [8, 9, 10],
      winter: [11, 0, 1]
    };
    if (!expected[season].includes(month)) {
      return { impact: 0.3, reason: `Season ${season} doesn't match current month ${month}` };
    }
    return { impact: 0, reason: '' };
  }

  private static analyzeLocation(location: string): { impact: number; reason: string } {
    const lower = location.toLowerCase();
    if (lower.includes('wedding') || lower.includes('gala') || lower.includes('ceremony')) {
      return { impact: 0.4, reason: `Formal venue: ${location}` };
    }
    if (lower.includes('beach') || lower.includes('park') || lower.includes('gym')) {
      return { impact: 0.2, reason: `Casual venue: ${location}` };
    }
    return { impact: 0, reason: '' };
  }

  private static analyzeTemporal(timestamp: number): { impact: number; reason: string } {
    const hour = new Date(timestamp).getHours();
    if (hour >= 22 || hour < 6) {
      return { impact: 0.1, reason: `Night hours (${hour}:00)` };
    }
    return { impact: 0, reason: '' };
  }

  private static analyzeCulturalCoherence(season: ValidSeason): { impact: number; reason: string } {
    if (season === 'summer') {
      return { impact: 0.2, reason: 'Cultural garments may be heavy in summer' };
    }
    return { impact: 0, reason: '' };
  }
}
```

```typescript
export class IdentityComposer {
  static compose(
    identityAnalysis: IdentityAnalysis,
    wardrobeAnalysis: WardrobeAnalysis,
    trendImpact: number
  ): HybridIdentity | null {
    if (identityAnalysis.active && !wardrobeAnalysis.active) {
      return {
        explicit: identityAnalysis.identities,
        wardrobe: 'none',
        confidence: 1.0,
        source: 'EXPLICIT'
      };
    }
    if (!identityAnalysis.active && wardrobeAnalysis.active) {
      const confidence = Math.max(0.1, wardrobeAnalysis.score * (1 - trendImpact));
      return {
        explicit: Object.freeze([]),
        wardrobe: wardrobeAnalysis.descriptor,
        confidence,
        source: 'WARDROBE'
      };
    }
    if (identityAnalysis.active && wardrobeAnalysis.active) {
      const explicitWeight = 0.7;
      const wardrobeWeight = 0.3;
      const wardrobeConfidence = wardrobeAnalysis.score * (1 - trendImpact);
      const confidence = (1.0 * explicitWeight) + (wardrobeConfidence * wardrobeWeight);
      return {
        explicit: identityAnalysis.identities,
        wardrobe: wardrobeAnalysis.descriptor,
        confidence: Math.min(0.99, confidence),
        source: 'HYBRID'
      };
    }
    return null;
  }
}
```

```typescript
import {
  DecisionInput,
  CoverageAnalysis,
  IdentityAnalysis,
  WardrobeAnalysis,
  TrendAnalysis,
  HybridIdentity,
  FinalDecision,
  DecisionStatus,
  IdentitySource,
  ProcessingStage
} from '../types';

export class DecisionComposer {
  static compose(
    input: DecisionInput,
    coverage: CoverageAnalysis,
    identity: IdentityAnalysis,
    wardrobe: WardrobeAnalysis,
    trend: TrendAnalysis,
    hybrid: HybridIdentity | null,
    stageTimings: Partial<Record<ProcessingStage, number>>,
    cacheHit: boolean,
    startTime: number
  ): FinalDecision {
    let status: DecisionStatus = 'APPROVED';
    if (coverage.blocked) {
      status = 'BLOCKED';
    } else if (!hybrid && !identity.active && !wardrobe.active) {
      status = 'NEEDS_REVIEW';
    }

    let finalIdentity: FinalDecision['identity'] = null;
    let source: IdentitySource = 'UNDETERMINED';

    if (hybrid) {
      source = hybrid.source;
      finalIdentity = hybrid;
    } else if (identity.active) {
      source = 'EXPLICIT';
      finalIdentity = identity.identities.length === 1
        ? identity.identities[0]
        : identity.identities;
    } else if (wardrobe.active) {
      source = 'WARDROBE';
      finalIdentity = wardrobe.descriptor;
    }

    const constraints: string[] = [];
    if (coverage.blocked) {
      constraints.push(`CMCS_BLOCKED:${coverage.uncovered.join(',')}`);
    }
    if (trend.maxImpact > 0) {
      constraints.push(`TREND_IMPACT:${trend.maxImpact.toFixed(2)}`);
    }
    if (identity.active) {
      constraints.push(`IDENTITY_VERSION:${identity.version}`);
      if (identity.isPartial) {
        constraints.push('PARTIAL_IDENTITY_MATCH');
      }
    }
    if (wardrobe.active) {
      constraints.push(`WARDROBE_CONFIDENCE:${wardrobe.score.toFixed(2)}`);
    }
    if (coverage.covered.length > 0) {
      constraints.push(`COVERED_REGIONS:${coverage.covered.join(',')}`);
    }

    const timestamp = Math.floor(Date.now() / 1000) * 1000;
    const hash = this.generateHash({
      inputHash: this.generateInputHash(input),
      identity: finalIdentity,
      source,
      constraints: constraints.sort(),
      timestamp: Math.floor(timestamp / 1000)
    });

    return {
      status,
      identity: finalIdentity,
      source,
      constraints: Object.freeze(constraints.sort()),
      timestamp,
      hash,
      metadata: {
        processingMs: performance.now() - startTime,
        cacheHit,
        correlationId: input.correlationId,
        stageTimings
      }
    };
  }

  private static generateInputHash(input: DecisionInput): string {
    const normalized = {
      identity: Array.isArray(input.declaredIdentity)
        ? [...input.declaredIdentity].sort()
        : input.declaredIdentity,
      items: [...input.clothingItems].sort(),
      patterns: input.clothingPatterns ? [...input.clothingPatterns].sort() : [],
      location: input.location,
      season: input.season,
      timestamp: Math.floor(input.timestamp / 1000),
      cultural: input.culturalRequirements
        ? input.culturalRequirements.map(r => ({
            culture: r.culture,
            regions: [...r.regions].sort()
          })).sort((a, b) => a.culture.localeCompare(b.culture))
        : []
    };
    const json = JSON.stringify(normalized);
    let hash = 0;
    for (let i = 0; i < json.length; i++) {
      hash = ((hash << 5) - hash) + json.charCodeAt(i);
      hash |= 0;
    }
    return Math.abs(hash).toString(36).padStart(13, '0');
  }

  private static generateHash(data: any): string {
    const normalized = JSON.stringify(data, (key, value) => {
      if (Array.isArray(value)) {
        return [...value].sort();
      }
      if (value && typeof value === 'object') {
        const sorted = Object.keys(value).sort();
        const result: Record<string, any> = {};
        for (const k of sorted) {
          result[k] = value[k];
        }
        return result;
      }
      return value;
    });
    let hash = 0;
    for (let i = 0; i < normalized.length; i++) {
      hash = ((hash << 5) - hash) + normalized.charCodeAt(i);
      hash |= 0;
    }
    return Math.abs(hash).toString(36).padStart(13, '0');
  }
}
```

```typescript
import { FinalDecision } from '../types/decisions';
import { MAX_CACHE_SIZE, CACHE_TTL_MS } from '../constants/patterns';

export class OptimizedCache {
  private cache = new Map<string, { value: FinalDecision; expiry: number }>();
  private accessTimes = new Map<string, number>();
  private hits = 0;
  private misses = 0;
  private size = 0;

  get(key: string): FinalDecision | null {
    const item = this.cache.get(key);
    if (!item) {
      this.misses++;
      return null;
    }
    if (Date.now() > item.expiry) {
      this.delete(key);
      this.misses++;
      return null;
    }
    this.accessTimes.set(key, Date.now());
    this.hits++;
    return item.value;
  }

  set(key: string, value: FinalDecision, ttl: number = CACHE_TTL_MS): void {
    if (this.size >= MAX_CACHE_SIZE) this.evictOldest();
    this.cache.set(key, { value, expiry: Date.now() + ttl });
    this.accessTimes.set(key, Date.now());
    this.size++;
  }

  delete(key: string): void {
    if (this.cache.delete(key)) {
      this.accessTimes.delete(key);
      this.size--;
    }
  }

  clear(): void { this.cache.clear(); this.accessTimes.clear(); this.size = 0; this.hits = 0; this.misses = 0; }

  private evictOldest(): void {
    let oldestKey: string | null = null;
    let oldestTime = Date.now();
    for (const [key, time] of this.accessTimes.entries()) {
      if (time < oldestTime) { oldestTime = time; oldestKey = key; }
    }
    if (oldestKey) this.delete(oldestKey);
  }

  stats(): { size: number; hitRate: number; hits: number; misses: number } {
    const total = this.hits + this.misses;
    return {
      size: this.size,
      hits: this.hits,
      misses: this.misses,
      hitRate: total > 0 ? this.hits / total : 0
    };
  }
}
```

src/resilience/circuit-breaker.ts

```typescript
export class CircuitBreaker {
  private state: 'CLOSED' | 'OPEN' | 'HALF_OPEN' = 'CLOSED';
  private failures = 0;
  private halfOpenAttempts = 0;
  private nextAttempt = 0;
  private stats = { successes: 0, failures: 0, trips: 0 };

  constructor(
    private readonly failureThreshold: number = 5,
    private readonly resetTimeout: number = 60000,
    private readonly halfOpenMaxAttempts: number = 3
  ) {}

  async execute<T>(operation: () => Promise<T>, fallback?: () => Promise<T>): Promise<T> {
    if (this.state === 'OPEN') {
      if (Date.now() > this.nextAttempt) {
        this.state = 'HALF_OPEN';
        this.halfOpenAttempts = 0;
      } else {
        if (fallback) return fallback();
        throw new Error('Circuit breaker is OPEN');
      }
    }

    try {
      const result = await operation();
      if (this.state === 'HALF_OPEN') {
        this.halfOpenAttempts++;
        if (this.halfOpenAttempts >= this.halfOpenMaxAttempts) {
          this.failures = 0;
          this.state = 'CLOSED';
        }
      } else {
        this.failures = 0;
      }
      this.stats.successes++;
      return result;
    } catch (error) {
      this.recordFailure();
      if (fallback) return fallback();
      throw error;
    }
  }

  private recordFailure(): void {
    this.failures++;
    this.stats.failures++;
    if (this.state === 'HALF_OPEN') {
      this.state = 'OPEN';
      this.nextAttempt = Date.now() + this.resetTimeout;
      this.stats.trips++;
    } else if (this.state === 'CLOSED' && this.failures >= this.failureThreshold) {
      this.state = 'OPEN';
      this.nextAttempt = Date.now() + this.resetTimeout;
      this.stats.trips++;
    }
  }

  getState(): string { return this.state; }
  getStats() { return { ...this.stats, state: this.state }; }
}
```

```typescript
import { RingBuffer } from '../structures/ring-buffer';
import { InputValidator } from '../validation/input-validator';
import { CoverageAnalyzer } from '../analysis/coverage-analyzer';
import { analyzeDeclaredIdentity } from '../core/identity';
import { WardrobeAnalyzer } from '../analysis/wardrobe-analyzer';
import { TrendAnalyzer } from '../analysis/trend-analyzer';
import { createHybridIdentity } from '../core/hybrid';
import { DeterministicHasher } from '../core/hashing';
import { OptimizedCache } from '../caching/optimized-cache';
import { CircuitBreaker } from '../resilience/circuit-breaker';
import { DecisionInput, FinalDecision, ProcessingStage, DecisionStatus, IdentitySource, HybridIdentity, CriticalRegion } from '../types';
import { BATCH_SIZE, VERSION } from '../constants';

export class IdentityDecisionEngine {
  private static instance: IdentityDecisionEngine;
  private cache = new OptimizedCache();
  private processingLocks = new Map<string, Promise<FinalDecision>>();
  private metrics = {
    totalRequests: 0,
    cacheHits: 0,
    processingTimes: new RingBuffer<number>(1000),
    errorCounts: new Map<string, number>()
  };
  private circuitBreaker = new CircuitBreaker();

  private constructor() {}

  static getInstance(): IdentityDecisionEngine {
    if (!this.instance) this.instance = new IdentityDecisionEngine();
    return this.instance;
  }

  async resolve(input: any): Promise<FinalDecision> {
    const startTime = performance.now();
    this.metrics.totalRequests++;

    try {
      const validation = InputValidator.validate(input);
      if (!validation.valid) {
        return this.createErrorDecision('VALIDATION_FAILED', {
          error: validation.error,
          details: validation.details,
          processingMs: performance.now() - startTime
        });
      }

      const validatedInput = validation.data;
      const cacheKey = validation.hash;
      const cached = this.cache.get(cacheKey);
      if (cached) {
        this.metrics.cacheHits++;
        return cached;
      }

      const existingPromise = this.processingLocks.get(cacheKey);
      if (existingPromise) return await existingPromise;

      const processingPromise = this.circuitBreaker.execute(
        () => this.processDecision(validatedInput, startTime),
        () => Promise.resolve(this.createErrorDecision('CIRCUIT_BREAKER_OPEN', {
          error: 'Service temporarily unavailable',
          processingMs: performance.now() - startTime
        }))
      );

      this.processingLocks.set(cacheKey, processingPromise);
      try {
        const decision = await processingPromise;
        if (decision.status === 'APPROVED' || decision.status === 'BLOCKED') {
          this.cache.set(cacheKey, decision);
        }
        return decision;
      } finally {
        this.processingLocks.delete(cacheKey);
      }
    } catch (error) {
      this.recordError(error);
      return this.createErrorDecision('PROCESSING_ERROR', {
        error: error instanceof Error ? error.message : 'Unknown error',
        processingMs: performance.now() - startTime
      });
    } finally {
      const processingTime = performance.now() - startTime;
      this.metrics.processingTimes.push(processingTime);
    }
  }

  async resolveBatch(inputs: any[]): Promise<FinalDecision[]> {
    const batches: any[][] = [];
    for (let i = 0; i < inputs.length; i += BATCH_SIZE) {
      batches.push(inputs.slice(i, i + BATCH_SIZE));
    }
    const results: FinalDecision[] = [];
    for (const batch of batches) {
      const batchResults = await Promise.all(batch.map(inp => this.resolve(inp)));
      results.push(...batchResults);
    }
    return results;
  }

  private async processDecision(input: DecisionInput, startTime: number): Promise<FinalDecision> {
    const stageTimings: Partial<Record<ProcessingStage, number>> = {};

    // CMCS
    const cmcsStart = performance.now();
    const coverage = CoverageAnalyzer.analyze(input.clothingItems, input.culturalRequirements);
    stageTimings.CMCS = performance.now() - cmcsStart;
    if (coverage.blocked) {
      return this.composeFinalDecision(input, coverage, null, null, null, null, stageTimings, false, startTime);
    }

    // CDIL
    const cdilStart = performance.now();
    const identity = await analyzeDeclaredIdentity(input.declaredIdentity);
    stageTimings.CDIL = performance.now() - cdilStart;

    // WDLS – pass critical regions
    const criticalRegions = [...Object.keys(coverage.coverageMap)] as CriticalRegion[];
    const wdlsStart = performance.now();
    const wardrobe = WardrobeAnalyzer.analyze(
      input.clothingItems,
      input.clothingPatterns || [],
      input.season as any,
      coverage.coverageMap,
      criticalRegions
    );
    stageTimings.WDLS = performance.now() - wdlsStart;

    // CTIF
    const ctifStart = performance.now();
    const trend = TrendAnalyzer.analyze(
      input.season as any,
      input.location,
      input.timestamp,
      wardrobe.score,
      identity.active
    );
    stageTimings.CTIF = performance.now() - ctifStart;

    // Hybrid
    const hybridStart = performance.now();
    const hybrid = createHybridIdentity(identity, wardrobe, trend.maxImpact);
    stageTimings.COMPOSITION = performance.now() - hybridStart;

    return this.composeFinalDecision(input, coverage, identity, wardrobe, trend, hybrid, stageTimings, false, startTime);
  }

  private composeFinalDecision(
    input: DecisionInput,
    coverage: { covered: readonly string[]; uncovered: readonly string[]; blocked: boolean; coverageMap: Record<CriticalRegion, boolean> },
    identity: { active: boolean; identities: readonly string[]; isPartial: boolean; version: string } | null,
    wardrobe: { active: boolean; descriptor: string; score: number } | null,
    trend: { maxImpact: number } | null,
    hybrid: HybridIdentity | null,
    stageTimings: Partial<Record<ProcessingStage, number>>,
    cacheHit: boolean,
    startTime: number
  ): FinalDecision {
    let status: DecisionStatus = 'APPROVED';
    if (coverage.blocked) status = 'BLOCKED';
    else if (!hybrid && !identity?.active && !wardrobe?.active) status = 'NEEDS_REVIEW';

    let finalIdentity: FinalDecision['identity'] = null;
    let source: IdentitySource = 'UNDETERMINED';

    if (hybrid) {
      source = hybrid.source;
      finalIdentity = hybrid;
    } else if (identity?.active) {
      source = 'EXPLICIT';
      finalIdentity = identity.identities.length === 1 ? identity.identities[0] : identity.identities;
    } else if (wardrobe?.active) {
      source = 'WARDROBE';
      finalIdentity = wardrobe.descriptor;
    }

    const constraints: string[] = [];
    if (coverage.blocked) constraints.push(`CMCS_BLOCKED:${coverage.uncovered.join(',')}`);
    if (trend?.maxImpact) constraints.push(`TREND_IMPACT:${trend.maxImpact.toFixed(2)}`);
    if (identity?.active) {
      constraints.push(`IDENTITY_VERSION:${identity.version}`);
      if (identity.isPartial) constraints.push('PARTIAL_IDENTITY_MATCH');
    }
    if (wardrobe?.active) constraints.push(`WARDROBE_CONFIDENCE:${wardrobe.score.toFixed(2)}`);
    if (coverage.covered.length) constraints.push(`COVERED_REGIONS:${coverage.covered.join(',')}`);
    constraints.sort();

    const timestamp = Math.floor(Date.now() / 1000) * 1000;
    const hash = DeterministicHasher.hashObject({
      type: 'FINAL_DECISION',
      inputHash: DeterministicHasher.generateInputHash(input as any),
      identity: finalIdentity ? (typeof finalIdentity === 'object' ? JSON.stringify(finalIdentity) : finalIdentity) : null,
      source,
      constraints,
      timestamp: Math.floor(timestamp / 1000)
    });

    return {
      status,
      identity: finalIdentity,
      source,
      constraints: Object.freeze(constraints),
      timestamp,
      hash,
      metadata: {
        processingMs: performance.now() - startTime,
        cacheHit,
        correlationId: input.correlationId,
        stageTimings
      }
    };
  }

  private createErrorDecision(errorType: string, metadata: { error: string; details?: unknown; processingMs: number }): FinalDecision {
    const timestamp = Math.floor(Date.now() / 1000) * 1000;
    const hash = DeterministicHasher.hashObject({
      type: 'ERROR',
      errorType,
      error: metadata.error,
      timestamp: Math.floor(timestamp / 1000)
    });
    return {
      status: 'NEEDS_REVIEW',
      identity: null,
      source: 'UNDETERMINED',
      constraints: Object.freeze([`ERROR:${errorType}`]),
      timestamp,
      hash,
      metadata: { processingMs: metadata.processingMs, cacheHit: false, stageTimings: {} },
      errorDetails: metadata.error,
      partialData: { coverage: { head: false, torso: false, arms: false, legs: false } }
    };
  }

  private recordError(error: unknown): void {
    const key = error instanceof Error ? error.constructor.name : 'UnknownError';
    this.metrics.errorCounts.set(key, (this.metrics.errorCounts.get(key) || 0) + 1);
  }

  clearCache(): void { this.cache.clear(); }

  getMetrics() {
    const times = this.metrics.processingTimes.toArray();
    const avg = times.length ? times.reduce((a,b) => a+b, 0) / times.length : 0;
    const sorted = [...times].sort((a,b) => a-b);
    const p95 = times.length ? sorted[Math.floor(0.95 * times.length)] : 0;
    const hitRate = this.metrics.totalRequests ? (this.metrics.cacheHits / this.metrics.totalRequests) * 100 : 0;
    return {
      totalRequests: this.metrics.totalRequests,
      cacheHitRate: `${hitRate.toFixed(2)}%`,
      avgProcessingTime: `${avg.toFixed(2)}ms`,
      p95ProcessingTime: `${p95.toFixed(2)}ms`,
      errorDistribution: Object.fromEntries(this.metrics.errorCounts),
      cacheStats: this.cache.stats(),
      circuitBreakerState: this.circuitBreaker.getState()
    };
  }

  async runSelfTest(): Promise<void> {
    if (process.env.NODE_ENV !== 'development') return;
    const testInput = {
      declaredIdentity: ['muslim', 'indian'],
      clothingItems: ['hijab', 'khanzu', 'leather shoes'],
      clothingPatterns: ['pattern_solid'],
      location: 'london',
      season: 'winter',
      timestamp: Date.now(),
      culturalRequirements: [
        { culture: 'muslim', criticalRegions: ['head', 'torso', 'arms', 'legs'] }
      ]
    };
    const decision1 = await this.resolve(testInput);
    const decision2 = await this.resolve(testInput);
    if (decision1.hash !== decision2.hash) {
      throw new Error('Self‑test failed: determinism violation');
    }
    console.log('Self‑test passed: determinism verified');
  }
}
```

```typescript
import { IdentityDecisionEngine } from '../engine/identity-decision-engine';
import { ValidSeason, CriticalRegion } from '../types/core';
import { SEASON_MAP } from '../constants/seasons';

export class BeautyAIEngine {
  private engine = IdentityDecisionEngine.getInstance();
  private featureFlags = new Map<string, boolean>();

  async validateVirtualTryOn(
    userId: string,
    virtualItems: {
      clothing: string[];
      makeup: string;
      accessories?: string[];
      hairStyle?: string;
    },
    userContext: {
      identities?: string[];
      location: string;
      season?: string;
      culturalPreferences?: Array<{ culture: string; criticalRegions: string[]; }>;
    },
    options?: { strictMode?: boolean; includeSuggestions?: boolean; }
  ): Promise<{
    approved: boolean;
    confidence: number;
    decisionId: string;
    culturalNotes: string[];
    suggestions?: string[];
    metadata: { processingTime: number; cacheHit: boolean; };
  }> {
    const startTime = performance.now();
    const allItems = [
      ...virtualItems.clothing,
      `makeup:${virtualItems.makeup}`,
      ...(virtualItems.accessories || []),
      ...(virtualItems.hairStyle ? [`hair:${virtualItems.hairStyle}`] : [])
    ];

    const input = {
      declaredIdentity: userContext.identities,
      clothingItems: allItems,
      location: userContext.location,
      season: userContext.season || this.detectSeason(),
      timestamp: Date.now(),
      culturalRequirements: userContext.culturalPreferences?.map(pref => ({
        culture: pref.culture,
        regions: pref.criticalRegions as CriticalRegion[]
      })),
      correlationId: `beauty-ai-${userId}-${Date.now()}`
    };

    const decision = await this.engine.resolve(input);
    const culturalNotes = this.extractCulturalNotes(decision);
    let suggestions: string[] | undefined;
    if (options?.includeSuggestions && decision.status !== 'APPROVED') {
      suggestions = this.generateSuggestions(decision);
    }

    return {
      approved: decision.status === 'APPROVED',
      confidence: this.extractConfidence(decision),
      decisionId: decision.hash,
      culturalNotes,
      suggestions,
      metadata: {
        processingTime: decision.metadata.processingMs,
        cacheHit: decision.metadata.cacheHit
      }
    };
  }

  async batchValidate(validations: Array<{ userId: string; virtualItems: any; userContext: any; }>): Promise<Array<{ userId: string; result: any; }>> {
    const inputs = validations.map(v => ({
      declaredIdentity: v.userContext.identities,
      clothingItems: v.virtualItems.clothing,
      location: v.userContext.location,
      season: v.userContext.season || this.detectSeason(),
      timestamp: Date.now(),
      culturalRequirements: v.userContext.culturalPreferences?.map((pref: any) => ({
        culture: pref.culture,
        regions: pref.criticalRegions
      })),
      correlationId: `batch-${v.userId}`
    }));
    const decisions = await this.engine.resolveBatch(inputs);
    return decisions.map((d, i) => ({
      userId: validations[i].userId,
      result: {
        approved: d.status === 'APPROVED',
        confidence: this.extractConfidence(d),
        decisionId: d.hash
      }
    }));
  }

  private detectSeason(): ValidSeason {
    const month = new Date().getMonth();
    return SEASON_MAP[month];
  }

  private extractCulturalNotes(decision: any): string[] {
    const notes: string[] = [];
    if (decision.constraints.some((c: string) => c.includes('PARTIAL_IDENTITY_MATCH')))
      notes.push('Some cultural identities were partially recognized');
    if (decision.constraints.some((c: string) => c.includes('TREND_IMPACT')))
      notes.push('Seasonal or location context considered');
    if (decision.source === 'EXPLICIT') notes.push('Based on explicit cultural declaration');
    else if (decision.source === 'WARDROBE') notes.push('Inferred from clothing patterns');
    else if (decision.source === 'HYBRID') notes.push('Combined explicit and wardrobe analysis');
    return notes;
  }

  private generateSuggestions(decision: any): string[] {
    const suggestions: string[] = [];
    if (decision.constraints.some((c: string) => c.startsWith('CMCS_BLOCKED')))
      suggestions.push('Consider adding garments to improve coverage');
    if (decision.constraints.some((c: string) => c.startsWith('TREND_IMPACT')))
      suggestions.push('This outfit may be more appropriate in a different season or location');
    if (decision.status === 'NEEDS_REVIEW')
      suggestions.push('Unable to determine cultural appropriateness - please provide more context');
    return suggestions;
  }

  private extractConfidence(decision: any): number {
    if (typeof decision.identity === 'object' && decision.identity && 'confidence' in decision.identity)
      return decision.identity.confidence;
    if (decision.status === 'APPROVED') return 1.0;
    if (decision.status === 'BLOCKED') return 0.0;
    return 0.5;
  }

  enableFeature(feature: string): void { this.featureFlags.set(feature, true); }
  disableFeature(feature: string): void { this.featureFlags.set(feature, false); }
  isFeatureEnabled(feature: string): boolean { return this.featureFlags.get(feature) || false; }
}
```
