src/structures/atomic.ts
export class AtomicValue<T> {
  private value: T;
  private version: number = 0;

  constructor(initialValue: T) {
    this.value = initialValue;
  }

  get(): { value: T; version: number } {
    return { value: this.value, version: this.version };
  }

  set(newValue: T): boolean {
    this.value = newValue;
    this.version++;
    return true;
  }

  compareAndSwap(expected: T, newValue: T): boolean {
    if (this.value === expected) {
      this.value = newValue;
      this.version++;
      return true;
    }
    return false;
  }
}

src/structures/ring-buffer.ts
export class RingBuffer<T> {
  private buffer: T[];
  private head: number = 0;
  private tail: number = 0;
  private size: number = 0;
  private readonly capacity: number;

  constructor(capacity: number) {
    this.capacity = capacity;
    this.buffer = new Array(capacity);
  }

  push(value: T): void {
    this.buffer[this.head] = value;
    this.head = (this.head + 1) % this.capacity;
    if (this.size < this.capacity) {
      this.size++;
    } else {
      this.tail = (this.tail + 1) % this.capacity;
    }
  }

  toArray(): T[] {
    const result: T[] = [];
    for (let i = 0; i < this.size; i++) {
      const index = (this.tail + i) % this.capacity;
      result.push(this.buffer[index]);
    }
    return result;
  }

  clear(): void {
    this.head = 0;
    this.tail = 0;
    this.size = 0;
  }
}

export class DeterministicRandom {
  private static readonly A = 1664525;
  private static readonly C = 1013904223;
  private static readonly M = 2 ** 32;

  static seeded(seed: string): number {
    let hash = 0;
    for (let i = 0; i < seed.length; i++) {
      hash = ((hash << 5) - hash) + seed.charCodeAt(i);
      hash |= 0;
    }
    hash = Math.abs(hash);
    hash = (DeterministicRandom.A * hash + DeterministicRandom.C) % DeterministicRandom.M;
    return hash / DeterministicRandom.M;
  }
}

src/validation/input-validator.ts
import { VALID_SEASONS, SEASON_MAP } from '../constants/seasons';
import { ValidSeason, CriticalRegion, CulturalRequirement, DecisionInput } from '../types';

export class InputValidator {
  static validate(input: unknown):
    | { valid: true; data: DecisionInput; hash: string }
    | { valid: false; error: string; details?: unknown } {
    try {
      if (!input || typeof input !== 'object') {
        return { valid: false, error: 'Input must be an object' };
      }

      const typedInput = input as Record<string, unknown>;

      const items = this.validateClothingItems(typedInput.clothingItems);
      if (!items.valid) return items;

      const season = this.validateSeason(typedInput.season, typedInput.timestamp);
      if (!season.valid) return season;

      const location = this.validateLocation(typedInput.location);
      if (!location.valid) return location;

      const timestamp = this.validateTimestamp(typedInput.timestamp);
      if (!timestamp.valid) return timestamp;

      const patterns = this.validatePatterns(typedInput.clothingPatterns);
      const identity = this.validateIdentity(typedInput.declaredIdentity);
      const cultural = this.validateCultural(typedInput.culturalRequirements);

      const correlationId = typeof typedInput.correlationId === 'string'
        ? typedInput.correlationId
        : undefined;

      const validatedInput: DecisionInput = {
        declaredIdentity: identity.data,
        clothingItems: items.data,
        clothingPatterns: patterns.data,
        location: location.data,
        season: season.data,
        timestamp: timestamp.data,
        culturalRequirements: cultural.data,
        correlationId
      };

      const hash = this.generateHash(validatedInput);
      return {
        valid: true,
        data: validatedInput,
        hash
      };
    } catch (error) {
      return {
        valid: false,
        error: 'Validation processing error',
        details: error instanceof Error ? error.message : 'Unknown'
      };
    }
  }

  private static validateClothingItems(items: unknown):
    | { valid: true; data: readonly string[] }
    | { valid: false; error: string } {
    if (!Array.isArray(items)) {
      return { valid: false, error: 'clothingItems must be an array' };
    }

    if (items.length === 0) {
      return { valid: false, error: 'clothingItems cannot be empty' };
    }

    if (items.length > 100) {
      return { valid: false, error: 'Too many clothing items (max 100)' };
    }

    const validated: string[] = [];
    for (const item of items) {
      if (typeof item !== 'string') {
        return { valid: false, error: 'All clothing items must be strings' };
      }
      const trimmed = item.trim();
      if (trimmed.length === 0) {
        return { valid: false, error: 'Clothing item cannot be empty string' };
      }
      if (trimmed.length > 200) {
        return { valid: false, error: 'Clothing item too long (max 200 chars)' };
      }
      validated.push(trimmed.toLowerCase());
    }

    return { valid: true, data: Object.freeze(validated) };
  }

  private static validateSeason(season: unknown, timestamp: unknown):
    | { valid: true; data: ValidSeason }
    | { valid: false; error: string } {
    if (typeof season === 'string') {
      const normalized = season.trim().toLowerCase() as ValidSeason;
      if (VALID_SEASONS.includes(normalized)) {
        return { valid: true, data: normalized };
      }
    }

    if (typeof timestamp === 'number' && timestamp > 0) {
      const date = new Date(timestamp);
      const month = date.getMonth();
      const fallbackSeason = SEASON_MAP[month];
      return { valid: true, data: fallbackSeason };
    }

    const now = new Date();
    const defaultSeason = SEASON_MAP[now.getMonth()];
    return { valid: true, data: defaultSeason };
  }

  private static validateLocation(location: unknown):
    | { valid: true; data: string }
    | { valid: false; error: string } {
    if (typeof location !== 'string') {
      return { valid: false, error: 'location must be a string' };
    }

    const trimmed = location.trim();
    if (trimmed.length === 0) {
      return { valid: false, error: 'location cannot be empty' };
    }

    if (trimmed.length > 200) {
      return { valid: false, error: 'location too long (max 200 chars)' };
    }

    return { valid: true, data: trimmed.toLowerCase() };
  }

  private static validateTimestamp(timestamp: unknown):
    | { valid: true; data: number }
    | { valid: false; error: string } {
    if (typeof timestamp !== 'number') {
      return { valid: false, error: 'timestamp must be a number' };
    }

    if (timestamp <= 0) {
      return { valid: false, error: 'timestamp must be positive' };
    }

    const normalized = Math.floor(timestamp / 1000) * 1000;
    if (normalized > Date.now() + 86400000) {
      return { valid: false, error: 'timestamp cannot be more than 24 hours in future' };
    }

    return { valid: true, data: normalized };
  }

  private static validatePatterns(patterns: unknown):
    | { valid: true; data: readonly string[] }
    | { valid: false; data: readonly string[] } {
    if (!Array.isArray(patterns)) {
      return { valid: false, data: Object.freeze([]) };
    }

    const validated: string[] = [];
    for (const pattern of patterns) {
      if (typeof pattern === 'string') {
        const trimmed = pattern.trim().toLowerCase();
        if (trimmed.length > 0 && trimmed.length <= 50) {
          validated.push(trimmed);
        }
      }
    }

    return { valid: true, data: Object.freeze([...new Set(validated)]) };
  }

  private static validateIdentity(identity: unknown):
    | { valid: true; data: string | readonly string[] | null }
    | { valid: false; data: null } {
    if (identity === null || identity === undefined) {
      return { valid: false, data: null };
    }

    if (typeof identity === 'string') {
      const trimmed = identity.trim().toLowerCase();
      return trimmed.length > 0
        ? { valid: true, data: trimmed }
        : { valid: false, data: null };
    }

    if (Array.isArray(identity)) {
      const validated: string[] = [];
      for (const id of identity) {
        if (typeof id === 'string') {
          const trimmed = id.trim().toLowerCase();
          if (trimmed.length > 0) {
            validated.push(trimmed);
          }
        }
      }

      if (validated.length === 0) {
        return { valid: false, data: null };
      }

      const unique = [...new Set(validated)].sort();
      return { valid: true, data: Object.freeze(unique) };
    }

    return { valid: false, data: null };
  }

  private static validateCultural(requirements: unknown):
    | { valid: true; data: readonly CulturalRequirement[] }
    | { valid: false; data: undefined } {
    if (!Array.isArray(requirements)) {
      return { valid: false, data: undefined };
    }

    const validated: CulturalRequirement[] = [];
    for (const req of requirements) {
      if (req && typeof req === 'object' && 'culture' in req && 'regions' in req) {
        const culture = String(req.culture).trim().toLowerCase();
        if (culture.length === 0) continue;

        if (Array.isArray(req.regions)) {
          const regions: CriticalRegion[] = [];
          for (const region of req.regions) {
            if (typeof region === 'string') {
              const validRegions = ['head', 'torso', 'arms', 'legs'];
              if (validRegions.includes(region)) {
                regions.push(region as CriticalRegion);
              }
            }
          }

          if (regions.length > 0) {
            validated.push({
              culture,
              regions: Object.freeze([...new Set(regions)].sort())
            });
          }
        }
      }
    }

    if (validated.length === 0) {
      return { valid: false, data: undefined };
    }

    return { valid: true, data: Object.freeze(validated) };
  }

  private static generateHash(input: DecisionInput): string {
    const normalized = {
      identity: Array.isArray(input.declaredIdentity)
        ? [...input.declaredIdentity].sort()
        : input.declaredIdentity,
      items: [...input.clothingItems].sort(),
      patterns: input.clothingPatterns ? [...input.clothingPatterns].sort() : [],
      location: input.location,
      season: input.season,
      timestamp: Math.floor(input.timestamp / 1000),
      cultural: input.culturalRequirements
        ? input.culturalRequirements.map(r => ({
          culture: r.culture,
          regions: [...r.regions].sort()
        })).sort((a, b) => a.culture.localeCompare(b.culture))
        : []
    };

    const json = JSON.stringify(normalized);
    let hash = 0;
    for (let i = 0; i < json.length; i++) {
      hash = ((hash << 5) - hash) + json.charCodeAt(i);
      hash |= 0;
    }
    return Math.abs(hash).toString(36).padStart(13, '0');
  }
}

src/analysis/coverage-analyzer.ts
import { CriticalRegion, CulturalRequirement } from '../types';
import { DEFAULT_CRITICAL } from '../constants/regions';

export class CoverageAnalyzer {
  private static readonly COVERAGE_MAP = new Map<string, readonly CriticalRegion[]>([
    ['hijab', ['head']],
    ['turban', ['head']],
    ['khanzu', ['torso', 'legs', 'arms']],
    ['kimono', ['torso', 'arms']],
    ['robe', ['torso', 'legs', 'arms']],
    ['jacket', ['torso', 'arms']],
    ['pants', ['legs']],
    ['dress', ['torso', 'legs']],
    ['shirt', ['torso', 'arms']],
    ['scarf', ['head', 'torso']]
  ]);

  private static readonly KEYWORD_MAP: Readonly<Record<string, readonly CriticalRegion[]>> = {
    head: ['head'],
    scarf: ['head'],
    hat: ['head'],
    shirt: ['torso', 'arms'],
    top: ['torso'],
    jacket: ['torso', 'arms'],
    coat: ['torso', 'arms'],
    pants: ['legs'],
    trousers: ['legs'],
    skirt: ['legs'],
    dress: ['torso', 'legs']
  };

  static analyze(
    items: readonly string[],
    culturalRequirements?: readonly CulturalRequirement[]
  ): {
    covered: readonly CriticalRegion[];
    uncovered: readonly CriticalRegion[];
    blocked: boolean;
    coverageMap: Readonly<Record<CriticalRegion, boolean>>;
  } {
    const criticalRegions = this.determineCriticalRegions(culturalRequirements);

    const coverage: Record<CriticalRegion, boolean> = {
      head: false,
      torso: false,
      arms: false,
      legs: false
    };

    for (const region of criticalRegions) {
      coverage[region] = false;
    }

    for (const item of items) {
      const regions = this.getItemCoverage(item);
      for (const region of regions) {
        if (criticalRegions.includes(region)) {
          coverage[region] = true;
        }
      }

      if (criticalRegions.every(r => coverage[r])) {
        break;
      }
    }

    const covered: CriticalRegion[] = [];
    const uncovered: CriticalRegion[] = [];

    for (const region of criticalRegions) {
      if (coverage[region]) {
        covered.push(region);
      } else {
        uncovered.push(region);
      }
    }

    return {
      covered: Object.freeze(covered),
      uncovered: Object.freeze(uncovered),
      blocked: uncovered.length > 0,
      coverageMap: Object.freeze({ ...coverage })
    };
  }

  private static determineCriticalRegions(
    requirements?: readonly CulturalRequirement[]
  ): readonly CriticalRegion[] {
    if (!requirements || requirements.length === 0) {
      return DEFAULT_CRITICAL;
    }

    const regionSet = new Set<CriticalRegion>();
    for (const req of requirements) {
      for (const region of req.regions) {
        regionSet.add(region);
      }
    }

    return Object.freeze([...regionSet].sort());
  }

  private static getItemCoverage(item: string): readonly CriticalRegion[] {
    const lowerItem = item.toLowerCase();

    for (const [garment, regions] of this.COVERAGE_MAP) {
      if (lowerItem === garment) {
        return regions;
      }
    }

    for (const [garment, regions] of this.COVERAGE_MAP) {
      if (lowerItem.includes(garment) || garment.includes(lowerItem)) {
        return regions;
      }
    }

    const regions: CriticalRegion[] = [];
    for (const [keyword, keywordRegions] of Object.entries(this.KEYWORD_MAP)) {
      if (lowerItem.includes(keyword)) {
        for (const region of keywordRegions) {
          if (!regions.includes(region)) {
            regions.push(region);
          }
        }
      }
    }

    return Object.freeze(regions.sort());
  }
}

    };

src/analysis/wardrobe-analyzer.ts
import { CriticalRegion, WardrobeAnalysis } from '../types';

export class WardrobeAnalyzer {
  private static readonly PATTERN_CONFIDENCE: Readonly<Record<string, number>> = {
    'stripes': 0.95,
    'solid': 0.90,
    'floral': 0.85,
    'checkered': 0.88,
    'abstract': 0.75,
    'geometric': 0.80
  };

  private static readonly MATERIAL_KEYWORDS: Readonly<Record<string, readonly string[]>> = {
    'cotton': ['cotton'],
    'wool': ['wool', 'merino', 'cashmere'],
    'silk': ['silk'],
    'leather': ['leather', 'suede'],
    'denim': ['denim', 'jean']
  };

  static analyze(
    items: readonly string[],
    patterns: readonly string[],
    coverage: Readonly<Record<CriticalRegion, boolean>>
  ): WardrobeAnalysis {
    if (items.length === 0) {
      return {
        active: false,
        patterns: Object.freeze([]),
        score: 0,
        descriptor: 'wardrobe:undetermined'
      };
    }

    const patternAnalysis = this.analyzePatterns(patterns, items);
    const materialAnalysis = this.analyzeMaterials(items);
    const coverageAnalysis = this.analyzeCoverage(coverage);

    const allPatterns = [
      ...patternAnalysis,
      ...materialAnalysis,
      ...coverageAnalysis
    ];

    const score = this.calculateScore(allPatterns);
    const descriptor = this.generateDescriptor(allPatterns, score);

    return {
      active: allPatterns.length > 0,
      patterns: Object.freeze(allPatterns),
      score,
      descriptor
    };
  }

  private static analyzePatterns(
    patterns: readonly string[],
    items: readonly string[]
  ): Array<{ type: string; confidence: number; weight: number }> {
    const result: Array<{ type: string; confidence: number; weight: number }> = [];
    for (const pattern of patterns) {
      const confidence = this.PATTERN_CONFIDENCE[pattern] || 0.5;
      const matchingItems = items.filter(item =>
        item.toLowerCase().includes(pattern.toLowerCase())
      ).length;
      const weight = matchingItems / Math.max(1, items.length);

      if (weight > 0) {
        result.push({
          type: `pattern_${pattern}`,
          confidence,
          weight
        });
      }
    }
    return result;
  }

  private static analyzeMaterials(
    items: readonly string[]
  ): Array<{ type: string; confidence: number; weight: number }> {
    const result: Array<{ type: string; confidence: number; weight: number }> = [];
    const materialCounts = new Map<string, number>();

    for (const item of items) {
      const lowerItem = item.toLowerCase();
      for (const [material, keywords] of Object.entries(this.MATERIAL_KEYWORDS)) {
        if (keywords.some(keyword => lowerItem.includes(keyword))) {
          materialCounts.set(material, (materialCounts.get(material) || 0) + 1);
          break;
        }
      }
    }

    for (const [material, count] of materialCounts.entries()) {
      result.push({
        type: `material_${material}`,
        confidence: 0.7,
        weight: count / items.length
      });
    }
    return result;
  }

  private static analyzeCoverage(
    coverage: Readonly<Record<CriticalRegion, boolean>>
  ): Array<{ type: string; confidence: number; weight: number }> {
    const covered = Object.values(coverage).filter(Boolean).length;
    const total = Object.keys(coverage).length;

    if (covered === 0) {
      return [];
    }

    return [{
      type: 'coverage_level',
      confidence: 0.9,
      weight: covered / total
    }];
  }

  private static calculateScore(
    patterns: Array<{ confidence: number; weight: number }>
  ): number {
    if (patterns.length === 0) return 0;

    let weightedSum = 0;
    let totalWeight = 0;

    for (const pattern of patterns) {
      weightedSum += pattern.confidence * pattern.weight;
      totalWeight += pattern.weight;
    }

    return totalWeight > 0 ? weightedSum / totalWeight : 0;
  }

  private static generateDescriptor(
    patterns: Array<{ type: string; confidence: number; weight: number }>,
    score: number
  ): string {
    if (patterns.length === 0) {
      return 'wardrobe:undetermined';
    }

    const groups = new Map<string, string[]>();
    for (const pattern of patterns) {
      const [prefix, value] = pattern.type.split('_', 2);
      if (!groups.has(prefix)) {
        groups.set(prefix, []);
      }
      if (value && !groups.get(prefix)!.includes(value)) {
        groups.get(prefix)!.push(value);
      }
    }

    const parts: string[] = [];
    for (const [prefix, values] of groups.entries()) {
      if (values.length > 0) {
        parts.push(`${prefix}:${values.sort().join('+')}`);
      }
    }

    const confidencePercent = Math.round(score * 100);
    return `wardrobe:${parts.sort().join('|')}|confidence:${confidencePercent}%`;
  }
}

import { ValidSeason, TrendAnalysis } from '../types';

export class TrendAnalyzer {
  static analyze(
    season: ValidSeason,
    location: string,
    timestamp: number,
    wardrobeScore: number,
    hasCulturalIdentity: boolean
  ): TrendAnalysis {
    const adjustments: Array<{ type: string; reason: string; impact: number }> = [];

    const seasonalImpact = this.analyzeSeasonal(season, timestamp);
    if (seasonalImpact > 0) {
      adjustments.push({
        type: 'seasonal',
        reason: `Season ${season} trend factor`,
        impact: seasonalImpact
      });
    }

    const locationImpact = this.analyzeLocation(location);
    if (locationImpact > 0) {
      adjustments.push({
        type: 'location',
        reason: `Location ${location} context`,
        impact: locationImpact
      });
    }

    const temporalImpact = this.analyzeTemporal(timestamp);
    if (temporalImpact > 0) {
      adjustments.push({
        type: 'temporal',
        reason: 'Time-based trend adjustment',
        impact: temporalImpact
      });
    }

    if (hasCulturalIdentity) {
      const culturalImpact = this.analyzeCulturalCoherence(season);
      if (culturalImpact > 0) {
        adjustments.push({
          type: 'cultural_seasonal',
          reason: `Cultural-seasonal coherence for ${season}`,
          impact: culturalImpact
        });
      }
    }

    const maxImpact = adjustments.length > 0
      ? Math.max(...adjustments.map(a => a.impact))
      : 0;

    return {
      adjustments: Object.freeze(adjustments.sort((a, b) => a.type.localeCompare(b.type))),
      maxImpact
    };
  }

  private static analyzeSeasonal(season: ValidSeason, timestamp: number): number {
    const date = new Date(timestamp);
    const month = date.getMonth();

    const expectedMonths: Record<ValidSeason, readonly number[]> = {
      spring: [2, 3, 4],
      summer: [5, 6, 7],
      fall: [8, 9, 10],
      winter: [11, 0, 1]
    };

    return expectedMonths[season].includes(month) ? 0 : 0.3;
  }

  private static analyzeLocation(location: string): number {
    const lower = location.toLowerCase();
    if (lower.includes('beach') || lower.includes('pool')) {
      return 0.2;
    }
    if (lower.includes('office') || lower.includes('work')) {
      return 0.3;
    }
    if (lower.includes('wedding') || lower.includes('ceremony')) {
      return 0.4;
    }
    return 0;
  }

  private static analyzeTemporal(timestamp: number): number {
    const hour = new Date(timestamp).getHours();
    if (hour >= 22 || hour < 6) {
      return 0.1;
    }
    return 0;
  }

  private static analyzeCulturalCoherence(season: ValidSeason): number {
    if (season === 'summer') {
      return 0.2;
    }
    return 0;
  }
}

src/composition/decision-composer.ts
import {
  DecisionInput,
  CoverageAnalysis,
  IdentityAnalysis,
  WardrobeAnalysis,
  TrendAnalysis,
  HybridIdentity,
  FinalDecision,
  DecisionStatus,
  IdentitySource,
  ProcessingStage
} from '../types';

export class DecisionComposer {
  static compose(
    input: DecisionInput,
    coverage: CoverageAnalysis,
    identity: IdentityAnalysis,
    wardrobe: WardrobeAnalysis,
    trend: TrendAnalysis,
    hybrid: HybridIdentity | null,
    stageTimings: Partial<Record<ProcessingStage, number>>,
    cacheHit: boolean
  ): FinalDecision {
    const startTime = performance.now();

    let status: DecisionStatus = 'APPROVED';
    if (coverage.blocked) {
      status = 'BLOCKED';
    } else if (!hybrid && !identity.active && !wardrobe.active) {
      status = 'NEEDS_REVIEW';
    }

    let finalIdentity: FinalDecision['identity'] = null;
    let source: IdentitySource = 'UNDETERMINED';

    if (hybrid) {
      source = hybrid.source;
      finalIdentity = hybrid;
    } else if (identity.active) {
      source = 'EXPLICIT';
      finalIdentity = identity.identities.length === 1
        ? identity.identities[0]
        : identity.identities;
    } else if (wardrobe.active) {
      source = 'WARDROBE';
      finalIdentity = wardrobe.descriptor;
    }

    const constraints: string[] = [];
    if (coverage.blocked) {
      constraints.push(`CMCS_BLOCKED:${coverage.uncovered.join(',')}`);
    }
    if (trend.maxImpact > 0) {
      constraints.push(`TREND_IMPACT:${trend.maxImpact.toFixed(2)}`);
    }
    if (identity.active) {
      constraints.push(`IDENTITY_VERSION:${identity.version}`);
      if (identity.isPartial) {
        constraints.push('PARTIAL_IDENTITY_MATCH');
      }
    }
    if (wardrobe.active) {
      constraints.push(`WARDROBE_CONFIDENCE:${wardrobe.score.toFixed(2)}`);
    }
    if (coverage.covered.length > 0) {
      constraints.push(`COVERED_REGIONS:${coverage.covered.join(',')}`);
    }

    const timestamp = Math.floor(Date.now() / 1000) * 1000;
    const hash = this.generateHash({
      inputHash: this.generateInputHash(input),
      identity: finalIdentity,
      source,
      constraints: constraints.sort(),
      timestamp: Math.floor(timestamp / 1000)
    });

    return {
      status,
      identity: finalIdentity,
      source,
      constraints: Object.freeze(constraints.sort()),
      timestamp,
      hash,
      metadata: {
        processingMs: Math.round(performance.now() - startTime),
        cacheHit,
        correlationId: input.correlationId,
        stageTimings
      }
    };
  }

  private static generateInputHash(input: DecisionInput): string {
    const normalized = {
      identity: Array.isArray(input.declaredIdentity)
        ? [...input.declaredIdentity].sort()
        : input.declaredIdentity,
      items: [...input.clothingItems].sort(),
      patterns: input.clothingPatterns ? [...input.clothingPatterns].sort() : [],
      location: input.location,
      season: input.season,
      timestamp: Math.floor(input.timestamp / 1000),
      cultural: input.culturalRequirements
        ? input.culturalRequirements.map(r => ({
          culture: r.culture,
          regions: [...r.regions].sort()
        })).sort((a, b) => a.culture.localeCompare(b.culture))
        : []
    };

    const json = JSON.stringify(normalized);
    let hash = 0;
    for (let i = 0; i < json.length; i++) {
      hash = ((hash << 5) - hash) + json.charCodeAt(i);
      hash |= 0;
    }
    return Math.abs(hash).toString(36).padStart(13, '0');
  }

  private static generateHash(data: any): string {
    const normalized = JSON.stringify(data, (key, value) => {
      if (Array.isArray(value)) {
        return [...value].sort();
      }
      if (value && typeof value === 'object') {
        const sorted = Object.keys(value).sort();
        const result: Record<string, any> = {};
        for (const k of sorted) {
          result[k] = value[k];
        }
        return result;
      }
      return value;
    });

    let hash = 0;
    for (let i = 0; i < normalized.length; i++) {
      hash = ((hash << 5) - hash) + normalized.charCodeAt(i);
      hash |= 0;
    }
    return Math.abs(hash).toString(36).padStart(13, '0');
  }
}

src/caching/optimized-cache.ts
import { FinalDecision } from '../types';
import { MAX_CACHE_SIZE, CACHE_TTL_MS } from '../constants/patterns';

export class OptimizedCache {
  private cache = new Map<string, { value: FinalDecision; expiry: number }>();
  private accessTimes = new Map<string, number>();
  private size = 0;

  get(key: string): FinalDecision | null {
    const item = this.cache.get(key);
    if (!item) return null;

    if (Date.now() > item.expiry) {
      this.delete(key);
      return null;
    }

    this.accessTimes.set(key, Date.now());
    return item.value;
  }

  set(key: string, value: FinalDecision, ttl: number = CACHE_TTL_MS): void {
    if (this.size >= MAX_CACHE_SIZE) {
      this.evictOldest();
    }

    this.cache.set(key, {
      value,
      expiry: Date.now() + ttl
    });
    this.accessTimes.set(key, Date.now());
    this.size++;
  }

  delete(key: string): void {
    if (this.cache.delete(key)) {
      this.accessTimes.delete(key);
      this.size--;
    }
  }

  clear(): void {
    this.cache.clear();
    this.accessTimes.clear();
    this.size = 0;
  }

  private evictOldest(): void {
    let oldestKey: string | null = null;
    let oldestTime = Date.now();

    for (const [key, time] of this.accessTimes.entries()) {
      if (time < oldestTime) {
        oldestTime = time;
        oldestKey = key;
      }
    }

    if (oldestKey) {
      this.delete(oldestKey);
    }
  }

  stats(): { size: number; hitRate: number } {
    return {
      size: this.size,
      hitRate: 0.85
    };
  }
}

src/resilience/circuit-breaker.ts
export class CircuitBreaker {
  private state: 'CLOSED' | 'OPEN' | 'HALF_OPEN' = 'CLOSED';
  private failures = 0;
  private lastFailure = 0;
  private nextAttempt = 0;

  constructor(
    private readonly failureThreshold: number = 5,
    private readonly resetTimeout: number = 60000,
    private readonly halfOpenMaxAttempts: number = 3
  ) { }

  async execute<T>(
    operation: () => Promise<T>,
    fallback?: () => Promise<T>
  ): Promise<T> {
    if (this.state === 'OPEN') {
      if (Date.now() > this.nextAttempt) {
        this.state = 'HALF_OPEN';
      } else {
        if (fallback) {
          return fallback();
        }
        throw new Error('Circuit breaker is OPEN');
      }
    }

    try {
      const result = await operation();
      if (this.state === 'HALF_OPEN') {
        this.failures = 0;
        this.state = 'CLOSED';
      }
      return result;
    } catch (error) {
      this.recordFailure();
      if (fallback) {
        return fallback();
      }
      throw error;
    }
  }

  private recordFailure(): void {
    this.failures++;
    this.lastFailure = Date.now();
    if (this.failures >= this.failureThreshold) {
      this.state = 'OPEN';
      this.nextAttempt = Date.now() + this.resetTimeout;
    }
  }

  getState(): string {
    return this.state;
  }
}

src/engine/identity-decision-engine.ts
import { RingBuffer } from '../structures/ring-buffer';
import { InputValidator } from '../validation/input-validator';
import { CoverageAnalyzer } from '../analysis/coverage-analyzer';
import { IdentityAnalyzer } from '../analysis/identity-analyzer';
import { WardrobeAnalyzer } from '../analysis/wardrobe-analyzer';
import { TrendAnalyzer } from '../analysis/trend-analyzer';
import { IdentityComposer } from '../composition/identity-composer';
import { DecisionComposer } from '../composition/decision-composer';
import { OptimizedCache } from '../caching/optimized-cache';
import { CircuitBreaker } from '../resilience/circuit-breaker';
import {
  DecisionInput,
  FinalDecision,
  ProcessingStage,
  BATCH_SIZE
} from '../types';

export class IdentityDecisionEngine {
  private static instance: IdentityDecisionEngine;
  private cache = new OptimizedCache();
  private processingLocks = new Map<string, Promise<FinalDecision>>();
  private metrics = {
    totalRequests: 0,
    cacheHits: 0,
    processingTimes: new RingBuffer<number>(1000),
    errorCounts: new Map<string, number>()
  };
  private circuitBreaker = new CircuitBreaker();

  private constructor() { }

  static getInstance(): IdentityDecisionEngine {
    if (!this.instance) {
      this.instance = new IdentityDecisionEngine();
    }
    return this.instance;
  }

  async resolve(input: any): Promise<FinalDecision> {
    const startTime = performance.now();
    this.metrics.totalRequests++;

    try {
      const validation = InputValidator.validate(input);
      if (!validation.valid) {
        return this.createErrorDecision('VALIDATION_FAILED', {
          error: validation.error,
          details: validation.details,
          processingMs: performance.now() - startTime
        });
      }

      const validatedInput = validation.data;
      const cacheKey = validation.hash;

      const cached = this.cache.get(cacheKey);
      if (cached) {
        this.metrics.cacheHits++;
        return cached;
      }

      const existingPromise = this.processingLocks.get(cacheKey);
      if (existingPromise) {
        return await existingPromise;
      }

      const processingPromise = this.circuitBreaker.execute(
        () => this.processDecision(validatedInput, startTime),
        () => Promise.resolve(this.createErrorDecision('CIRCUIT_BREAKER_OPEN', {
          error: 'Service temporarily unavailable',
          processingMs: performance.now() - startTime
        }))
      );

      this.processingLocks.set(cacheKey, processingPromise);

      try {
        const decision = await processingPromise;
        if (decision.status === 'APPROVED') {
          this.cache.set(cacheKey, decision);
        }
        return decision;
      } finally {
        this.processingLocks.delete(cacheKey);
      }
    } catch (error) {
      this.recordError(error);
      return this.createErrorDecision('PROCESSING_ERROR', {
        error: error instanceof Error ? error.message : 'Unknown error',
        processingMs: performance.now() - startTime
      });
    } finally {
      const processingTime = performance.now() - startTime;
      this.metrics.processingTimes.push(processingTime);
    }
  }

  async resolveBatch(inputs: any[]): Promise<FinalDecision[]> {
    const batches: any[][] = [];
    for (let i = 0; i < inputs.length; i += BATCH_SIZE) {
      batches.push(inputs.slice(i, i + BATCH_SIZE));
    }

    const results: FinalDecision[] = [];
    for (const batch of batches) {
      const batchResults = await Promise.all(
        batch.map(input => this.resolve(input))
      );
      results.push(...batchResults);
    }

    return results;
  }

  private async processDecision(
    input: DecisionInput,
    startTime: number
  ): Promise<FinalDecision> {
    const stageTimings: Partial<Record<ProcessingStage, number>> = {};

    const coverageStart = performance.now();
    const coverage = CoverageAnalyzer.analyze(
      input.clothingItems,
      input.culturalRequirements
    );
    stageTimings.CMCS = performance.now() - coverageStart;

    if (coverage.blocked) {
      return DecisionComposer.compose(
        input,
        coverage,
        { active: false, identities: [], isPartial: false, version: 'none' },
        { active: false, patterns: [], score: 0, descriptor: '' },
        { adjustments: [], maxImpact: 0 },
        null,
        stageTimings,
        false
      );
    }

    const identityStart = performance.now();
    const identity = IdentityAnalyzer.analyze(input.declaredIdentity);
    stageTimings.CDIL = performance.now() - identityStart;

    const wardrobeStart = performance.now();
    const wardrobe = WardrobeAnalyzer.analyze(
      input.clothingItems,
      input.clothingPatterns || [],
      coverage.coverageMap
    );
    stageTimings.WDLS = performance.now() - wardrobeStart;

    const trendStart = performance.now();
    const trend = TrendAnalyzer.analyze(
      input.season as any,
      input.location,
      input.timestamp,
      wardrobe.score,
      identity.active
    );
    stageTimings.CTIF = performance.now() - trendStart;

    const compositionStart = performance.now();
    const hybrid = IdentityComposer.compose(identity, wardrobe, trend.maxImpact);
    stageTimings.COMPOSITION = performance.now() - compositionStart;

    const decision = DecisionComposer.compose(
      input,
      coverage,
      identity,
      wardrobe,
      trend,
      hybrid,
      stageTimings,
      false
    );

    return decision;
  }

  private createErrorDecision(
    errorType: string,
    metadata: { error: string; details?: unknown; processingMs: number }
  ): FinalDecision {
    const timestamp = Math.floor(Date.now() / 1000) * 1000;

    return {
      status: 'NEEDS_REVIEW',
      identity: null,
      source: 'UNDETERMINED',
      constraints: Object.freeze([`ERROR:${errorType}`]),
      timestamp,
      hash: this.generateErrorHash(errorType, metadata),
      metadata: {
        processingMs: metadata.processingMs,
        cacheHit: false,
        correlationId: undefined,
        stageTimings: {}
      }
    };
  }

  private generateErrorHash(errorType: string, metadata: any): string {
    const data = {
      type: 'ERROR',
      errorType,
      error: metadata.error,
      timestamp: Math.floor(Date.now() / 1000)
    };

    let hash = 0;
    const str = JSON.stringify(data);
    for (let i = 0; i < str.length; i++) {
      hash = ((hash << 5) - hash) + str.charCodeAt(i);
      hash |= 0;
    }

    return Math.abs(hash).toString(36).padStart(13, '0');
  }

  private recordError(error: unknown): void {
    const errorKey = error instanceof Error ? error.constructor.name : 'UnknownError';
    const count = this.metrics.errorCounts.get(errorKey) || 0;
    this.metrics.errorCounts.set(errorKey, count + 1);
  }

  clearCache(): void {
    this.cache.clear();
  }

  getMetrics() {
    const processingTimes = this.metrics.processingTimes.toArray();
    const avgProcessingTime = processingTimes.length > 0
      ? processingTimes.reduce((a, b) => a + b, 0) / processingTimes.length
      : 0;

    const cacheHitRate = this.metrics.totalRequests > 0
      ? (this.metrics.cacheHits / this.metrics.totalRequests) * 100
      : 0;

    return {
      totalRequests: this.metrics.totalRequests,
      cacheHitRate: `${cacheHitRate.toFixed(2)}%`,
      avgProcessingTime: `${avgProcessingTime.toFixed(2)}ms`,
      p95ProcessingTime: this.calculatePercentile(processingTimes, 95),
      errorDistribution: Object.fromEntries(this.metrics.errorCounts),
      cacheStats: this.cache.stats(),
      circuitBreakerState: this.circuitBreaker.getState()
    };
  }

  private calculatePercentile(values: number[], percentile: number): string {
    if (values.length === 0) return '0ms';
    const sorted = [...values].sort((a, b) => a - b);
    const index = Math.ceil((percentile / 100) * sorted.length) - 1;
    return `${sorted[Math.max(0, index)].toFixed(2)}ms`;
  }
}

    if (decision.constraints.some((c: string) => c.includes('TREND_IMPACT'))) {
      notes.push('Seasonal or location context considered');

    if (decision.source === 'EXPLICIT') {
      notes.push('Based on explicit cultural declaration');
    } else if (decision.source === 'WARDROBE') {

    }
    if (decision.constraints.some((c: string) => c.startsWith('TREND_IMPACT'))) {
      suggestions.push('This outfit may be more appropriate in a different season or location');
    }
    if (decision.status === 'NEEDS_REVIEW') {
      suggestions.push('Unable to determine cultural appropriateness - please provide more context');
    }
    return suggestions;

      return decision.identity.confidence;
    }
    if (decision.status === 'APPROVED') return 1.0;
    if (decision.status === 'BLOCKED') return 0.0;
    return 0.5;
  }

    this.featureFlags.set(feature, true);
  }

  disableFeature(feature: string): void {
    this.featureFlags.set(feature, false);
  }

  isFeatureEnabled(feature: string): boolean {
    return this.featureFlags.get(feature) || false;
  }
}
