class AnalysisSandbox:
    """Enforces module isolation with no side-effects"""
    
    def __init__(self):
        self.original_stdout = sys.stdout
        self.original_stderr = sys.stderr
        self.isolated_environment = {}
    
    @contextmanager
    def isolated_execution(self):
        """Execute skin analysis in isolated context"""
        # Redirect all output to prevent side effects
        sys.stdout = open(os.devnull, 'w')
        sys.stderr = open(os.devnull, 'w')
        
        try:
            yield self.isolated_environment
        finally:
            # Restore original state
            sys.stdout.close()
            sys.stderr.close()
            sys.stdout = self.original_stdout
            sys.stderr = self.original_stderr
            self.isolated_environment.clear()
    
    def validate_no_side_effects(self, module) -> bool:
        """Verify module has no persistent state"""
        # Check for hidden state
        forbidden_attrs = ['_history', '_previous', '_cache', '_state']
        for attr in forbidden_attrs:
            if hasattr(module, attr):
                return False
        return True

class PreconditionGates:
    """EXACT IMPLEMENTATION OF SPECIFICATION SECTION 3"""
    
    @staticmethod
    def check_all_gates(input_data: SkinAnalysisInput, 
                       config: SkinAnalysisConfig) -> Tuple[bool, Optional[str]]:
        """
        Check all gates per specification
        Returns: (all_passed, refusal_flag)
        """
        gates = [
            ("LIGHTING_INSUFFICIENT", PreconditionGates._check_lighting_gate, 
             [input_data, config]),
            ("BODY_UNSTABLE", PreconditionGates._check_body_stability_gate, 
             [input_data, config]),
            ("OCCLUSION_TOO_HIGH", PreconditionGates._check_skin_visibility_gate, 
             [input_data, config]),
            ("SENSOR_NOISE", PreconditionGates._check_sensor_quality_gate, 
             [input_data, config]),
            ("CONFIDENCE_LAYER_VETO", PreconditionGates._check_confidence_layer_gate, 
             [input_data, config])
        ]
        
        for flag_name, gate_func, args in gates:
            if not gate_func(*args):
                return False, flag_name
        
        return True, None
    
    @staticmethod
    def _check_lighting_gate(input_data: SkinAnalysisInput, 
                           config: SkinAnalysisConfig) -> bool:
        """Gate: Lighting Confidence ≥ threshold"""
        # Environmental confidence from confidence layer
        if input_data.environmental_confidence < config.LIGHTING_CONFIDENCE_THRESHOLD:
            return False
        
        # Additional lighting quality checks
        ambient = input_data.lighting_metadata.get('ambient_light', 0)
        if ambient < config.LIGHTING_AMBIENT_MIN:
            return False
        
        return True
    
    @staticmethod
    def _check_body_stability_gate(input_data: SkinAnalysisInput,
                                 config: SkinAnalysisConfig) -> bool:
        """Gate: Body Calibration Stability - pose/orientation stable, low motion"""
        cal = input_data.body_calibration
        pose_stable = cal.get('pose_stability', 0) >= config.BODY_STABILITY_THRESHOLD
        low_motion = cal.get('motion_magnitude', 1.0) <= 0.1
        return pose_stable and low_motion
    
    @staticmethod
    def _check_skin_visibility_gate(input_data: SkinAnalysisInput,
                                  config: SkinAnalysisConfig) -> bool:
        """Gate: Skin Surface Visibility ≥ X% relevant skin visible"""
        visible_ratio = input_data.body_calibration.get('skin_visible_ratio', 0)
        return visible_ratio >= config.MIN_SKIN_VISIBILITY
    
    @staticmethod
    def _check_sensor_quality_gate(input_data: SkinAnalysisInput,
                                 config: SkinAnalysisConfig) -> bool:
        """Gate: Sensor Signal Quality - exposure, ISO, focus within bounds"""
        sensor = input_data.sensor_metadata
        exposure = sensor.get('exposure', 0)
        iso = sensor.get('iso', 0)
        focus_confidence = sensor.get('focus_confidence', 0)
        
        exposure_valid = (config.SENSOR_EXPOSURE_MIN <= exposure <= 
                         config.SENSOR_EXPOSURE_MAX)
        iso_valid = iso <= config.SENSOR_ISO_MAX
        focus_valid = focus_confidence >= config.SENSOR_FOCUS_MIN
        
        return exposure_valid and iso_valid and focus_valid
    
    @staticmethod
    def _check_confidence_layer_gate(input_data: SkinAnalysisInput,
                                   config: SkinAnalysisConfig) -> bool:
        """Gate: Confidence Layer Approval"""
        # Environmental confidence already checked in lighting gate
        # Add internal confidence check (simplified)
        if input_data.raw_frame.size > 0:
            frame_quality = np.mean(input_data.raw_frame)
            internal_confidence = 1.0 if (config.IMAGE_INTENSITY_MIN < 
                                        frame_quality < 
                                        config.IMAGE_INTENSITY_MAX) else 0.0
        else:
            internal_confidence = 0.0
        
        # Combined confidence check (per specification)
        combined_confidence = (input_data.environmental_confidence + 
                             internal_confidence) / 2
        return combined_confidence >= config.LIGHTING_CONFIDENCE_THRESHOLD

class PurePerceptionEngine:
    """PURE PERCEPTION - NO INTERPRETATION"""
    
    @staticmethod
    def compute_skin_state_vector(frame: np.ndarray) -> np.ndarray:
        """
        Compute the 5D skin state vector EXACTLY as specified
        Returns: [texture uniformity, surface reflectance, 
                 chromatic balance deviation, moisture proxy, 
                 micro-variation density]
        """
        if frame.size == 0:
            return np.full(SkinAnalysisConfig.VECTOR_DIMENSIONS, 0.5, 
                          dtype=np.float32)
        
        # Convert to appropriate color space if needed
        if len(frame.shape) == 3:
            gray = np.mean(frame, axis=2).astype(np.float32)
            color_channels = frame.astype(np.float32)
        else:
            gray = frame.astype(np.float32)
            color_channels = None
        
        # Compute all components
        texture_uniformity = PurePerceptionEngine._compute_texture_uniformity(gray)
        surface_reflectance = PurePerceptionEngine._compute_surface_reflectance(gray)
        chromatic_deviation = PurePerceptionEngine._compute_chromatic_deviation(
            color_channels)
        moisture_proxy = PurePerceptionEngine._compute_moisture_proxy(gray)
        micro_variation = PurePerceptionEngine._compute_micro_variation(gray)
        
        vector = np.array([
            texture_uniformity,
            surface_reflectance,
            chromatic_deviation,
            moisture_proxy,
            micro_variation
        ], dtype=np.float32)
        
        return np.clip(vector, 0.0, 1.0)
    
    @staticmethod
    def _compute_texture_uniformity(gray: np.ndarray) -> float:
        """Texture uniformity [0,1] - higher = more uniform"""
        if gray.size == 0:
            return 0.5
        
        if SCIPY_AVAILABLE:
            # Local standard deviation as inverse of uniformity
            local_mean = uniform_filter(gray, size=5)
            local_var = uniform_filter(gray**2, size=5) - local_mean**2
            local_std = np.sqrt(np.maximum(local_var, 0))
            uniformity = 1.0 - np.clip(np.mean(local_std) / 20.0, 0, 1)
        else:
            # Fallback implementation
            uniformity = 1.0 - np.clip(np.std(gray) / 50.0, 0, 1)
        
        return float(uniformity)
    
    @staticmethod
    def _compute_surface_reflectance(gray: np.ndarray) -> float:
        """Surface reflectance [0,1] - higher = more reflective"""
        if gray.size == 0:
            return 0.5
        
        # Normalize to [0, 1]
        if gray.max() > 0:
            reflectance = np.mean(gray) / 255.0
        else:
            reflectance = 0.5
        
        return float(np.clip(reflectance, 0, 1))
    
    @staticmethod
    def _compute_chromatic_deviation(color_channels: Optional[np.ndarray]) -> float:
        """Chromatic balance deviation [0,1] - higher = more deviation"""
        if color_channels is None or color_channels.size == 0:
            return 0.5
        
        # Compute deviation from balanced RGB
        r_mean = np.mean(color_channels[..., 0])
        g_mean = np.mean(color_channels[..., 1])
        b_mean = np.mean(color_channels[..., 2])
        
        max_val = max(r_mean, g_mean, b_mean)
        if max_val == 0:
            return 0.5
        
        ratios = [r_mean / max_val, g_mean / max_val, b_mean / max_val]
        deviation = np.std(ratios)
        return float(np.clip(deviation, 0, 1))
    
    @staticmethod
    def _compute_moisture_proxy(gray: np.ndarray) -> float:
        """Moisture proxy [0,1] via specular highlights"""
        if gray.size == 0:
            return 0.5
        
        # Bright pixels indicate specular highlights (moisture proxy)
        bright_threshold = 0.9 * 255
        bright_ratio = np.sum(gray > bright_threshold) / gray.size
        return float(np.clip(bright_ratio * 3, 0, 1))
    
    @staticmethod
    def _compute_micro_variation(gray: np.ndarray) -> float:
        """Micro-variation density [0,1] - higher = more variation"""
        if gray.size == 0:
            return 0.5
        
        if SCIPY_AVAILABLE:
            grad_x = sobel(gray, axis=0)
            grad_y = sobel(gray, axis=1)
            grad_mag = np.sqrt(grad_x**2 + grad_y**2)
            variation = np.clip(np.mean(grad_mag) / 30.0, 0, 1)
        else:
            # Simple gradient approximation
            grad_x = np.diff(gray, axis=0)
            grad_y = np.diff(gray, axis=1)
            if grad_x.size > 0 and grad_y.size > 0:
                # Pad to original size
                grad_x = np.pad(grad_x, ((0, 1), (0, 0)))
                grad_y = np.pad(grad_y, ((0, 0), (0, 1)))
                grad_mag = np.sqrt(grad_x**2 + grad_y**2)
                variation = np.clip(np.mean(grad_mag) / 30.0, 0, 1)
            else:
                variation = 0.5
        
        return float(variation)
    
    @staticmethod
    def detect_contamination(frame: np.ndarray, 
                           sensor_metadata: Dict) -> List[str]:
        """
        Detect contamination flags for CFRP weighting ONLY
        Returns EXACT flags from specification
        """
        flags = []
        
        # Check each contamination type from specification
        if frame.size > 0 and frame.mean() > 230:
            flags.append("SENSOR_SATURATION")
        
        if sensor_metadata.get("motion_blur_estimate", 0) > 0.5:
            flags.append("MOTION_BLUR")
        
        if sensor_metadata.get("noise_estimate", 0) > 0.3:
            flags.append("ENVIRONMENTAL_NOISE")
        
        # Check for makeup (simplified)
        if len(frame.shape) == 3:
            color_std = np.std(frame, axis=(0, 1))
            if np.any(color_std < 10):  # Very uniform color
                flags.append("MAKEUP_INTERFERENCE")
        
        # Check for harsh lighting
        if frame.size > 0:
            lighting_variance = np.var(frame)
            if lighting_variance > 5000:
                flags.append("HARSH_LIGHT_DISTORTION")
        
        return flags
    
    @staticmethod
    def compute_confidence_score(signal_quality: float,
                               contamination_flags: List[str]) -> float:
        """
        Compute confidence score per specification rules:
        - 0 hard refusal
        - low confidence compresses vector toward neutral
        - contamination flags affect weighting
        """
        if signal_quality == 0:
            return 0.0
        
        # Apply contamination penalties
        base_confidence = signal_quality
        if contamination_flags:
            # Each flag reduces confidence by 20%
            penalty = 0.8 ** len(contamination_flags)
            base_confidence *= penalty
        
        return float(np.clip(base_confidence, 0, 1))

                    f"dimensions, got {len(self.skin_state_vector)}"

    def to_specification_format(self) -> Dict:
        """EXACT REFUSAL SCHEMA FROM SPECIFICATION SECTION 4"""
        output = {
            "skin_state_vector": (
                None if self.skin_state_vector is None 
                else self.skin_state_vector.tolist()
            ),
            "confidence_score": float(self.confidence_score),
            "contamination_flags": self.contamination_flags.copy()
        }
        
        # Validate against specification schema
        required_keys = SkinAnalysisConfig.OUTPUT_SCHEMA.keys()
        if not all(key in output for key in required_keys):
            raise ValueError(f"Output missing required keys: {required_keys}")
        
        return output
    
    def is_refusal(self) -> bool:
        """Check if this is a refusal per specification"""
        return self.confidence_score == 0 or self.skin_state_vector is None

class CirvannaSkinAnalysis:
    """COMPLETE SPECIFICATION IMPLEMENTATION"""
    
    def __init__(self, config: Optional[SkinAnalysisConfig] = None):
        self.config = config or SkinAnalysisConfig()
        self.gates = PreconditionGates()
        self.engine = PurePerceptionEngine()
        self.sandbox = AnalysisSandbox()
        
        # Validate no side effects
        if not self.sandbox.validate_no_side_effects(self):
            raise RuntimeError("Module has side effects - violates specification")
    
    def analyze(self, input_data: SkinAnalysisInput) -> SkinAnalysisOutput:
        """
        Main analysis with EXACT three outcomes per Specification Section 5:
        1. Valid Skin State: gates pass, confidence > 0
        2. Suppressed Output: gates pass, confidence veto by layer
        3. Hard Refusal: any gate fails
        """
        # Execute in sandbox (Specification Section 6)
        with self.sandbox.isolated_execution():
            return self._analyze_isolated(input_data)
    
    def _analyze_isolated(self, input_data: SkinAnalysisInput) -> SkinAnalysisOutput:
        """Isolated analysis with no side effects"""
        
        # STEP 1: PRECONDITION GATES
        gates_passed, refusal_flag = self.gates.check_all_gates(
            input_data, self.config
        )
        
        if not gates_passed:
            # HARD REFUSAL - any gate fails (Specification Section 5)
            return SkinAnalysisOutput(
                skin_state_vector=None,
                confidence_score=0.0,
                contamination_flags=[refusal_flag] if refusal_flag else ["GATE_FAILURE"],
                refusal_flag=refusal_flag,
                gate_failure=True
            )
        
        # STEP 2: PURE PERCEPTION
        # Compute skin state vector (no interpretation)
        skin_vector = self.engine.compute_skin_state_vector(input_data.raw_frame)
        
        # Detect contamination (for CFRP weighting only)
        contamination_flags = self.engine.detect_contamination(
            input_data.raw_frame,
            input_data.sensor_metadata
        )
        
        # Compute signal quality (independent metric)
        frame_quality = self._compute_frame_quality(input_data.raw_frame)
        sensor_quality = self._compute_sensor_quality(input_data.sensor_metadata)
        signal_quality = (frame_quality + sensor_quality) / 2
        
        # STEP 3: CONFIDENCE SCORE
        # Apply specification rules exactly
        confidence_score = self.engine.compute_confidence_score(
            signal_quality,
            contamination_flags
        )
        
        # Confidence = 0 hard refusal (per specification)
        if confidence_score == 0:
            return SkinAnalysisOutput(
                skin_state_vector=None,
                confidence_score=0.0,
                contamination_flags=contamination_flags + ["CONFIDENCE_ZERO_REFUSAL"],
                gate_failure=False
            )
        
        # STEP 4: APPLY COMPRESSION
        # "low confidence compresses vector toward neutral"
        if confidence_score < 0.5:  # Low confidence threshold
            neutral = np.full(self.config.VECTOR_DIMENSIONS, 0.5, dtype=np.float32)
            compression_factor = confidence_score * 2  # Map [0, 0.5] to [0, 1]
            skin_vector = (compression_factor * skin_vector + 
                          (1 - compression_factor) * neutral)
        
        # STEP 5: VALID OUTPUT
        return SkinAnalysisOutput(
            skin_state_vector=skin_vector,
            confidence_score=confidence_score,
            contamination_flags=contamination_flags,
            refusal_flag=None,
            gate_failure=False
        )
    
    def _compute_frame_quality(self, frame: np.ndarray) -> float:
        """Compute frame signal quality"""
        if frame.size == 0:
            return 0.0
        
        # Check for valid signal range
        mean_intensity = np.mean(frame)
        if (mean_intensity < self.config.IMAGE_INTENSITY_MIN or 
            mean_intensity > self.config.IMAGE_INTENSITY_MAX):
            return 0.0
        
        # Check contrast
        if mean_intensity > 0:
            contrast = np.std(frame) / 128.0
        else:
            contrast = 0
        
        return float(np.clip(contrast, 0, 1))
    
    def _compute_sensor_quality(self, sensor_metadata: Dict) -> float:
        """Compute sensor signal quality"""
        iso = sensor_metadata.get('iso', 0)
        focus = sensor_metadata.get('focus_confidence', 0)
        exposure = sensor_metadata.get('exposure', 0)
        
        # ISO penalty
        iso_quality = max(0, 1 - iso / 3200.0)
        
        # Exposure quality
        exposure_quality = (1.0 if (self.config.SENSOR_EXPOSURE_MIN <= exposure <= 
                                  self.config.SENSOR_EXPOSURE_MAX) 
                          else 0.0)
        
        # Combined quality
        return float(np.clip((iso_quality + focus + exposure_quality) / 3, 0, 1))

# ============================================================================

class ConfidenceLayer:
    """SEPARATE MODULE with veto authority"""
    
    def __init__(self, veto_threshold: float = 0.1):
        self.veto_threshold = veto_threshold
    
    def evaluate(self, skin_output: SkinAnalysisOutput,
                additional_data: Dict) -> Dict:
        """
        Confidence layer evaluation with veto authority
        Returns CFRP-ready data per specification
        """
        # If skin analysis already refused, pass through
        if skin_output.is_refusal():
            return {
                "skin_state_vector": None,
                "confidence_score": 0.0,
                "contamination_flags": skin_output.contamination_flags,
                "confidence_layer_action": "PASS_THROUGH_REFUSAL",
                "timestamp": time.time()
            }
        
        # Confidence layer's internal evaluation
        layer_confidence = self._internal_evaluation(skin_output, additional_data)
        
        # VETO DECISION (Confidence Layer authority)
        if layer_confidence < self.veto_threshold:
            return {
                "skin_state_vector": None,  # SUPPRESSED OUTPUT
                "confidence_score": 0.0,
                "contamination_flags": ["CONFIDENCE_LAYER_VETO"],
                "confidence_layer_action": "VETO_APPLIED",
                "veto_reason": f"layer_confidence={layer_confidence:.3f}",
                "timestamp": time.time()
            }
        
        # Approved - apply weighting for CFRP
        weighted_confidence = skin_output.confidence_score * layer_confidence
        
        return {
            "skin_state_vector": (
                skin_output.skin_state_vector.tolist() 
                if skin_output.skin_state_vector is not None 
                else None
            ),
            "confidence_score": weighted_confidence,
            "contamination_flags": skin_output.contamination_flags,
            "confidence_layer_action": "APPROVED",
            "weighting_factor": layer_confidence,
            "timestamp": time.time()
        }
    
    def _internal_evaluation(self, skin_output: SkinAnalysisOutput,
                           additional_data: Dict) -> float:
        """Confidence layer internal logic"""
        # Start with environmental confidence
        base_confidence = additional_data.get("environmental_confidence", 1.0)
        
        # Downweight for contamination flags (per specification)
        if skin_output.contamination_flags:
            contamination_penalty = 0.8 ** len(skin_output.contamination_flags)
            base_confidence *= contamination_penalty
        
        # Additional confidence layer logic
        temporal_consistency = additional_data.get("temporal_consistency", 1.0)
        multi_modal_agreement = additional_data.get("multi_modal_agreement", 1.0)
        
        combined_confidence = (base_confidence * temporal_consistency * 
                             multi_modal_agreement)
        return float(np.clip(combined_confidence, 0, 1))

class CFRPIntegration:
    """COMPLETE CFRP INTEGRATION PER SPECIFICATION SECTION 8"""
    
    @staticmethod
    def process_skin_analysis(skin_output: SkinAnalysisOutput,
                            confidence_layer_result: Dict,
                            body_calibration: Dict,
                            lighting_metadata: Dict) -> Dict:
        """
        CFRP integration per Specification Section 8:
        - If confidence = 0 or veto: ignore skin analysis
        - If contamination flags present: downweight signal
        - Integrates body + lighting + skin vectors
        """
        # Rule 1: Ignore if confidence = 0 or veto
        if (confidence_layer_result.get("confidence_score", 0) == 0 or
            confidence_layer_result.get("confidence_layer_action") == "VETO_APPLIED"):
            return CFRPIntegration._create_fallback_output(
                body_calibration,
                lighting_metadata,
                "SKIN_ANALYSIS_IGNORED"
            )
        
        # Rule 2: Downweight if contamination flags present
        contamination_flags = confidence_layer_result.get("contamination_flags", [])
        weighting_factor = 1.0
        if contamination_flags:
            # Each contamination flag reduces weight by 20%
            weighting_factor = 0.8 ** len(contamination_flags)
        
        # Rule 3: Integrate vectors
        composite_vector = CFRPIntegration._create_composite_vector(
            skin_output,
            body_calibration,
            lighting_metadata,
            weighting_factor
        )
        
        return {
            "composite_perception_vector": composite_vector,
            "skin_analysis_weight": weighting_factor,
            "confidence_score": confidence_layer_result.get("confidence_score", 0),
            "contamination_flags": contamination_flags,
            "body_calibration_used": body_calibration,
            "lighting_adjustment_used": lighting_metadata,
            "integration_timestamp": time.time()
        }
    
    @staticmethod
    def _create_fallback_output(body_calibration: Dict,
                              lighting_metadata: Dict,
                              reason: str) -> Dict:
        """Fallback output when skin analysis is ignored"""
        return {
            "composite_perception_vector": None,
            "skin_analysis_weight": 0.0,
            "fallback_reason": reason,
            "body_calibration_used": body_calibration,
            "lighting_adjustment_used": lighting_metadata,
            "confidence_score": 0.0,
            "contamination_flags": [reason],
            "timestamp": time.time()
        }
    
    @staticmethod
    def _create_composite_vector(skin_output: SkinAnalysisOutput,
                               body_calibration: Dict,
                               lighting_metadata: Dict,
                               weighting_factor: float) -> Optional[List[float]]:
        """Integrate body + lighting + skin vectors"""
        if skin_output.skin_state_vector is None:
            return None
        
        # Extract relevant components from body calibration
        body_vector = [
            body_calibration.get('pose_stability', 0.5),
            body_calibration.get('orientation_confidence', 0.5),
            body_calibration.get('surface_mapping_confidence', 0.5)
        ]
        
        # Extract relevant components from lighting
        lighting_vector = [
            lighting_metadata.get('ambient_light', 0.5) / 100.0,
            lighting_metadata.get('light_uniformity', 0.5),
            lighting_metadata.get('intensity_correction', 1.0)
        ]
        
        # Combine vectors with weighting
        skin_vector = skin_output.skin_state_vector.tolist()
        combined = skin_vector + body_vector + lighting_vector
        
        # Apply weighting factor to skin components only
        for i in range(len(skin_vector)):
            combined[i] = combined[i] * weighting_factor
        
        return combined

def complete_specification_pipeline(raw_inputs: Dict) -> Dict:
    """COMPLETE PIPELINE PER SPECIFICATION"""
    try:
        # Step 1: Create input contract
        input_data = SkinAnalysisInput(**raw_inputs)
        
        # Step 2: Skin Analysis (Specification Sections 1-7)

        skin_output = analyzer.analyze(input_data)
        
        # Step 3: Confidence Layer (Specification Section 8)

        confidence_result = confidence_layer.evaluate(skin_output, raw_inputs)
        
        # Step 4: CFRP Integration (Specification Section 8)
