class BayesianLayer:
    """
    Hierarchical Beta‑Binomial model with persistent storage in Redis.
    Falls back to in‑memory dictionaries if Redis is unavailable or not configured.
    """
    def __init__(self, redis_client: Optional['redis.Redis'] = None, key_prefix: str = "fci:bayes:"):
        self.redis = redis_client
        self.prefix = key_prefix
        self._lock = threading.RLock()
        # In‑memory fallback (used if redis is None)
        self._mem_category: Dict[str, Tuple[float, float]] = {}
        self._mem_brand_category: Dict[Tuple[str, str], Tuple[float, float]] = {}
        self._mem_user_brand_category: Dict[Tuple[str, str, str], Tuple[float, float]] = {}
        # Global prior (fixed)
        self.global_alpha = 1.0
        self.global_beta = 1.0

    def _redis_key(self, level: str, *parts) -> str:
        """Generate Redis key for a given level and identifiers."""
        return self.prefix + level + ":" + ":".join(str(p) for p in parts)

    def _get_counts(self, key: str) -> Tuple[float, float]:
        """Retrieve (alpha, beta) from Redis or in‑memory."""
        if self.redis:
            data = self.redis.hgetall(key)
            if data:
                return float(data.get(b'alpha', 0.0)), float(data.get(b'beta', 0.0))
            return 0.0, 0.0
        else:
            return 0.0, 0.0  # will be handled by caller using in‑memory dicts

    def _set_counts(self, key: str, alpha: float, beta: float):
        """Store (alpha, beta) in Redis or in‑memory."""
        if self.redis:
            self.redis.hset(key, mapping={'alpha': alpha, 'beta': beta})
        # in‑memory is handled separately

    def update(self, category: str, brand: str, user: str, successes: float, trials: float):
        """Update all levels with a new observation."""
        if trials <= 0 or successes < 0 or successes > trials:
            raise ValueError("Invalid successes/trials")
        with self._lock:
            # Update user‑brand‑category
            ubc_key = self._redis_key("ubc", user, brand, category)
            if self.redis:
                a_ubc, b_ubc = self._get_counts(ubc_key)
                self._set_counts(ubc_key, a_ubc + successes, b_ubc + (trials - successes))
            else:
                key = (user, brand, category)
                a_ubc, b_ubc = self._mem_user_brand_category.get(key, (0.0, 0.0))
                self._mem_user_brand_category[key] = (a_ubc + successes, b_ubc + (trials - successes))

            # Update brand‑category
            bc_key = self._redis_key("bc", brand, category)
            if self.redis:
                a_bc, b_bc = self._get_counts(bc_key)
                self._set_counts(bc_key, a_bc + successes, b_bc + (trials - successes))
            else:
                key = (brand, category)
                a_bc, b_bc = self._mem_brand_category.get(key, (0.0, 0.0))
                self._mem_brand_category[key] = (a_bc + successes, b_bc + (trials - successes))

            # Update category
            cat_key = self._redis_key("cat", category)
            if self.redis:
                a_c, b_c = self._get_counts(cat_key)
                self._set_counts(cat_key, a_c + successes, b_c + (trials - successes))
            else:
                a_c, b_c = self._mem_category.get(category, (0.0, 0.0))
                self._mem_category[category] = (a_c + successes, b_c + (trials - successes))

    def predict(self, category: str, brand: str, user: str) -> float:
        """Return posterior mean for the given user, brand, category."""
        with self._lock:
            # Get brand‑category prior
            bc_key = self._redis_key("bc", brand, category)
            if self.redis:
                a_bc, b_bc = self._get_counts(bc_key)
                if a_bc + b_bc == 0:
                    # fallback to global
                    a_bc, b_bc = self.global_alpha, self.global_beta
            else:
                key_bc = (brand, category)
                a_bc, b_bc = self._mem_brand_category.get(key_bc, (self.global_alpha, self.global_beta))

            # Get user‑brand‑category data
            ubc_key = self._redis_key("ubc", user, brand, category)
            if self.redis:
                a_ubc, b_ubc = self._get_counts(ubc_key)
            else:
                key_ubc = (user, brand, category)
                a_ubc, b_ubc = self._mem_user_brand_category.get(key_ubc, (0.0, 0.0))

            # Posterior mean
            trials_ubc = a_ubc + b_ubc
            if trials_ubc > 0:
                posterior_alpha = a_bc + a_ubc
                posterior_beta = b_bc + b_ubc
            else:
                posterior_alpha = a_bc
                posterior_beta = b_bc

            return posterior_alpha / (posterior_alpha + posterior_beta)

class NeuralResidual:
    """Wraps any sklearn‑compatible regressor and caps output to ±0.1."""
    def __init__(self, model: Any, feature_names: List[str]):
        self.model = model
        self.feature_names = feature_names

    def predict(self, features: Dict[str, Any]) -> float:
        try:
            X = np.array([[features[name] for name in self.feature_names]])
            residual = self.model.predict(X)[0]
        except Exception as e:
            logger.error(f"Neural residual failed: {e}")
            residual = 0.0
        return max(-0.1, min(0.1, float(residual)))

# Fast Calibrator with local‑variance confidence intervals
# ----------------------------------------------------------------------
class FastCalibrator:
    """
    Calibrates raw scores using isotonic regression.
    Confidence intervals are approximated using the variance of the
    k‑nearest training scores (local binomial variance).
    """
    def __init__(self, k: int = 50, method: str = "isotonic"):
        self.k = k
        self.method = method
        self.isotonic = None
        self.X_train = None
        self.y_train = None
        self.nn = None

    def fit(self, X: np.ndarray, y: np.ndarray):
        """X: raw scores (n_samples,), y: binary labels."""
        X = X.ravel()
        self.isotonic = IsotonicRegression(out_of_bounds="clip")
        self.isotonic.fit(X, y)
        # Store training data for nearest neighbor search
        self.X_train = X.reshape(-1, 1)
        self.y_train = y
        self.nn = NearestNeighbors(n_neighbors=min(self.k, len(X)))
        self.nn.fit(self.X_train)

    def predict_proba(self, scores: np.ndarray) -> np.ndarray:
        """Return calibrated probabilities."""
        scores = np.atleast_1d(scores)
        return self.isotonic.predict(scores).clip(0, 1)

    def confidence_interval(self, scores: np.ndarray, confidence: float = 0.95) -> Tuple[np.ndarray, np.ndarray]:
        """
        Approximate confidence intervals using local binomial variance.
        For each score, find k nearest training scores, compute the empirical
        variance of the binary labels in that neighborhood, and assume asymptotic
        normality to construct a confidence interval around the calibrated value.
        """
        scores = np.atleast_1d(scores)
        if self.nn is None or len(self.X_train) == 0:
            # fallback
            lower = np.clip(scores - 0.05, 0, 1)
            upper = np.clip(scores + 0.05, 0, 1)
            return lower, upper

        # Get k nearest neighbors for each score
        X_query = scores.reshape(-1, 1)
        distances, indices = self.nn.kneighbors(X_query)

        z = 1.96  # for 95% CI (approx)
        lower = []
        upper = []
        for i, idx in enumerate(indices):
            y_neigh = self.y_train[idx]
            p_hat = np.mean(y_neigh)
            # binomial variance
            var = p_hat * (1 - p_hat) / len(y_neigh)
            std = np.sqrt(var)
            calibrated = self.isotonic.predict([scores[i]])[0]
            # CI around calibrated value using the same std (heuristic)
            ci_lower = calibrated - z * std
            ci_upper = calibrated + z * std
            lower.append(ci_lower)
            upper.append(ci_upper)
        lower = np.clip(lower, 0, 1)
        upper = np.clip(upper, 0, 1)
        return lower, upper

class RuleBasedFrictionPolicy(FrictionPolicy):
    """Rule‑based policy with guardrails."""
    def __init__(self, thresholds: Optional[Dict[str, float]] = None):
        self.thresholds = thresholds or {
            "fci_high": 0.8,
            "fci_medium": 0.65,
            "bmc_low": 0.6,
            "gsr_low": 0.5
        }

    def choose_action(self, state: Dict[str, Any]) -> FrictionAction:
        if state["gsr"] < self.thresholds["gsr_low"]:
            return FrictionAction.SUGGEST_RECALIBRATION
        if state["bmc"] < self.thresholds["bmc_low"]:
            return FrictionAction.SUGGEST_ALTERNATIVE
        fci = state["fci"]
        if fci >= self.thresholds["fci_high"]:
            return FrictionAction.AUTO_CONFIRM
        elif fci >= self.thresholds["fci_medium"]:
            return FrictionAction.RECOMMEND_ONLY
        else:
            return FrictionAction.SUGGEST_ALTERNATIVE

        self._rng = np.random.RandomState(42)

class MetricsCollector:
    """Base interface for metrics collection."""
    def add_prediction(self, prediction_id: str, result: Dict[str, Any]):
        raise NotImplementedError

    def add_outcome(self, prediction_id: str, fit_success: bool, return_reason: Optional[str] = None):
        raise NotImplementedError

    def compute_metrics(self) -> Dict[str, float]:
        raise NotImplementedError

class InMemoryMetricsCollector(MetricsCollector):
    """Simple in‑memory collector for testing."""
    def __init__(self):
        self.predictions = {}  # id -> result
        self.outcomes = {}     # id -> (success, reason)

    def add_prediction(self, prediction_id: str, result: Dict[str, Any]):
        self.predictions[prediction_id] = result

    def add_outcome(self, prediction_id: str, fit_success: bool, return_reason: Optional[str] = None):
        self.outcomes[prediction_id] = (fit_success, return_reason)

    def compute_metrics(self) -> Dict[str, float]:
        # Join predictions and outcomes
        ids = set(self.predictions.keys()) & set(self.outcomes.keys())
        if not ids:
            return {}
        y_true = []
        y_pred = []
        ci_widths = []
        for pid in ids:
            y_true.append(1 if self.outcomes[pid][0] else 0)
            y_pred.append(self.predictions[pid]["fci_final"])
            ci_widths.append(self.predictions[pid]["ci_width"])
        y_true = np.array(y_true)
        y_pred = np.array(y_pred)
        ci_widths = np.array(ci_widths)

        metrics = {}
        metrics["brier_score"] = float(np.mean((y_true - y_pred) ** 2))
        metrics["ece"] = self._expected_calibration_error(y_true, y_pred)
        metrics["mean_ci_width"] = float(np.mean(ci_widths))
        metrics["fit_return_rate"] = 1.0 - float(np.mean(y_true))
        metrics["false_confidence_rate"] = self._false_confidence_rate(y_true, y_pred, threshold=0.8)
        return metrics

    def _expected_calibration_error(self, y_true, y_pred, n_bins=10):
        bins = np.linspace(0, 1, n_bins + 1)
        bin_indices = np.digitize(y_pred, bins[1:-1])
        ece = 0.0
        for i in range(n_bins):
            mask = (bin_indices == i)
            if np.any(mask):
                bin_acc = np.mean(y_true[mask])
                bin_conf = np.mean(y_pred[mask])
                ece += np.abs(bin_acc - bin_conf) * np.sum(mask) / len(y_true)
        return float(ece)

    def _false_confidence_rate(self, y_true, y_pred, threshold=0.8):
        high_conf = y_pred >= threshold
        if not np.any(high_conf):
            return 0.0
        false_pos = np.sum((y_pred[high_conf] >= threshold) & (y_true[high_conf] == 0))
        return float(false_pos / np.sum(high_conf))

class PostgresMetricsCollector(MetricsCollector):
    """Persistent metrics collector using PostgreSQL."""
    def __init__(self, conn_pool: 'psycopg2.pool.SimpleConnectionPool'):
        self.pool = conn_pool
        self._init_db()

    def _init_db(self):
        conn = self.pool.getconn()
        try:
            with conn.cursor() as cur:
                cur.execute("""
                    CREATE TABLE IF NOT EXISTS fci_predictions (
                        prediction_id TEXT PRIMARY KEY,
                        result JSONB,
                        created_at TIMESTAMP DEFAULT NOW()
                    );
                    CREATE TABLE IF NOT EXISTS fci_outcomes (
                        prediction_id TEXT PRIMARY KEY REFERENCES fci_predictions(prediction_id),
                        fit_success BOOLEAN,
                        return_reason TEXT,
                        created_at TIMESTAMP DEFAULT NOW()
                    );
                """)
                conn.commit()
        finally:
            self.pool.putconn(conn)

    def add_prediction(self, prediction_id: str, result: Dict[str, Any]):
        conn = self.pool.getconn()
        try:
            with conn.cursor() as cur:
                cur.execute(
                    "INSERT INTO fci_predictions (prediction_id, result) VALUES (%s, %s)",
                    (prediction_id, json.dumps(result))
                )
                conn.commit()
        finally:
            self.pool.putconn(conn)

    def add_outcome(self, prediction_id: str, fit_success: bool, return_reason: Optional[str] = None):
        conn = self.pool.getconn()
        try:
            with conn.cursor() as cur:
                cur.execute(
                    "INSERT INTO fci_outcomes (prediction_id, fit_success, return_reason) VALUES (%s, %s, %s)",
                    (prediction_id, fit_success, return_reason)
                )
                conn.commit()
        finally:
            self.pool.putconn(conn)

    def compute_metrics(self) -> Dict[str, float]:
        conn = self.pool.getconn()
        try:
            with conn.cursor() as cur:
                cur.execute("""
                    SELECT p.result->>'fci_final' as fci,
                           p.result->>'ci_width' as ci_width,
                           o.fit_success
                    FROM fci_predictions p
                    JOIN fci_outcomes o ON p.prediction_id = o.prediction_id
                """)
                rows = cur.fetchall()
                if not rows:
                    return {}
                y_true = []
                y_pred = []
                ci_widths = []
                for row in rows:
                    y_true.append(1 if row[2] else 0)
                    y_pred.append(float(row[0]))
                    ci_widths.append(float(row[1]))
                y_true = np.array(y_true)
                y_pred = np.array(y_pred)
                ci_widths = np.array(ci_widths)
                # Compute metrics (same as in‑memory)
                collector = InMemoryMetricsCollector()  # reuse logic
                # We'd need to set its internal state; simpler: compute directly
                metrics = {}
                metrics["brier_score"] = float(np.mean((y_true - y_pred) ** 2))
                metrics["ece"] = collector._expected_calibration_error(y_true, y_pred)
                metrics["mean_ci_width"] = float(np.mean(ci_widths))
                metrics["fit_return_rate"] = 1.0 - float(np.mean(y_true))
                metrics["false_confidence_rate"] = collector._false_confidence_rate(y_true, y_pred, 0.8)
                return metrics
        finally:
            self.pool.putconn(conn)

class FitConfidenceIndex:
    def __init__(self,
                 structural_weights: Tuple[float, float, float, float] = (0.3, 0.25, 0.25, 0.2),
                 bayesian: Optional[BayesianLayer] = None,
                 neural_residual: Optional[NeuralResidual] = None,
                 calibrator: Optional[FastCalibrator] = None,
                 friction_policy: Optional[FrictionPolicy] = None,
                 metrics_collector: Optional[MetricsCollector] = None):
        self.structural_weights = structural_weights
        self.bayesian = bayesian or BayesianLayer()
        self.neural = neural_residual
        self.calibrator = calibrator
        self.friction = friction_policy or RuleBasedFrictionPolicy()
        self.metrics = metrics_collector or InMemoryMetricsCollector()

    def predict(self,
                body: BodyModel,
                garment: GarmentModel,
                brand: BrandData,
                user: UserHistory,
                category: str,
                brand_name: str,
                user_id: str,
                extra_features: Optional[Dict[str, Any]] = None,
                prediction_id: Optional[str] = None) -> Dict[str, Any]:
        """Run FCI pipeline and return result."""
        try:
            w1, w2, w3, w4 = self.structural_weights
            s_struct = w1 * body.bmc + w2 * garment.gsr + w3 * brand.bfs + w4 * user.phfa
            s_struct = np.clip(s_struct, 0.0, 1.0)

            s_bayes = self.bayesian.predict(category, brand_name, user_id)

            delta = 0.0
            if self.neural is not None:
                features = {
                    "bmc": body.bmc,
                    "gsr": garment.gsr,
                    "bfs": brand.bfs,
                    "phfa": user.phfa,
                    "s_struct": s_struct,
                    "s_bayes": s_bayes,
                    "price_bucket": garment.price_bucket,
                    "rigidity": garment.rigidity_index,
                    "brand_volatility": brand.brand_volatility,
                    "user_tolerance": user.user_tolerance_index,
                }
                if extra_features:
                    features.update(extra_features)
                delta = self.neural.predict(features)
            s_hybrid = np.clip(s_bayes + delta, 0.0, 1.0)

            if self.calibrator is not None:
                fci_calibrated = self.calibrator.predict_proba(np.array([s_hybrid]))[0]
                ci_lower, ci_upper = self.calibrator.confidence_interval(np.array([s_hybrid]))
                ci_lower, ci_upper = ci_lower[0], ci_upper[0]
            else:
                fci_calibrated = s_hybrid
                ci_lower, ci_upper = s_hybrid - 0.05, s_hybrid + 0.05
                ci_lower, ci_upper = np.clip([ci_lower, ci_upper], 0.0, 1.0)

            fci_final = fci_calibrated
            if body.bmc < 0.6:
                fci_final = min(fci_final, 0.75)
            if garment.gsr < 0.5:
                fci_final = min(fci_final, 0.70)

            state = {
                "fci": fci_final,
                "ci_width": ci_upper - ci_lower,
                "bmc": body.bmc,
                "gsr": garment.gsr,
                "price": garment.price_bucket,
                "brand_volatility": brand.brand_volatility,
                "rigidity": garment.rigidity_index,
                "user_tolerance": user.user_tolerance_index,
            }
            action = self.friction.choose_action(state)

            result = {
                "prediction_id": prediction_id,
                "bmc": body.bmc,
                "gsr": garment.gsr,
                "bfs": brand.bfs,
                "phfa": user.phfa,
                "s_struct": s_struct,
                "s_bayes": s_bayes,
                "delta_residual": delta,
                "s_hybrid": s_hybrid,
                "fci_calibrated": fci_calibrated,
                "fci_final": fci_final,
                "ci_lower": ci_lower,
                "ci_upper": ci_upper,
                "ci_width": ci_upper - ci_lower,
                "friction_action": int(action),
                "friction_action_name": action.name,
                "timestamp": datetime.utcnow().isoformat(),
            }

            if prediction_id:
                self.metrics.add_prediction(prediction_id, result)

            logger.info(f"FCI final: {fci_final:.4f}, action: {action.name}")
            return result
        except Exception as e:
            logger.exception("FCI prediction failed")
            return {"prediction_id": prediction_id, "error": str(e), "fci_final": 0.5, "friction_action": 2}

    def record_outcome(self, prediction_id: str, fit_success: bool, return_reason: Optional[str] = None):
        self.metrics.add_outcome(prediction_id, fit_success, return_reason)
        logger.info(f"Outcome recorded: id={prediction_id}, success={fit_success}, reason={return_reason}")
