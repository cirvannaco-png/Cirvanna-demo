"""
Simulated Redis client for snapshot storage and caching.
"""
import asyncio
import time
from typing import Any, Dict, Optional

class RedisClient:
    _instance = None
    _data: Dict[str, Any] = {}
    _locks: Dict[str, asyncio.Lock] = {}

    @classmethod
    def get_instance(cls):
        if cls._instance is None:
            cls._instance = cls()
        return cls._instance

    async def set(self, key: str, value: Any, ttl: Optional[int] = None):
        self._data[key] = {
            'value': value,
            'expiry': time.time() + ttl if ttl else None
        }

    async def get(self, key: str) -> Optional[Any]:
        if key not in self._data:
            return None
        entry = self._data[key]
        if entry['expiry'] and time.time() > entry['expiry']:
            del self._data[key]
            return None
        return entry['value']

    async def delete(self, key: str):
        if key in self._data:
            del self._data[key]

    async def acquire_lock(self, lock_name: str, timeout: int = 10) -> bool:
        if lock_name not in self._locks:
            self._locks[lock_name] = asyncio.Lock()
        try:
            await asyncio.wait_for(self._locks[lock_name].acquire(), timeout=timeout)
            return True
        except asyncio.TimeoutError:
            return False

    async def release_lock(self, lock_name: str):
        if lock_name in self._locks and self._locks[lock_name].locked():
            self._locks[lock_name].release()

    _bucket: Dict[str, List[Dict]] = {}

    async def put_event(self, user_id: str, event: Dict) -> str:
        key = f"events/{user_id}/{uuid.uuid4()}.json"
        if user_id not in self._bucket:
            self._bucket[user_id] = []
        self._bucket[user_id].append(event)
        return key

        if user_id not in self._bucket:
            return []
        return [e for e in self._bucket[user_id] if e.get('timestamp', 0) > timestamp]

"""
CFRPActor – Single-threaded message processing lane.
"""
import asyncio
import logging
from typing import Callable, Optional, Dict

class CFRPActor:
    def __init__(self, user_id: str):
        self.user_id = user_id
        self.queue = asyncio.Queue(maxsize=1000)
        self.running = False
        self.task: Optional[asyncio.Task] = None
        self.signal_states: Dict = {}
        self.reinforcement_log: list = []
        logging.info(f"Actor created for {user_id}")

    async def start(self):
        if self.running:
            return
        self.running = True
        self.task = asyncio.create_task(self._run())

    async def stop(self):
        if not self.running:
            return
        self.running = False
        await self.queue.put(None)
        if self.task:
            await self.task

    async def _run(self):
        while self.running:
            try:
                msg = await self.queue.get()
                if msg is None:
                    break
                await msg()
                self.queue.task_done()
            except asyncio.CancelledError:
                break
            except Exception as e:
                logging.error(f"Actor {self.user_id} error: {e}")

    async def send(self, func: Callable):
        future = asyncio.Future()
        async def wrapper():
            try:
                result = await func()
                future.set_result(result)
            except Exception as e:
                future.set_exception(e)
        await self.queue.put(wrapper)
        return await future

"""
EpistemicInvariants – Mathematical guarantees for CFRP's reasoning.
"""
from decimal import Decimal
from typing import List

class EpistemicInvariantViolation(Exception):
    """Raised when an epistemic invariant is violated"""
    pass

class EpistemicInvariants:
    MAX_CONFIDENCE_JUMP = Decimal('0.3')
    MAX_DAILY_DECAY = Decimal('0.05')
    MIN_CONFIDENCE = Decimal('0.1')
    MAX_CFRP_INFLUENCE = Decimal('0.3')
    MIN_USI_AUTONOMY = Decimal('0.7')
    MAX_DOMINANCE_RATIO = Decimal('10.0')

    @classmethod
    def validate_confidence_change(cls, old: Decimal, new: Decimal, success: bool):
        delta = new - old
        if success:
            max_increase = cls.MAX_CONFIDENCE_JUMP * (Decimal('1') - old)
            if delta > max_increase:
                raise EpistemicInvariantViolation(
                    f"Confidence increase {delta} > allowed {max_increase}"
                )
        else:
            max_decrease = cls.MAX_CONFIDENCE_JUMP * old
            if delta < -max_decrease:
                raise EpistemicInvariantViolation(
                    f"Confidence decrease {-delta} > allowed {max_decrease}"
                )

    @classmethod
    def validate_decay(cls, old: Decimal, new: Decimal, days_elapsed: Decimal):
        if new < cls.MIN_CONFIDENCE:
            raise EpistemicInvariantViolation(
                f"Confidence {new} < minimum {cls.MIN_CONFIDENCE}"
            )
        if days_elapsed > Decimal('0'):
            max_decay = cls.MAX_DAILY_DECAY * days_elapsed * old
            actual_decay = old - new
            if actual_decay > max_decay:
                raise EpistemicInvariantViolation(
                    f"Decay {actual_decay} > allowed {max_decay}"
                )

    @classmethod
    def validate_cfrp_influence(cls, usi_weight: Decimal, cfrp_adjustment: Decimal) -> Decimal:
        bounded = max(-cls.MAX_CFRP_INFLUENCE, min(cls.MAX_CFRP_INFLUENCE, cfrp_adjustment))
        final = usi_weight * (Decimal('1') + bounded)
        min_weight = usi_weight * cls.MIN_USI_AUTONOMY
        if final < min_weight:
            return min_weight
        return final

    @classmethod
    def validate_no_dominance(cls, weights: List[Decimal]):
        if len(weights) < 2:
            return
        non_zero = [w for w in weights if w > Decimal('0')]
        if len(non_zero) < 2:
            return
        max_w = max(non_zero)
        min_w = min(non_zero)
        if min_w == Decimal('0'):
            return
        ratio = max_w / min_w
        if ratio > cls.MAX_DOMINANCE_RATIO:
            raise EpistemicInvariantViolation(
                f"Signal dominance ratio {ratio} > allowed {cls.MAX_DOMINANCE_RATIO}"
            )

"""
EpistemicContext – Adaptive context for CFRP.
"""
import time
from dataclasses import dataclass, field
from typing import Dict, Optional

@dataclass
class EpistemicContext:
    cultural_sensitivity: float = 0.0
    wardrobe_stability: float = 0.5
    exploration_budget: float = 0.2
    signal_volatility: float = 0.0
    last_context_update: float = field(default_factory=time.time)

    @classmethod
    async def from_user_id(cls, user_id: str, redis) -> 'EpistemicContext':
        key = f"context:{user_id}"
        cached = await redis.get(key)
        if cached:
            return EpistemicContext(**cached)
        return EpistemicContext()

    async def save(self, user_id: str, redis):
        key = f"context:{user_id}"
        await redis.set(key, {
            'cultural_sensitivity': self.cultural_sensitivity,
            'wardrobe_stability': self.wardrobe_stability,
            'exploration_budget': self.exploration_budget,
            'signal_volatility': self.signal_volatility,
            'last_context_update': self.last_context_update
        }, ttl=86400)

    def compute_adaptive_rates(self) -> Dict[str, float]:
        base_lr = 0.1
        learning_rate = base_lr * (1 - self.cultural_sensitivity * 0.5)
        base_decay = 0.05
        confidence_decay = base_decay * (1 - self.wardrobe_stability * 0.3)
        exploration_boost = self.exploration_budget * 0.2
        return {
            'learning_rate': max(0.01, min(0.3, learning_rate)),
            'confidence_decay': max(0.01, min(0.1, confidence_decay)),
            'exploration_boost': exploration_boost,
            'volatility_multiplier': 1 + self.signal_volatility * 0.5
        }

"""
LearningPrior – Hierarchical learning priors.
"""
import numpy as np
from dataclasses import dataclass, field
from typing import Tuple, Dict, Optional

@dataclass
class LearningPrior:
    global_prior: np.ndarray = field(default_factory=lambda: np.array([0.5]))
    segment_prior: np.ndarray = field(default_factory=lambda: np.array([0.5]))
    user_posterior: np.ndarray = field(default_factory=lambda: np.array([0.5]))

    ALPHA_RANGE = (0.4, 0.8)
    BETA_RANGE = (0.1, 0.4)
    GAMMA_RANGE = (0.05, 0.2)

    @classmethod
    def compute_posterior(cls, user_posterior: np.ndarray,
                          segment_prior: Optional[np.ndarray] = None,
                          global_prior: Optional[np.ndarray] = None,
                          conflict_threshold: float = 0.3) -> Tuple[np.ndarray, Dict[str, float]]:
        if segment_prior is None:
            segment_prior = np.array([0.5])
        if global_prior is None:
            global_prior = np.array([0.5])

        has_conflict = False
        if np.abs(user_posterior[0] - segment_prior[0]) > conflict_threshold:
            has_conflict = True

        if has_conflict or segment_prior is None:
            alpha = 0.7
            beta = 0.2
            gamma = 0.1
        else:
            alpha = 0.6
            beta = 0.3
            gamma = 0.1

        total = alpha + beta + gamma
        alpha, beta, gamma = alpha/total, beta/total, gamma/total

        posterior = (alpha * user_posterior + beta * segment_prior + gamma * global_prior)
        posterior = np.clip(posterior, 0, 1)

        metadata = {
            'alpha': alpha,
            'beta': beta,
            'gamma': gamma,
            'has_conflict': has_conflict,
            'learning_rate_multiplier': 0.5 if has_conflict else 1.0
        }
        return posterior, metadata

"""
EpistemicState – Confidence state with reinforcement and decay.
"""
import time
import math
from decimal import Decimal
from dataclasses import dataclass, field
from typing import Dict, Optional, Any

from src.core.epistemic_invariants import EpistemicInvariants
from src.core.learning_prior import LearningPrior
from src.core.epistemic_context import EpistemicContext

@dataclass
class EpistemicState:
    signal_id: str
    confidence: Decimal = Decimal('0.5')
    rolling_baseline: Decimal = Decimal('0.5')
    last_reinforced: float = field(default_factory=time.time)
    reinforcement_count: int = 0
    baseline_window: int = 100
    learning_metadata: Dict[str, Any] = field(default_factory=dict)
    context_features: Dict[str, float] = field(default_factory=dict)

    def reinforce(self, success: bool, value: Decimal,
                  prior: Optional[LearningPrior] = None,
                  context: Optional[EpistemicContext] = None) -> 'EpistemicState':
        success_value = Decimal('1') if success else Decimal('0')

        if self.reinforcement_count < self.baseline_window:
            weight = Decimal('1') / Decimal(str(self.reinforcement_count + 1))
            new_baseline = self.rolling_baseline * (Decimal('1') - weight) + success_value * weight
        else:
            new_baseline = self.rolling_baseline * Decimal('0.99') + success_value * Decimal('0.01')

        if context:
            rates = context.compute_adaptive_rates()
            learning_rate = Decimal(str(rates['learning_rate']))
        else:
            learning_rate = Decimal('0.1')

        if prior:
            user_posterior = np.array([float(self.confidence)])
            posterior, metadata = LearningPrior.compute_posterior(
                user_posterior=user_posterior,
                segment_prior=prior.segment_prior,
                global_prior=prior.global_prior
            )
            if metadata.get('has_conflict', False):
                learning_rate *= Decimal('0.5')
            base_confidence = Decimal(str(posterior[0]))
            self.learning_metadata = metadata
        else:
            base_confidence = self.confidence

        if success:
            if success_value > new_baseline:
                increase = learning_rate * (Decimal('1') - base_confidence)
                new_confidence = base_confidence + increase * value
            else:
                new_confidence = base_confidence
        else:
            decrease = learning_rate * base_confidence
            new_confidence = base_confidence - decrease * value

        new_confidence = max(Decimal('0'), min(Decimal('1'), new_confidence))

        EpistemicInvariants.validate_confidence_change(
            old=self.confidence,
            new=new_confidence,
            success=success
        )

        return EpistemicState(
            signal_id=self.signal_id,
            confidence=new_confidence,
            rolling_baseline=new_baseline,
            last_reinforced=time.time(),
            reinforcement_count=self.reinforcement_count + 1,
            learning_metadata=self.learning_metadata,
            context_features=self.context_features
        )

    def decay(self, current_time: float, context: Optional[EpistemicContext] = None) -> 'EpistemicState':
        days_elapsed = Decimal(str((current_time - self.last_reinforced) / 86400))

        if days_elapsed <= Decimal('0'):
            return self

        if context:
            rates = context.compute_adaptive_rates()
            decay_rate = Decimal(str(rates['confidence_decay']))
        else:
            decay_rate = Decimal('0.05')

        halflife_days = 30 / (1 + float(decay_rate) * 10)
        decay_factor = Decimal(str(math.exp(-float(days_elapsed / Decimal(str(halflife_days))))))

        new_confidence = self.confidence * decay_factor

        EpistemicInvariants.validate_decay(
            old=self.confidence,
            new=new_confidence,
            days_elapsed=days_elapsed
        )

        return EpistemicState(
            signal_id=self.signal_id,
            confidence=new_confidence,
            rolling_baseline=self.rolling_baseline,
            last_reinforced=self.last_reinforced,
            reinforcement_count=self.reinforcement_count,
            learning_metadata=self.learning_metadata,
            context_features=self.context_features
        )

"""
CFRPReinforcementLog – Event-sourced log (non-enterprise).
"""
import time
import uuid
from typing import Dict, List, Optional

class CFRPReinforcementLog:
    def __init__(self, user_id: str):
        self.user_id = user_id
        self.events: List[Dict] = []
        self.last_snapshot_time = time.time()
        self.snapshot_interval = 300

    def append(self, signal_id: str, success: bool, value: float):
        event = {
            'timestamp': time.time(),
            'signal_id': signal_id,
            'success': success,
            'value': value,
            'event_id': str(uuid.uuid4())
        }
        self.events.append(event)

    def get_since(self, since_timestamp: float) -> List[Dict]:
        return [e for e in self.events if e['timestamp'] > since_timestamp]

    def needs_snapshot(self) -> bool:
        return time.time() - self.last_snapshot_time > self.snapshot_interval

    def create_snapshot(self, signal_states: Dict) -> Dict:
        self.last_snapshot_time = time.time()
        return {
            'timestamp': self.last_snapshot_time,
            'user_id': self.user_id,
            'signal_states': {
                sid: {
                    'confidence': float(s.confidence),
                    'rolling_baseline': float(s.rolling_baseline),
                    'last_reinforced': s.last_reinforced,
                    'reinforcement_count': s.reinforcement_count
                } for sid, s in signal_states.items()
            },
            'last_event_id': self.events[-1]['event_id'] if self.events else None
        }

"""
CFRPInstance – Single authoritative CFRP instance per user.
"""
import asyncio
import time
import logging
from decimal import Decimal
from typing import Dict, List, Optional

from src.actor.cfrp_actor import CFRPActor
from src.core.epistemic_state import EpistemicState
from src.core.reinforcement_log import CFRPReinforcementLog
from src.infrastructure.redis_client import RedisClient

class CFRPInstance:
    def __init__(self, user_id: str):
        self.user_id = user_id
        self.actor = CFRPActor(user_id)
        self.signal_states: Dict[str, EpistemicState] = {}
        self.reinforcement_log = CFRPReinforcementLog(user_id)
        self.running = False
        self.maintenance_task: Optional[asyncio.Task] = None
        self.shutdown_event = asyncio.Event()
        self.redis = RedisClient.get_instance()

    async def start(self):
        if self.running:
            return
        await self.actor.start()
        await self._load_state()
        self.running = True
        self.maintenance_task = asyncio.create_task(self._maintenance_loop())

    async def stop(self):
        if not self.running:
            return
        self.running = False
        self.shutdown_event.set()
        if self.maintenance_task:
            self.maintenance_task.cancel()
            try:
                await self.maintenance_task
            except asyncio.CancelledError:
                pass
        await self.actor.stop()
        await self._flush_state()

    async def _load_state(self):
        # In production: load from Redis snapshot + replay events
        snapshot = await self.redis.get(f"snapshot:{self.user_id}")
        if snapshot:
            # Reconstruct signal_states from snapshot
            pass

    async def _flush_state(self):
        if self.reinforcement_log.needs_snapshot():
            snapshot = self.reinforcement_log.create_snapshot(self.signal_states)
            await self.redis.set(f"snapshot:{self.user_id}", snapshot)

    async def _maintenance_loop(self):
        while not self.shutdown_event.is_set():
            try:
                await asyncio.wait(
                    [asyncio.create_task(asyncio.sleep(60)),
                     asyncio.create_task(self.shutdown_event.wait())],
                    return_when=asyncio.FIRST_COMPLETED
                )
                if self.shutdown_event.is_set():
                    break
                await self._perform_maintenance()
            except asyncio.CancelledError:
                break

    async def _perform_maintenance(self):
        for signal_id in list(self.signal_states.keys()):
            await self.actor.send(lambda: self._apply_decay(signal_id))
        if self.reinforcement_log.needs_snapshot():
            snapshot = self.reinforcement_log.create_snapshot(self.signal_states)
            await self.redis.set(f"snapshot:{self.user_id}", snapshot)

    async def _apply_decay(self, signal_id: str):
        if signal_id in self.signal_states:
            state = self.signal_states[signal_id]
            self.signal_states[signal_id] = state.decay(time.time())

    async def reinforce(self, signal_id: str, success: bool, value: float = 1.0) -> Dict:
        return await self.actor.send(lambda: self._reinforce_internal(signal_id, success, value))

    async def _reinforce_internal(self, signal_id: str, success: bool, value: float) -> Dict:
        if signal_id not in self.signal_states:
            self.signal_states[signal_id] = EpistemicState(signal_id=signal_id)
        old_state = self.signal_states[signal_id]
        new_state = old_state.reinforce(success, Decimal(str(value)))
        self.signal_states[signal_id] = new_state
        self.reinforcement_log.append(signal_id, success, value)
        return {
            'success': True,
            'old_confidence': float(old_state.confidence),
            'new_confidence': float(new_state.confidence),
            'reinforcement_count': new_state.reinforcement_count
        }

    async def get_advice(self, signal_ids: List[str]) -> Dict:
        return await self.actor.send(lambda: self._get_advice_internal(signal_ids))

    async def _get_advice_internal(self, signal_ids: List[str]) -> Dict:
        advisories = {}
        for signal_id in signal_ids:
            if signal_id not in self.signal_states:
                self.signal_states[signal_id] = EpistemicState(signal_id=signal_id)
            state = self.signal_states[signal_id]
            state = state.decay(time.time())
            self.signal_states[signal_id] = state
            advisory = self._generate_advisory(state)
            advisories[signal_id] = advisory
        return {'advisories': advisories}

    def _generate_advisory(self, state: EpistemicState) -> Dict:
        confidence_weight = float(state.confidence) - 0.5
        days_since = (time.time() - state.last_reinforced) / 86400
        recency_bonus = max(0, 0.1 * (1 - days_since / 7))
        data_bonus = min(0.1, state.reinforcement_count / 100)
        total_adjustment = confidence_weight + recency_bonus + data_bonus
        return {
            'confidence_weight': confidence_weight,
            'recency_bonus': recency_bonus,
            'data_bonus': data_bonus,
            'total_cfrp_adjustment': total_adjustment,
            'epistemic_notes': self._get_epistemic_notes(state)
        }

    def _get_epistemic_notes(self, state: EpistemicState) -> List[str]:
        notes = []
        if state.reinforcement_count < 10:
            notes.append("Limited data – high uncertainty")
        elif state.reinforcement_count > 100:
            notes.append("Well‑established signal")
        if state.confidence < Decimal('0.3'):
            notes.append("Low confidence – experimental")
        elif state.confidence > Decimal('0.8'):
            notes.append("High confidence – reliable")
        return notes

"""
CFRPManager – Manages CFRP instances with sticky sessions.
"""
import asyncio
import logging
from typing import Dict

from src.core.cfrp_instance import CFRPInstance

class CFRPManager:
    def __init__(self):
        self.instances: Dict[str, CFRPInstance] = {}
        self.lock = asyncio.Lock()
        self.running = False
        self.shutdown_event = asyncio.Event()

    async def get_instance(self, user_id: str) -> CFRPInstance:
        async with self.lock:
            if user_id not in self.instances:
                instance = CFRPInstance(user_id)
                await instance.start()
                self.instances[user_id] = instance
            return self.instances[user_id]

    async def start(self):
        self.running = True
        logging.info("CFRP Manager started")

    async def shutdown(self):
        if not self.running:
            return
        self.running = False
        self.shutdown_event.set()
        async with self.lock:
            shutdown_tasks = [inst.stop() for inst in self.instances.values()]
            if shutdown_tasks:
                await asyncio.gather(*shutdown_tasks, return_exceptions=True)
            self.instances.clear()
        logging.info("CFRP Manager shutdown complete")

"""
USIIntegration – Applies CFRP's epistemic gravity to USI suggestions.
"""
from decimal import Decimal
from typing import List, Dict

from src.core.cfrp_manager import CFRPManager
from src.core.epistemic_invariants import EpistemicInvariants

class USIIntegration:
    def __init__(self, cfrp_manager: CFRPManager):
        self.cfrp_manager = cfrp_manager

    async def get_weighted_suggestions(self, user_id: str, raw_suggestions: List[Dict]) -> List[Dict]:
        instance = await self.cfrp_manager.get_instance(user_id)
        signal_ids = [s['signal'] for s in raw_suggestions]
        advice_result = await instance.get_advice(signal_ids)
        advisories = advice_result.get('advisories', {})

        enhanced = []
        weights = []
        for suggestion in raw_suggestions:
            signal_id = suggestion['signal']
            usi_weight = Decimal(str(suggestion['usi_weight']))
            advisory = advisories.get(signal_id, {})
            cfrp_adjustment = Decimal(str(advisory.get('total_cfrp_adjustment', 0.0)))

            final_weight = EpistemicInvariants.validate_cfrp_influence(
                usi_weight=usi_weight,
                cfrp_adjustment=cfrp_adjustment
            )

            enhanced_suggestion = {
                **suggestion,
                'final_weight': float(final_weight),
                'cfrp_adjustment': float(cfrp_adjustment),
                'epistemic_notes': advisory.get('epistemic_notes', [])
            }
            enhanced.append(enhanced_suggestion)
            weights.append(final_weight)

        EpistemicInvariants.validate_no_dominance(weights)
        enhanced.sort(key=lambda x: x['final_weight'], reverse=True)
        return enhanced

    async def process_feedback(self, user_id: str, signal_id: str, success: bool, value: float = 1.0):
        instance = await self.cfrp_manager.get_instance(user_id)
        return await instance.reinforce(signal_id, success, value)

"""
Skin biology: layers, melanin types.
"""
from enum import Enum

class SkinLayer(Enum):
    EPIDERMIS = "epidermis"

    II = 2
    III = 3

    TONE_4 = 4
    TONE_5 = 5
    TONE_6 = 6
    TONE_7 = 7
    TONE_8 = 8
    TONE_9 = 9
    TONE_10 = 10

class SkinToneAnalyzer:
    @staticmethod
    def estimate_fitzpatrick_from_lab(lab_image: np.ndarray) -> FitzpatrickType:
        L_mean = np.mean(lab_image[..., 0])
        if L_mean > 80:
            return FitzpatrickType.I
        elif L_mean > 70:
            return FitzpatrickType.II
        elif L_mean > 60:
            return FitzpatrickType.III
        elif L_mean > 50:
            return FitzpatrickType.IV
        elif L_mean > 40:
            return FitzpatrickType.V
        else:
            return FitzpatrickType.VI

    @staticmethod
    def estimate_monk_from_rgb(rgb_image: np.ndarray) -> MonkSkinTone:
        gray = np.dot(rgb_image[...,:3], [0.299, 0.587, 0.114])
        mean_lum = np.mean(gray)
        idx = int(np.clip(mean_lum / 255 * 10, 0, 9))
        return MonkSkinTone(idx + 1)

        else:

"""
Skin texture: pore density, wrinkle depth, roughness.
"""
import numpy as np
from scipy import ndimage

class TextureAnalyzer:
    @staticmethod
    def pore_density(gray_image: np.ndarray, threshold: float = 0.7) -> float:
        edges = ndimage.sobel(gray_image)
        pores = (edges > threshold * edges.max()).sum()
        return pores / (gray_image.shape[0] * gray_image.shape[1])

    @staticmethod
    def wrinkle_depth(gray_image: np.ndarray) -> float:
        local_std = ndimage.generic_filter(gray_image, np.std, size=5)
        return float(np.mean(local_std))

    @staticmethod
    def surface_roughness(gray_image: np.ndarray) -> float:
        return float(np.mean(np.abs(gray_image - np.mean(gray_image))))

class StructuredLogger:

                'logger': record.name,
                'message': record.getMessage(),

                'shard_id': getattr(record, 'shard_id', None),

            }
            if record.exc_info:
                log_record['exception'] = self.formatException(record.exc_info)
            return json.dumps(log_record)

        self.logger.info(message, extra={'extra': context})

    def warning(self, message: str, **context):

"""
AnomalyDetector – Statistical anomaly detection.
"""
import numpy as np
from collections import deque
from typing import Dict

class AnomalyDetector:
    def __init__(self, window_size: int = 100, sigma_threshold: float = 3.0):
        self.window_size = window_size
        self.sigma_threshold = sigma_threshold
        self.history: Dict[str, deque] = {}

    async def check_anomaly(self, user_id: str, old_conf: float, new_conf: float, metadata=None) -> Dict:
        delta = new_conf - old_conf
        if user_id not in self.history:
            self.history[user_id] = deque(maxlen=self.window_size)
        self.history[user_id].append(delta)
        if len(self.history[user_id]) < 10:
            return {"anomaly": False, "reason": "insufficient_data"}
        mean = np.mean(self.history[user_id])
        std = np.std(self.history[user_id])
        if std == 0:
            return {"anomaly": False, "reason": "zero_std"}
        z_score = (delta - mean) / std
        is_anomaly = abs(z_score) > self.sigma_threshold
        return {"anomaly": is_anomaly, "z_score": float(z_score)}

"""
LRUCache – Simple LRU cache.
"""
from collections import OrderedDict
from typing import Any, Optional

class LRUCache:
    def __init__(self, maxsize: int = 10000):
        self.maxsize = maxsize
        self.cache = OrderedDict()

    def get(self, key: str) -> Optional[Any]:
        if key in self.cache:
            self.cache.move_to_end(key)
            return self.cache[key]
        return None

    def set(self, key: str, value: Any, ttl: int = 300):
        if len(self.cache) >= self.maxsize:
            self.cache.popitem(last=False)
        self.cache[key] = value
        self.cache.move_to_end(key)

import asyncio
from typing import Any

class ConnectionPool:
    def __init__(self, max_connections: int = 100):
        self.max_connections = max_connections
        self.connections = asyncio.Queue(maxsize=max_connections)
        self.in_use = set()

    async def acquire(self) -> Any:
        if not self.connections.empty():
            conn = await self.connections.get()
        elif len(self.in_use) < self.max_connections:
            conn = await self._create_connection()
        else:
            conn = await self.connections.get()
        self.in_use.add(id(conn))
        return conn

    async def release(self, conn: Any):
        if id(conn) in self.in_use:
            self.in_use.remove(id(conn))
            await self.connections.put(conn)
