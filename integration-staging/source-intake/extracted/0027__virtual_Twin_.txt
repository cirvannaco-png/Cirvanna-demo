class MeasurementExtractorInterface:

class StubMeasurementExtractor(MeasurementExtractorInterface):

class MeasurementEngine:
    """Refines measurements, applies brand corrections, and detects drift."""

    def __init__(self, extractor: MeasurementExtractorInterface, brand_corrections_path: Optional[Path] = None):
        self.extractor = extractor
        self.brand_corrections: Dict[str, Dict[str, float]] = {}
        self.brand_interaction_count: Dict[str, int] = {}
        self._load_brand_corrections(brand_corrections_path)

    def _load_brand_corrections(self, path: Optional[Path]) -> None:
        """Load correction matrices for pilot brands from a JSON file."""
        if path and path.exists():
            with open(path) as f:
                self.brand_corrections = json.load(f)
        else:
            # Default corrections for pilot brands (multiplicative factors)
            self.brand_corrections = {
                "brandA": {"chest": 1.02, "waist": 0.98, "hips": 1.01},
                "brandB": {"chest": 0.97, "waist": 1.03},
                # ... others would be loaded or default to identity
            }
            # Ensure all pilot brands have at least identity matrix
            for brand in Config.PILOT_BRANDS:
                if brand not in self.brand_corrections:
                    self.brand_corrections[brand] = {}

    def extract_measurements(self, twin: VirtualTwin) -> VirtualTwin:
        """Update twin.measurement_table using the extractor."""
        twin.measurement_table = self.extractor.extract(twin)
        return twin

    def apply_brand_corrections(self, twin: VirtualTwin, brand_id: str) -> VirtualTwin:
        """
        Adjust twin measurements using brand‑specific correction matrix.
        For unknown brands, start with identity and learn after 5–10 interactions.
        """
        # Get correction matrix for this brand (may be empty)
        correction = self.brand_corrections.get(brand_id, {})
        if not correction:
            logging.info("Brand %s unknown; using identity corrections (to be learned).", brand_id)

        # Apply corrections (multiplicative; could be additive depending on spec)
        for region, factor in correction.items():
            if region in twin.measurement_table:
                twin.measurement_table[region] *= factor

        # Store the applied correction matrix in the twin for later reference
        twin.brand_correction_matrix[brand_id] = correction.copy()

        # Update interaction count for this brand
        self.brand_interaction_count[brand_id] = self.brand_interaction_count.get(brand_id, 0) + 1

        # If we have enough interactions for an unknown brand, trigger learning
        if brand_id not in self.brand_corrections and self.brand_interaction_count[brand_id] >= 5:
            self._learn_brand_correction(twin, brand_id)

        return twin

    def _learn_brand_correction(self, twin: VirtualTwin, brand_id: str) -> None:
        """
        Stub: after sufficient interactions, compute average fit discrepancy
        and update the brand correction matrix.
        """
        # In production: collect user feedback (fit ratings) for this brand,
        # compute average discrepancy between predicted and actual fit,
        # and adjust the correction matrix accordingly.
        logging.info("Learning correction for brand %s after %d interactions",
                     brand_id, self.brand_interaction_count[brand_id])
        # For now, just create a placeholder correction
        self.brand_corrections[brand_id] = {"chest": 1.0, "waist": 1.0}  # identity
        # Save updated corrections to disk
        self._save_brand_corrections()

    def _save_brand_corrections(self, path: Optional[Path] = None) -> None:
        """Persist updated brand corrections."""
        save_path = path or Path("./config/brand_corrections.json")
        save_path.parent.mkdir(parents=True, exist_ok=True)
        with open(save_path, "w") as f:
            json.dump(self.brand_corrections, f, indent=2)

    def detect_drift(self, twin: VirtualTwin, new_weight: Optional[float] = None,
                     misfit_count: int = 0) -> bool:
        """
        Detect if recalibration is needed:
        - weight change >5% compared to last recorded weight (stored in twin)
        - or misfit_count >= MISFIT_THRESHOLD
        """
        if twin.last_known_weight is not None and new_weight is not None:
            change = abs(new_weight - twin.last_known_weight) / twin.last_known_weight
            if change > Config.WEIGHT_CHANGE_THRESHOLD:
                logging.warning("Weight change %.1f%% > threshold", change * 100)
                return True
        if misfit_count >= Config.MISFIT_THRESHOLD:
            logging.warning("Three consecutive misfits detected")
            return True
        return False

    def trigger_recalibration(self, twin: VirtualTwin) -> VirtualTwin:
        """Mark twin as needing recalibration and log event."""
        twin.drift_status = True
        logging.info("Recalibration triggered for user %s", twin.user_id)
        # In production: send notification to front‑end (e.g., via WebSocket)
        return twin

class BehavioralEngine:
    """Manages user fit preferences and generates minimal personalisation embedding."""

    @staticmethod
    def update_behavioral_profile(twin: VirtualTwin, fit_bias: float = None,
                                  silhouette: str = None, stretch: float = None) -> VirtualTwin:
        if fit_bias is not None:
            twin.behavioral_profile.fit_bias = max(-1.0, min(1.0, fit_bias))
        if silhouette is not None:
            twin.behavioral_profile.silhouette_preference = silhouette
        if stretch is not None:
            twin.behavioral_profile.stretch_tolerance = max(0.0, min(1.0, stretch))
        return twin

    @staticmethod
    def get_embedding(twin: VirtualTwin) -> List[float]:
        """Return a minimal numeric embedding for personalisation."""
        return [
            twin.behavioral_profile.fit_bias,
            0.0 if twin.behavioral_profile.silhouette_preference == "structured" else 1.0,
            twin.behavioral_profile.stretch_tolerance
        ]

class ARClickItModule:
    """Displays garment on virtual twin with fallback overlay if confidence low."""

    def __init__(self, fabric_engine: FabricInteractionEngine):
        self.fabric_engine = fabric_engine

    def display_ar(self, twin: VirtualTwin, garment: Any) -> Dict[str, Any]:
        """
        Main AR display logic. Returns a response dict for the frontend.
        """
        confidence = twin.overall_confidence()
        response = {
            "mode": "physics",
            "fallback_reason": None,
            "lighting_correction": False
        }

        if confidence < Config.RECONFIDENCE_THRESHOLD:
            logging.warning("Confidence %.2f < threshold. Switching to overlay mode.", confidence)
            response["mode"] = "overlay"
            response["fallback_reason"] = "low_confidence"
            MetricsMonitor.log_ar_fallback_trigger("low_confidence")
        elif not self.fabric_engine.can_use_physics():
            logging.info("Device capability insufficient. Using visual‑only AR.")
            response["mode"] = "visual_only"
            response["fallback_reason"] = "device_capability"
            MetricsMonitor.log_ar_fallback_trigger("device_capability")
        else:
            # Attempt full physics
            success = self.fabric_engine.simulate_fabric_interaction(garment, twin)
            if not success:
                logging.error("Physics simulation failed. Falling back to overlay.")
                response["mode"] = "overlay"
                response["fallback_reason"] = "physics_failure"
                MetricsMonitor.log_ar_fallback_trigger("physics_failure")
            else:
                response["mode"] = "physics"

        # Always apply minimal lighting correction in any mode
        response["lighting_correction"] = True
        return response

    @staticmethod
    def minimal_lighting_correction(image: Any) -> Any:
        """Stub: apply minimal lighting normalisation before AR overlay."""
        # In production: use OpenCV to adjust brightness/contrast
        return image

class PersistenceManager:
    """Handles encrypted storage, versioning, and privacy‑compliant deletion."""

    def __init__(self, storage_path: Path = Config.STORAGE_PATH):
        self.storage_path = storage_path
        self.storage_path.mkdir(parents=True, exist_ok=True)
        self.cipher = Fernet(Config.ENCRYPTION_KEY.encode())

    def _encrypt_data(self, data: bytes) -> bytes:
        return self.cipher.encrypt(data)

    def _decrypt_data(self, data: bytes) -> bytes:
        return self.cipher.decrypt(data)

    def save_encrypted_twin(self, twin: VirtualTwin) -> None:
        """
        Store twin data with encryption.
        Version minor is auto‑incremented.
        """
        # Increment version minor
        twin.version_minor += 1

        # Convert twin to dict, handling dataclasses and datetime
        data = asdict(twin)
        # Convert datetime to isoformat
        if data["last_calibrated"]:
            data["last_calibrated"] = data["last_calibrated"].isoformat()

        # Serialize to JSON bytes
        json_bytes = json.dumps(data, indent=2).encode('utf-8')

        # Encrypt
        encrypted = self._encrypt_data(json_bytes)

        # Write to file
        filename = self.storage_path / f"{twin.user_id}_{twin.version}.enc"
        with open(filename, 'wb') as f:
            f.write(encrypted)

        logging.info("Saved twin %s version %s", twin.user_id, twin.version)

    def load_twin(self, user_id: str, version: Optional[str] = None) -> Optional[VirtualTwin]:
        """
        Load the latest or specified version of a twin.
        Returns None if not found.
        """
        if version is None:
            # Find latest version: list files for user_id, extract version, pick highest
            pattern = f"{user_id}_v*.enc"
            files = list(self.storage_path.glob(pattern))
            if not files:
                return None
            # Extract version numbers and pick max
            def version_key(path):
                # filename: user_id_v1.2.enc -> (1,2)
                stem = path.stem
                ver_part = stem.split('_v')[1]
                major, minor = map(int, ver_part.split('.'))
                return (major, minor)
            latest_file = max(files, key=version_key)
        else:
            latest_file = self.storage_path / f"{user_id}_{version}.enc"
            if not latest_file.exists():
                return None

        # Read and decrypt
        with open(latest_file, 'rb') as f:
            encrypted = f.read()
        try:
            decrypted = self._decrypt_data(encrypted)
            data = json.loads(decrypted.decode('utf-8'))
        except Exception as e:
            raise PersistenceError(f"Failed to decrypt/parse twin: {e}") from e

        # Convert back to VirtualTwin (handle datetime)
        if data.get("last_calibrated"):
            data["last_calibrated"] = datetime.fromisoformat(data["last_calibrated"])
        # Reconstruct BehavioralProfile if stored as dict
        if "behavioral_profile" in data and isinstance(data["behavioral_profile"], dict):
            data["behavioral_profile"] = BehavioralProfile(**data["behavioral_profile"])
        twin = VirtualTwin(**data)
        return twin

    def delete_user_data(self, user_id: str) -> None:
        """Permanently delete all personal twin data for a user."""
        pattern = f"{user_id}_*.enc"
        for file in self.storage_path.glob(pattern):
            file.unlink()
        logging.info("Deleted all personal data for user %s", user_id)

    @staticmethod
    def keep_aggregated_stats(twin: VirtualTwin) -> None:
        """Send de‑identified measurements to analytics (stub)."""
        # In production: publish to Kafka / metrics pipeline
        stats = {
            "measurements": twin.measurement_table,
            "confidence": twin.overall_confidence(),
            "timestamp": datetime.utcnow().isoformat()
        }
        logging.debug("Aggregated stats (de‑identified): %s", stats)

class VirtualTwinOrchestrator:
    """Coordinates all modules to process user input and produce an interactive twin."""

    def __init__(self,
                 reconstruction_model: ReconstructionModelInterface,
                 measurement_extractor: MeasurementExtractorInterface,
                 brand_corrections_path: Optional[Path] = None):
        self.user_input = UserInputModule()
        self.reconstruction = BodyReconstructionEngine(reconstruction_model)
        self.measurement = MeasurementEngine(measurement_extractor, brand_corrections_path)
        self.behavior = BehavioralEngine()
        # Device capability would be passed per request; we'll store a default or require per call
        self.fabric_engine = None  # will be created per request with device capability
        self.ar = None  # will be created per request
        self.persistence = PersistenceManager()
        self.metrics = MetricsMonitor()

    def process_user_input(self,
                           images: List[Any],
                           video: Optional[Any],
                           height: float,
                           weight: Optional[float],
                           device_capability: float,
                           brand_id: Optional[str] = None,
                           misfit_count: int = 0,
                           behavioral_updates: Optional[Dict] = None) -> VirtualTwin:
        """
        Full pipeline:
        1. Validate input
        2. Reconstruct body
        3. Extract measurements
        4. Calculate confidence
        5. Apply brand corrections
        6. Update behavioral profile
        7. Detect drift
        8. Trigger AR display (return AR response)
        9. Persist twin
        10. Log metrics
        """
        # Step 1: Validate input
        valid, _ = self.user_input.capture_and_validate(images, video)
        if not valid:
            raise ValidationError("Image validation failed. Please provide more angles.")

        # Step 2: Reconstruct body
        try:
            twin = self.reconstruction.reconstruct_body(images, video, height)
        except ReconstructionError as e:
            self.metrics.log_reconstruction_failure(str(e))
            raise

        # Step 3: Extract measurements (refine)
        twin = self.measurement.extract_measurements(twin)

        # Step 4: Assign confidence
        twin.confidence_map = self.reconstruction.calculate_confidence(twin)

        # Step 5: Apply brand corrections (if brand provided)
        if brand_id:
            twin = self.measurement.apply_brand_corrections(twin, brand_id)

        # Step 6: Update behavioral profile
        if behavioral_updates:
            twin = self.behavior.update_behavioral_profile(twin, **behavioral_updates)

        # Store current weight for future drift detection
        twin.last_known_weight = weight

        # Step 7: Detect drift and recalibrate if needed
        if self.measurement.detect_drift(twin, weight, misfit_count):
            twin = self.measurement.trigger_recalibration(twin)
            # In production: notify frontend via response field

        # Step 8: AR try‑on (simulate with a dummy garment)
        self.fabric_engine = FabricInteractionEngine(device_capability)
        self.ar = ARClickItModule(self.fabric_engine)
        dummy_garment = {"id": "tshirt_001", "brand": brand_id}
        ar_response = self.ar.display_ar(twin, dummy_garment)
        twin.ar_response = ar_response  # attach to twin for return

        # Step 9: Persist twin
        self.persistence.save_encrypted_twin(twin)
        self.persistence.keep_aggregated_stats(twin)

        # Step 10: Log metrics
        if twin.overall_confidence() < Config.RECONFIDENCE_THRESHOLD:
            self.metrics.log_ar_fallback_trigger("low_confidence_post_reconstruction")
        # Additional metrics can be logged here

        return twin
