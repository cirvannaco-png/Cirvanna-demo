"""
Xyra Gateway Orchestrator – routes requests to agents, handles timeouts, retries, tracing.
"""
import asyncio
import time
import uuid
from typing import Dict, Any, Optional, List
from dataclasses import dataclass, field
from ..config import TariConfig
from ..connection_pool import SecureConnectionPool
from ..cache.predictive_cache import PredictiveCache
from ..privacy.manager import PrivacyManager
from .agents.conversation_agent import ConversationAgent
from .agents.retrieval_agent import RetrievalAgent
from .agents.outfit_agent import OutfitAgent
from .agents.tryon_agent import TryOnAgent
from .agents.learning_agent import LearningAgent
from .memory.memory_service import MemoryService
from .knowledge_graph import FashionKnowledgeGraph

@dataclass
class OrchestrationContext:
    request_id: str
    user_id: str
    query: str
    context: Dict[str, Any]
    start_time: float = field(default_factory=time.perf_counter)
    metadata: Dict[str, Any] = field(default_factory=dict)

class XyraGatewayOrchestrator:
    def __init__(self, config: TariConfig, cache: PredictiveCache,
                 privacy_manager: PrivacyManager, pool: SecureConnectionPool):
        self.config = config
        self.cache = cache
        self.privacy_manager = privacy_manager
        self.pool = pool
        # Agents
        self.conversation_agent = ConversationAgent(config, cache, privacy_manager, pool)
        self.retrieval_agent = RetrievalAgent(config, cache, privacy_manager, pool)
        self.outfit_agent = OutfitAgent(config, cache, privacy_manager, pool)
        self.tryon_agent = TryOnAgent(config, cache, privacy_manager, pool)
        self.learning_agent = LearningAgent(config, cache, privacy_manager, pool)
        self.memory_service = MemoryService(config, cache, privacy_manager, pool)
        self.knowledge_graph = FashionKnowledgeGraph(config)
        # State
        self._active_requests: Dict[str, OrchestrationContext] = {}

    async def process(self, user_id: str, query: str, context: Dict[str, Any]) -> Dict[str, Any]:
        request_id = str(uuid.uuid4())
        orch_ctx = OrchestrationContext(
            request_id=request_id,
            user_id=user_id,
            query=query,
            context=context
        )
        self._active_requests[request_id] = orch_ctx
        try:
            # Step 1: Intent parsing (conversation agent)
            intent = await self.conversation_agent.parse_intent(query, context)
            # Step 2: Retrieve user memory
            user_profile = await self.memory_service.get_user_profile(user_id)
            # Step 3: Style embedding generation (via retrieval agent)
            style_embedding = await self.retrieval_agent.generate_style_embedding(intent, user_profile)
            # Step 4: Semantic product retrieval
            products = await self.retrieval_agent.retrieve_products(style_embedding, intent, user_profile)
            # Step 5: Outfit composition
            outfits = await self.outfit_agent.compose_outfits(products, intent, user_profile)
            # Step 6: XARI ranking
            ranked_outfits = await self.learning_agent.rank_outfits(outfits, user_id, context)
            # Step 7: Generate variations (if requested)
            if context.get("generate_variations"):
                ranked_outfits = await self.outfit_agent.generate_variations(ranked_outfits, intent)
            # Step 8: Trigger try-on asynchronously (if user selects a look)
            tryon_job = None
            if context.get("auto_tryon") and ranked_outfits:
                selected_outfit = ranked_outfits[0]
                tryon_job = await self.tryon_agent.queue_tryon(user_id, selected_outfit, context)
            # Step 9: Prepare response
            elapsed_ms = (time.perf_counter() - orch_ctx.start_time) * 1000
            response = {
                "request_id": request_id,
                "intent": intent,
                "products": products[:12],
                "outfits": ranked_outfits[:4],
                "tryon_job": tryon_job,
                "performance": {"elapsed_ms": elapsed_ms}
            }
            # Step 10: Async feedback learning
            asyncio.create_task(self.learning_agent.record_interaction(user_id, query, response, context))
            return response
        except Exception as e:
            return {"error": str(e), "request_id": request_id}
        finally:
            self._active_requests.pop(request_id, None)

    async def health_check(self) -> Dict[str, Any]:
        return {
            "orchestrator": "healthy",
            "agents": {
                "conversation": await self.conversation_agent.health(),
                "retrieval": await self.retrieval_agent.health(),
                "outfit": await self.outfit_agent.health(),
                "tryon": await self.tryon_agent.health(),
                "learning": await self.learning_agent.health(),
            },
            "memory": await self.memory_service.health(),
            "knowledge_graph": await self.knowledge_graph.health()
        }

        """
        # Step 1: Vector search (top 200)
        vector_results = await self._vector_search(style_embedding, top_k=self.config.RETRIEVAL_TOP_K)
        # Step 2: BM25 (text match on product descriptions)
        bm25_results = await self._bm25_search(intent, top_k=self.config.RETRIEVAL_TOP_K)
        # Step 3: Merge with reciprocal rank fusion (RRF)

        # Step 4: Apply behavioral boost (based on user profile)

        # Step 5: Apply freshness and inventory score

        # Step 6: Diversity re-ranking (MMR)
        diverse = self._maximal_marginal_relevance(style_embedding, final, lambda_d=self.config.DIVERSITY_LAMBDA)
        # Step 7: Return top 12

    def _reciprocal_rank_fusion(self, list_a: List[Dict], list_b: List[Dict], k: int = 60) -> List[Dict]:
        scores = {}
        for rank, item in enumerate(list_a):
            rid = item["product_id"]
            scores[rid] = scores.get(rid, 0) + 1.0 / (k + rank + 1)
        for rank, item in enumerate(list_b):
            rid = item["product_id"]
            scores[rid] = scores.get(rid, 0) + 1.0 / (k + rank + 1)
        sorted_ids = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        merged = []
        for rid, score in sorted_ids:
            # Find the full item from either list
            item = next((x for x in list_a if x["product_id"] == rid), None)
            if not item:
                item = next((x for x in list_b if x["product_id"] == rid), None)
            if item:
                item["fusion_score"] = score
                merged.append(item)
        return merged

    def _apply_behavioral_boost(self, items: List[Dict], user_profile: Dict) -> List[Dict]:
        # Boost products similar to user's liked styles
        liked_styles = user_profile.get("likedStyles", [])
        for item in items:
            item_tags = item.get("metadata", {}).get("styleTags", [])
            boost = 1.0
            for tag in item_tags:
                if tag in liked_styles:
                    boost *= 1.2
            item["score"] = item.get("fusion_score", 0.5) * boost
        items.sort(key=lambda x: x["score"], reverse=True)
        return items

    def _apply_inventory_and_freshness(self, items: List[Dict]) -> List[Dict]:
        # Placeholder – filter out out-of-stock and boost new arrivals
        return items

    def _maximal_marginal_relevance(self, query_embedding: np.ndarray, items: List[Dict],
                                    lambda_d: float = 0.5) -> List[Dict]:
        """
        MMR algorithm for diversity.
        """
        if not items:
            return []
        selected = []
        remaining = items.copy()
        # Precompute item embeddings
        for item in remaining:
            if "embedding" not in item:
                item["embedding"] = np.random.randn(self.config.MAX_EMBEDDING_DIM).astype(np.float32)
                item["embedding"] /= np.linalg.norm(item["embedding"])
        # Select first item
        first = max(remaining, key=lambda x: x["score"])
        selected.append(first)
        remaining.remove(first)
        while remaining and len(selected) < self.config.FINAL_TOP_K:
            mmr_scores = []
            for item in remaining:
                rel = item["score"]
                sim_max = max([self._cosine_sim(item["embedding"], sel["embedding"]) for sel in selected])
                mmr = lambda_d * rel - (1 - lambda_d) * sim_max
                mmr_scores.append(mmr)
            best_idx = np.argmax(mmr_scores)
            selected.append(remaining[best_idx])
            remaining.pop(best_idx)
        return selected

    def _cosine_sim(self, a: np.ndarray, b: np.ndarray) -> float:
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b) + 1e-10))

    async def health(self) -> Dict:
        return {"agent": "retrieval", "status": "healthy"}

"""

class OutfitAgent:
    def __init__(self, config: TariConfig, cache: PredictiveCache,
                 privacy_manager: PrivacyManager, pool: SecureConnectionPool):
        self.config = config
        self.cache = cache
        self.privacy_manager = privacy_manager
        self.pool = pool
        self.body_pipeline = CirvannaOptimizedPipeline(config, cache, privacy_manager, pool)

    async def compose_outfits(self, products: List[Dict], intent: Dict,
                              user_profile: Dict) -> List[Dict]:
        """
        Generate complete coordinated looks.
        """
        # Separate by category (inferred from metadata)
        categories = defaultdict(list)
        for p in products:
            cat = p.get("metadata", {}).get("category", "unknown")
            categories[cat].append(p)
        # Define category priority
        required_cats = ["top", "bottom", "footwear"]
        optional_cats = ["outerwear", "accessory", "bag"]
        # Generate outfit candidates
        outfits = []
        # For simplicity, take combinations of top, bottom, footwear
        tops = categories.get("top", [])[:5]
        bottoms = categories.get("bottom", [])[:5]
        footwear = categories.get("footwear", [])[:5]
        for top, bottom, shoe in itertools.product(tops, bottoms, footwear):
            outfit = {
                "outfit_id": f"outfit_{len(outfits)}",
                "items": [top, bottom, shoe],
                "compatibility_score": await self._compute_outfit_compatibility([top, bottom, shoe]),
                "aesthetic_profile": self._merge_aesthetics([top, bottom, shoe]),
                "color_palette": self._merge_colors([top, bottom, shoe]),
                "fit_scores": await self._compute_fit_scores([top, bottom, shoe], user_profile),
                "tension_tags": []
            }
            # Add optional items if available
            if categories.get("outerwear"):
                outer = categories["outerwear"][0]
                outfit["items"].append(outer)
                outfit["compatibility_score"] *= 0.9
            outfits.append(outfit)
        # Sort by compatibility
        outfits.sort(key=lambda x: x["compatibility_score"], reverse=True)
        # Apply constraints
        constrained = await self._apply_constraints(outfits, user_profile, intent)
        return constrained[:10]  # return top 10 outfits

    async def _compute_outfit_compatibility(self, items: List[Dict]) -> float:
        """Compute overall compatibility using pairwise scores."""
        if len(items) < 2:
            return 0.5
        total = 0.0
        count = 0
        for i in range(len(items)):
            for j in range(i+1, len(items)):
                emb_i = items[i].get("embedding")
                emb_j = items[j].get("embedding")
                if emb_i is not None and emb_j is not None:
                    sim = self._cosine_sim(emb_i, emb_j)
                else:
                    sim = 0.5
                total += sim
                count += 1
        return total / count if count > 0 else 0.5

    def _cosine_sim(self, a: np.ndarray, b: np.ndarray) -> float:
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b) + 1e-10))

    def _merge_aesthetics(self, items: List[Dict]) -> List[str]:
        all_aesthetics = []
        for item in items:
            tags = item.get("metadata", {}).get("styleTags", [])
            all_aesthetics.extend(tags)
        # Return most common
        from collections import Counter
        return [tag for tag, _ in Counter(all_aesthetics).most_common(3)]

    def _merge_colors(self, items: List[Dict]) -> List[str]:
        all_colors = []
        for item in items:
            colors = item.get("metadata", {}).get("colors", [])
            all_colors.extend(colors)
        return list(set(all_colors))[:3]

    async def _compute_fit_scores(self, items: List[Dict], user_profile: Dict) -> Dict:
        # Use body calibration to estimate fit per item
        body_meas = user_profile.get("body_measurements", {})
        scores = {}
        for item in items:
            # Simplified: use similarity between item fit tags and user fit preferences
            fit_pref = user_profile.get("fitPreference", "regular")
            scores[item.get("product_id")] = 0.8 if fit_pref == "regular" else 0.6
        return scores

    async def _apply_constraints(self, outfits: List[Dict], user_profile: Dict, intent: Dict) -> List[Dict]:
        """
        Apply constraint engine: category uniqueness, color coherence, layering rules, etc.
        """
        valid = []
        for outfit in outfits:
            # Category uniqueness (no duplicate categories)
            categories = [item.get("metadata", {}).get("category") for item in outfit["items"]]
            if len(set(categories)) != len(categories):
                continue
            # Color coherence (simple: at least two items share a color)
            all_colors = self._merge_colors(outfit["items"])
            if len(all_colors) < 1:
                continue
            # Layering rule: if outerwear, must have top and bottom
            has_outer = any(cat == "outerwear" for cat in categories)
            has_top = any(cat == "top" for cat in categories)
            has_bottom = any(cat == "bottom" for cat in categories)
            if has_outer and (not has_top or not has_bottom):
                continue
            # Seasonal rule (simplified)
            season = intent.get("season", "spring")
            if season == "winter" and not any(cat == "outerwear" for cat in categories):
                continue
            valid.append(outfit)
        return valid

    async def generate_variations(self, outfits: List[Dict], intent: Dict) -> List[Dict]:
        """
        Generate slight variations of a selected look (darker, luxury, casual, avant-garde, warmer).
        """

            variant["variation_type"] = "luxury"
            variant["compatibility_score"] *= 0.95

                 privacy_manager: PrivacyManager, pool: SecureConnectionPool):
        self.config = config
        self.cache = cache
        self.privacy_manager = privacy_manager
        self.pool = pool
        self._job_queue = asyncio.Queue()
        self._workers = []
        self._render_cache = {}

    async def start(self):
        for _ in range(self.config.TRYON_GPU_WORKERS):
            worker = asyncio.create_task(self._worker_loop())
            self._workers.append(worker)

    async def queue_tryon(self, user_id: str, outfit: Dict, context: Dict) -> Dict:
        """Submit a try-on job to the async queue."""
        job_id = str(uuid.uuid4())
        resolution = self.config.FREE_TIER_TRYON_RESOLUTION
        if context.get("tier") == "premium":
            resolution = self.config.PREMIUM_TIER_TRYON_RESOLUTION
        job = {
            "job_id": job_id,
            "user_id": user_id,
            "outfit": outfit,
            "context": context,
            "resolution": resolution,
            "status": "queued",
            "created_at": time.time()
        }
        await self._job_queue.put(job)
        await self._store_job(job)
        return {"job_id": job_id, "status": "queued", "estimated_time_ms": self.config.MAX_TRYON_QUEUE_DELAY_MS}

    async def _worker_loop(self):
        while True:
            try:
                job = await self._job_queue.get()
                await self._process_job(job)
            except asyncio.CancelledError:
                break
            except Exception as e:
                job["status"] = "failed"
                job["error"] = str(e)
                await self._update_job(job)

    async def _process_job(self, job: Dict):
        job["status"] = "rendering"
        await self._update_job(job)
        # Simulate rendering (place – call actual try-on model)
        await asyncio.sleep(2)

        job["status"] = "completed"
        job["render_url"] = render_url
        job["completed_at"] = time.time()
        await self._update_job(job)
        # Cache result
        self._render_cache[job["job_id"]] = job

    async def get_result(self, job_id: str) -> Optional[Dict]:
        if job_id in self._render_cache:
            return self._render_cache[job_id]
        # Check storage
        return await self._fetch_job(job_id)

    async def _store_job(self, job: Dict):

            "silhouettePreferences": preferences.get("silhouette_preferences", []),
            "recentInteractions": behavioral[-10:],
            "semanticStyle": semantic.get("embedding", []),
            "session": session
        }
        return profile

    async def update_memory(self, user_id: str, interaction: Dict):
        """Update memory tiers based on interaction."""
        anonymized = self.privacy_manager.anonymize_user_id(user_id)
        # Session memory (short-term)
        self._session_memories[anonymized].update(interaction)
        # Preference memory (if explicit like/dislike)
        if "liked_style" in interaction:
            prefs = await self._get_preferences(anonymized)
            prefs["liked_styles"] = prefs.get("liked_styles", []) + [interaction["liked_style"]]
            await self._store_preferences(anonymized, prefs)
        # Behavioral memory (append)
        self._behavioral[anonymized].append({
            "timestamp": time.time(),
            "type": interaction.get("type"),
            "product_id": interaction.get("product_id")
        })
        if len(self._behavioral[anonymized]) > 100:
            self._behavioral[anonymized] = self._behavioral[anonymized][-100:]
        # Semantic memory (update embedding with temporal decay)
        if "style_embedding" in interaction:
            emb = interaction["style_embedding"]
            current = await self._get_semantic(anonymized)
            old_emb = current.get("embedding", [0]*384)
            # Exponential moving average with decay
            alpha = 0.3
            new_emb = [alpha * e + (1-alpha) * o for e, o in zip(emb, old_emb)]
            await self._store_semantic(anonymized, {"embedding": new_emb})

"""
Offline evaluation framework for retrieval and outfit quality.
"""
import numpy as np
from typing import List, Dict, Any
from ...config import TariConfig

class OfflineEvaluator:
    def __init__(self, config: TariConfig):
        self.config = config

    def compute_recall_at_k(self, retrieved_ids: List[str], relevant_ids: List[str], k: int) -> float:
        retrieved_k = retrieved_ids[:k]
        intersection = set(retrieved_k) & set(relevant_ids)
        return len(intersection) / len(relevant_ids) if relevant_ids else 0.0

    def compute_ndcg_at_k(self, retrieved_ids: List[str], relevance_scores: Dict[str, float], k: int) -> float:
        dcg = 0.0
        for i, pid in enumerate(retrieved_ids[:k]):
            rel = relevance_scores.get(pid, 0.0)
            dcg += rel / np.log2(i + 2)
        # Ideal DCG
        sorted_relevances = sorted(relevance_scores.values(), reverse=True)[:k]
        idcg = sum([rel / np.log2(i + 2) for i, rel in enumerate(sorted_relevances)])
        return dcg / idcg if idcg > 0 else 0.0

    def compute_mrr(self, retrieved_ids: List[str], relevant_ids: List[str]) -> float:
        for i, pid in enumerate(retrieved_ids):
            if pid in relevant_ids:
                return 1.0 / (i + 1)
        return 0.0

@asynccontextmanager
async def lifespan(app: FastAPI):
    config = TariConfig()
    pool = SecureConnectionPool.get_instance(config)
    try:
        await pool.initialize()
        # Initialize gateway and start try-on workers
        cache = PredictiveCache(config)

@app.middleware("http")
async def deduplication_middleware(request, call_next):
    if request.method != "POST":
        return await call_next(request)
    import hashlib
    body = await request.body()
    fingerprint = hashlib.sha256(
        f"{request.url.path}:{body}".encode()
    ).hexdigest()
    cache_key = f"req_dedup:{fingerprint}"
    pool = await SecureConnectionPool.get_instance()
    async with pool.get_redis() as redis_pool:
        cached = await redis_pool.get(cache_key)
        if cached:
            return JSONResponse(orjson.loads(cached))
        response = await call_next(request)
        if response.status_code == 200:
            await redis_pool.setex(
                cache_key,
                2,
                orjson.dumps({
                    "status": response.status_code,
                    "body": response.body.decode() if hasattr(response.body, 'decode') else str(response.body)
                })
            )
        return response
